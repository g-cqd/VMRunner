  /// The CRC-24 of RFC 4880 section 6.1, a byte at a time through `crc24Table`.
  static func crc24(_ bytes: [UInt8]) -> UInt32 {
      var crc: UInt32 = 0xB7_04CE
      for byte in bytes {
          crc = ((crc << 8) & 0xFF_FFFF) ^ crc24Table[Int(((crc >> 16) ^ UInt32(byte)) & 0xFF)]
