```swift
import Foundation
import GuestSeed

/// What the editor shows about a provisioning spec: one line per step the next start would pack, or why it would refuse.
/// A preview only: the start reads the spec and its scripts again and decides (`ProvisioningSeed.prepare`).
enum ProvisioningCheck: Equatable, Sendable {
    case ready([String])
    case refused(String)

    init(specAt path: String) {
        do {
            let spec = try ProvisioningSeed.compile(specAt: path, instanceID: "check").spec
            let steps = spec.steps.map { Self.line(for: $0) }
            self = .ready(spec.cloudInit.userData == nil ? steps : steps + ["Your own cloud-init data, merged with the steps"])
        } catch let error as ProvisioningSeedError {
            self = .refused(error.reason)
        } catch {
            self = .refused(error.localizedDescription)
        }
    }

    /// "line 7: first-boot, inline script", or "line 11: every-boot, scripts/hello.sh, sha256 9fac65956eed…".
    static func line(for step: ProvisioningStep) -> String {
        switch step.script {
        case .inline: "line \(step.line): \(step.phase.rawValue), inline script"
        case .file(let path, let sha256): "line \(step.line): \(step.phase.rawValue), \(path), sha256 \(sha256.prefix(12))…"
        }
    }
}
```

2. Create `Sources/VMRunner/Views/Editor/ProvisioningSection.swift` with exactly:

```swift
import SwiftUI

/// A Linux machine's provisioning spec (ADR-0012, experimental): the file, what a check of it finds, and the buttons to
/// choose, check or remove it. The check is a preview; the start reads the spec and its scripts again and decides.
struct ProvisioningSection: View {
    let specPath: String
    let installerAttached: Bool
    let choose: () -> Void
    let remove: () -> Void
    @State private var check: ProvisioningCheck?

    var body: some View {
        Section {
            if specPath.isEmpty {
                Text("No spec: the machine starts without a seed.").font(.caption).foregroundStyle(.secondary)
            } else {
                LabeledContent("Spec", value: URL(filePath: specPath).lastPathComponent)
                    .help(specPath)
                checkResult
                if installerAttached {
                    Text("Not used while the installer is attached: detach the installer to use it.")
                        .font(.caption).foregroundStyle(.secondary)
                }
            }
            HStack {
                Button(specPath.isEmpty ? "Choose Spec…" : "Change Spec…", action: choose)
                    .managedLock(.allowProvisioning)
                    .settingHelp(.provisioning)
                if !specPath.isEmpty {
                    Button("Check Again") { Task { await recheck() } }
                    Button("Remove", role: .destructive, action: remove)
                }
            }
        } header: {
            HelpHeader("Provisioning (experimental)", topic: .provisioning)
        }
        .task(id: specPath) { await recheck() }
    }

    @ViewBuilder private var checkResult: some View {
        switch check {
        case .ready(let lines)?:
            if lines.isEmpty {
                Text("No steps.").font(.caption).foregroundStyle(.secondary)
            }
            ForEach(Array(lines.enumerated()), id: \.offset) { _, line in
                Text(line).font(.caption.monospaced())
            }
        case .refused(let reason)?:
            Label(reason, systemImage: "exclamationmark.triangle").font(.caption).foregroundStyle(.red)
        case nil:
            ProgressView().controlSize(.small)
        }
    }

    /// Reads the spec and its scripts away from the main thread; the result replaces what is shown.
    private func recheck() async {
        guard !specPath.isEmpty else {
            check = nil
            return
        }
        let path = specPath
        check = nil
        let result = await Task.detached(priority: .utility) { ProvisioningCheck(specAt: path) }.value
        // A newer path cancelled this task while the check ran: its result is for a spec that is no longer shown.
        guard !Task.isCancelled else { return }
        check = result
    }
}
```
