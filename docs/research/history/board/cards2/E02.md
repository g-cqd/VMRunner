     ```swift
     import NetPolicy

     /// What the kill switch reads from the Mac's network path. A plain value, so the rules can be tested without a network.
     struct HostNetworkPath: Equatable, Sendable {
         /// Whether a VPN is up on the Mac, classified as the Mac VPN indicator does (`VPNState.classify`).
         var vpn = VPNState.unknown
         /// macOS marks the path as expensive: a phone's hotspot or a cellular link.
         var isExpensive = false
         /// Low Data Mode is on for the path.
         var isConstrained = false
     }

     /// Why the kill switch has disconnected the guest's network card.
     enum NetworkCutoffReason: Equatable, Sendable {
         /// The machine needs a VPN on the Mac, and none is up (or none is known to be up yet).
         case vpnRequired
         /// The machine must not use the Mac's VPN, and one is up (or the state is not known yet).
         case vpnForbidden
         /// The Mac's path is metered or in Low Data Mode, and the machine's switch says to cut then.
         case meteredPath
     }

     extension VMConfig {
         /// Whether the kill switch has a rule to watch for this machine: a network card, and a VPN rule or the metered switch.
         var hasNetworkCutoff: Bool {
             networkMode.hasNetworkDevice && (networkPolicy.vpn != .ignore || networkCutsOnMeteredPath)
         }

         /// Why the network card must be disconnected on `path`, or nil when it may stay connected. The VPN rule comes first,
         /// and an unknown VPN state breaks both VPN rules (`VPNRequirement.allowsNetwork(vpn:)`).
         func networkCutoffReason(on path: HostNetworkPath) -> NetworkCutoffReason? {
             guard networkMode.hasNetworkDevice else { return nil }
             let vpnRule = networkPolicy.vpn
             if !vpnRule.allowsNetwork(vpn: path.vpn) { return vpnRule == .required ? .vpnRequired : .vpnForbidden }
             return networkCutsOnMeteredPath && (path.isExpensive || path.isConstrained) ? .meteredPath : nil
         }
     }
     ```

     ```swift
     import NetPolicy
     import Network
     import Observation

     /// The kill switch of one run (roadmap 0010): it follows the Mac's network path and disconnects the guest's network card
     /// while the machine's rules say so (`VMConfig.networkCutoffReason(on:)`), then connects it again.
     ///
     /// It watches the Mac, not the guest: a VPN that is up on the Mac usually carries the guest's traffic, but split
     /// tunnelling can make that wrong either way (see `VPNMonitor`). Before the first sample the VPN state is unknown, so a
     /// VPN rule cuts the card from the start. It stops when it is released, with the machine that owns it.
     @MainActor @Observable
     final class NetworkCutoff {
         /// Produces a fresh stream of path samples when the watch starts.
         typealias Source = @Sendable () -> any AsyncSequence<HostNetworkPath, Never>

         /// Why the card is disconnected now; nil while it is connected.
         private(set) var reason: NetworkCutoffReason?

         @ObservationIgnored private let config: VMConfig
         @ObservationIgnored private let source: Source
         @ObservationIgnored private var connect: ((Bool) -> Void)?
         @ObservationIgnored private var task: Task<Void, Never>?

         init(config: VMConfig, source: @escaping Source = HostNetworkPath.systemPaths) {
             self.config = config
             self.source = source
         }

         isolated deinit {
             task?.cancel()
         }

         /// Applies the rules to an unknown path at once, then to every sample. `connect(false)` disconnects the card and
         /// `connect(true)` connects it again; each is called only when the answer changes. Idempotent.
         func start(connect: @escaping (Bool) -> Void) {
             guard task == nil else { return }
             self.connect = connect
             apply(HostNetworkPath())
             let makePaths = source
             task = Task { [weak self] in
                 for await path in makePaths() {
                     guard let self, !Task.isCancelled else { return }
                     self.apply(path)
                 }
             }
         }

         private func apply(_ path: HostNetworkPath) {
             let next = config.networkCutoffReason(on: path)
             guard next != reason else { return }
             let changesConnection = (next == nil) != (reason == nil)
             reason = next
             if changesConnection { connect?(next == nil) }
         }
     }

     extension HostNetworkPath {
         /// Samples of the Mac's network path from `NWPathMonitor`, a fresh stream per call: the VPN state of the path's usable
         /// interfaces (read as `VPNMonitor` reads them), and whether the path is expensive or constrained.
         static let systemPaths: @Sendable () -> any AsyncSequence<HostNetworkPath, Never> = {
             NWPathMonitor().map { path in
                 let up = path.status == .satisfied
                 let interfaces = path.availableInterfaces.map { InterfaceSnapshot(name: $0.name, isUp: up, addresses: nil) }
                 return HostNetworkPath(
                     vpn: VPNState.classify(interfaces), isExpensive: path.isExpensive, isConstrained: path.isConstrained)
             }
         }
     }
     ```

       ```swift
           /// Starts the kill switch once the framework has the machine, so its network card can be reached.
           private func startNetworkCutoff() {
               networkCutoff?.start { [weak self] connected in self?.setNetworkConnected(connected) }
           }

           /// Takes the attachment off the network card (`false`) or puts the one it had back (`true`), like pulling and
           /// replugging a cable. A change the framework refuses is reported to the delegate and leaves the card disconnected.
           private func setNetworkConnected(_ connected: Bool) {
               guard let device = machine.networkDevices.first else { return }
               if connected {
                   guard let attachment = detachedAttachment else { return }
                   device.attachment = attachment
                   detachedAttachment = nil
               } else if let attachment = device.attachment {
                   detachedAttachment = attachment
                   device.attachment = nil
               }
           }
       ```
