public struct RunnerArguments: Equatable, Sendable {
    /// How the container is attached to a network.
    public enum Network: String, Equatable, Sendable {
        /// No network device: the baseline run (pull, boot, `uname -a`, exit status).
        case none
        /// A network device whose frames go to a socketpair read by the runner itself.
        case loopback
    }

    public static let defaultImage = "docker.io/library/alpine:3.24.2"
    public static let defaultCPUs = 2
    public static let defaultMemoryMiB = 512
    public static let cpuRange = 1...16
    public static let memoryRange = 128...16384
    static let maximumImageBytes = 255
    static let maximumCommandArguments = 64
    static let maximumCommandArgumentBytes = 4096
    static let maximumEnvironmentVariables = 64

    public static let usage = """
        usage: ContainerRunner <scratch-directory> [--network none|loopback] [--image <reference>] [--command <argument>]...
                               [--cpus <n>] [--memory <MiB>] [--env <NAME=value>]... [--json]

        Every file the runner creates (kernel, image store, container root) lives under <scratch-directory>.
        --network none      pull the image, boot it, run the command, print the exit status (default)
        --network loopback  the same, with a network device backed by a socketpair instead of a network; the container
                            pings its gateway and the runner reports the frames the device delivered
        --image             the image to run (default \(defaultImage))
        --command           one argument of the command to run, without a shell; repeat it for each argument. Without it the
                            runner runs `uname -a` and expects the architecture in the output; with it the run passes when
                            the command exits with status 0
        --cpus              virtual processors, \(cpuRange.lowerBound) to \(cpuRange.upperBound) (default \(defaultCPUs))
        --memory            memory in MiB, \(memoryRange.lowerBound) to \(memoryRange.upperBound) (default \(defaultMemoryMiB))
        --env               an environment variable NAME=value for the command; repeat it, once per name
        --json              print one JSON event per line on standard output instead of the report (phase, output, exited,
                            error, finished) and stop the container on a `stop` line or the end of standard input (status 75)
        """

    /// The absolute, standardized scratch directory.
    public let scratch: URL
    public let network: Network
    /// The image reference to pull and run.
    public let image: String
    /// The command, one element per argument; empty runs the baseline `uname -a`.
    public let command: [String]
    public let cpus: Int
    public let memoryMiB: Int
    /// `NAME=value` entries, validated, each name once.
    public let environment: [String]
    /// Whether to print events instead of the report, and to watch standard input for a stop.
    public let json: Bool

    /// Reads `arguments` (the program name already removed). A relative scratch directory is taken from `currentDirectory`.
    ///
    /// - Throws: ``RunnerArgumentError``.
    public static func parse(_ arguments: [String], currentDirectory: URL) throws(RunnerArgumentError) -> RunnerRequest {
        var scratch: String?
        var network = Network.none
        var image = defaultImage
        var command: [String] = []
        var cpus = defaultCPUs
        var memoryMiB = defaultMemoryMiB
        var environment: [String] = []
        var json = false
        var index = 0
        /// The value after `option`; a value may start with a dash (`--command -c`).
        func value(after option: String) throws(RunnerArgumentError) -> String {
            guard index < arguments.count else { throw .missingValue(option: option) }
            index += 1
            return arguments[index - 1]
        }
        while index < arguments.count {
            let argument = arguments[index]
            index += 1
            switch argument {
            case "-h", "--help":
                return .help
            case "--network":
                let name = try value(after: argument)
                guard let chosen = Network(rawValue: name) else { throw .unknownNetwork(name) }
                network = chosen
            case "--image":
                image = try validImage(try value(after: argument))
            case "--command":
                try appendCommandArgument(try value(after: argument), to: &command)
            case "--cpus":
                let text = try value(after: argument)
                guard let count = integer(text, in: cpuRange) else { throw .invalidCPUs(text) }
                cpus = count
            case "--memory":
                let text = try value(after: argument)
                guard let mebibytes = integer(text, in: memoryRange) else { throw .invalidMemory(text) }
                memoryMiB = mebibytes
            case "--env":
                try appendEnvironment(try value(after: argument), to: &environment)
            case "--json":
                json = true
            case _ where argument.hasPrefix("-"):
                throw .unknownOption(argument)
            default:
                guard scratch == nil else { throw .extraArgument(argument) }
                scratch = argument
            }
        }
        guard let scratch, !scratch.isEmpty else { throw .missingScratchDirectory }
        let directory = URL(fileURLWithPath: scratch, isDirectory: true, relativeTo: currentDirectory).standardizedFileURL
        return .run(
            Self(
                scratch: directory, network: network, image: image, command: command, cpus: cpus, memoryMiB: memoryMiB,
                environment: environment, json: json))
    }

    /// The name of a `NAME=value` entry: everything before the first `=`, or all of it when there is none.
    static func name(of entry: String) -> Substring {
        entry.prefix { $0 != "=" }
    }

    /// `base` with each entry of the environment applied: an entry replaces the one of the same name, or is added at the end.
    public func environment(merging base: [String]) -> [String] {
        var merged = base
        for entry in environment {
            let name = Self.name(of: entry)
            if let position = merged.firstIndex(where: { Self.name(of: $0) == name }) {
                merged[position] = entry
            } else {
                merged.append(entry)
            }
        }
        return merged
    }

    private static func validImage(_ reference: String) throws(RunnerArgumentError) -> String {
        let forbidden = reference.unicodeScalars.contains { $0.properties.isWhitespace || $0.properties.generalCategory == .control }
        guard !reference.isEmpty, reference.utf8.count <= maximumImageBytes, !forbidden else { throw .invalidImage }
        return reference
    }

    private static func appendCommandArgument(_ argument: String, to command: inout [String]) throws(RunnerArgumentError) {
        guard command.count < maximumCommandArguments else { throw .tooManyCommandArguments }
        guard argument.utf8.count <= maximumCommandArgumentBytes, !argument.utf8.contains(0) else {
            throw .invalidCommandArgument(position: command.count + 1)
        }
        command.append(argument)
    }

    private static func appendEnvironment(_ entry: String, to environment: inout [String]) throws(RunnerArgumentError) {
        let key = name(of: entry)
        let value = entry.dropFirst(key.count + 1)
        let wellFormed = entry.utf8.count > key.utf8.count && isName(key) && !value.utf8.contains(0) && !value.utf8.contains(10)
        guard wellFormed else { throw .invalidEnvironment(name: String(key.prefix(64))) }
        guard environment.count < maximumEnvironmentVariables else { throw .tooManyEnvironmentVariables }
        guard !environment.contains(where: { name(of: $0) == key }) else { throw .duplicateEnvironment(name: String(key)) }
        environment.append(entry)
    }

    /// Letters, digits and underscores, not starting with a digit.
    private static func isName(_ name: Substring) -> Bool {
        guard let first = name.utf8.first, first == 95 || isLetter(first) else { return false }
        return name.utf8.allSatisfy { $0 == 95 || isLetter($0) || isDigit($0) }
    }

    private static func isLetter(_ byte: UInt8) -> Bool { (65...90).contains(byte) || (97...122).contains(byte) }
    private static func isDigit(_ byte: UInt8) -> Bool { (48...57).contains(byte) }

    /// `text` as a whole number inside `range`; digits only, so no sign, no spaces.
    private static func integer(_ text: String, in range: ClosedRange<Int>) -> Int? {
        guard text.utf8.count <= 6, !text.isEmpty, text.utf8.allSatisfy(isDigit), let number = Int(text), range.contains(number) else {
            return nil
        }
        return number
    }
}

public import Foundation

/// A line the runner writes to standard output in `--json` mode: one JSON object per line, in the order things happened.
///
/// `time` is seconds on the monotonic clock since the runner started, so a phase can also be read as the difference
/// between two events and does not depend on the launcher's clock.
public struct RunnerEvent: Codable, Equatable, Sendable {
    public enum Kind: String, Codable, Sendable {
        /// One step of the run finished; `name` says which (`kernel`, `init-image`, `pull`, `create`, `boot`, `run`) and
        /// `seconds` how long it took.
        case phase
        /// A chunk of what the container wrote; `stream` says where and `text` is the chunk, decoded leniently.
        case output
        /// The container's process ended; `status` is its exit status.
        case exited
        /// Something failed around the container; `message` says what.
        case error
        /// The last line: `status` is the runner's own exit status.
        case finished
    }

    /// Which output stream of the container a chunk came from.
    public enum Stream: String, Codable, Sendable {
        case stdout
        case stderr
    }

    /// The most bytes one `output` event carries; a longer write is split.
    public static let maximumChunkBytes = 16384

    public var event: Kind
    public var time: Double
    public var name: String?
    public var seconds: Double?
    public var stream: Stream?
    public var text: String?
    public var status: Int32?
    public var message: String?

    public init(
        _ event: Kind, time: Double = 0, name: String? = nil, seconds: Double? = nil, stream: Stream? = nil, text: String? = nil,
        status: Int32? = nil, message: String? = nil
    ) {
        self.event = event
        self.time = time
        self.name = name
        self.seconds = seconds
        self.stream = stream
        self.text = text
        self.status = status
        self.message = message
    }

    /// The JSON object as one line, without the newline.
    public func line() throws -> String {
        let encoder = JSONEncoder()
        encoder.outputFormatting = [.sortedKeys, .withoutEscapingSlashes]
        let encoded = try encoder.encode(self)
        guard let line = String(bytes: encoded, encoding: .utf8) else {
            throw EncodingError.invalidValue(self, .init(codingPath: [], debugDescription: "an event is not valid UTF-8"))
        }
        return line
    }

    /// `duration` as fractional seconds.
    public static func seconds(of duration: Duration) -> Double {
        let parts = duration.components
        return Double(parts.seconds) + Double(parts.attoseconds) / 1e18
    }

    /// `data` cut into pieces of at most ``maximumChunkBytes`` bytes, in order. Empty data gives no piece.
    public static func chunks(of data: Data) -> [Data] {
        stride(from: 0, to: data.count, by: maximumChunkBytes).map { offset in
            let start = data.startIndex + offset
            return data.subdata(in: start..<min(start + maximumChunkBytes, data.endIndex))
        }
    }
}

/// Finds the `stop` line the launcher writes to the runner's standard input, in whatever pieces the bytes arrive.
public struct StopLineDetector: Sendable {
    /// A line longer than this cannot be `stop`; it is dropped instead of kept, so the buffer stays small.
    static let maximumLineBytes = 64

    private var pending = Data()
    /// Whether the rest of the current line is being skipped because it was too long.
    private var skipping = false

    public init() {}

    /// Adds `data` and tells whether a complete line of it was `stop` (a trailing carriage return and spaces are ignored).
    public mutating func feed(_ data: Data) -> Bool {
        pending.append(data)
        while let newline = pending.firstIndex(of: 10) {
            let line = pending[pending.startIndex..<newline]
            let text = String(bytes: line, encoding: .utf8)?.trimmingCharacters(in: .whitespacesAndNewlines)
            let isStop = !skipping && text == "stop"
            skipping = false
            pending = Data(pending[(newline + 1)...])
            if isStop { return true }
        }
        if pending.count > Self.maximumLineBytes {
            pending = Data()
            skipping = true
        }
        return false
    }
}

import ContainerRunnerCore
import Containerization
import Foundation
import Synchronization

/// Writes the runner's events to standard output, one JSON line each, and keeps the clock they are stamped with.
///
/// Standard error belongs to diagnostics, so a launcher can read events from standard output without filtering.
final class EventEmitter: Sendable {
    private let start = ContinuousClock.now
    /// Held while a line is written and stamped, so lines from the container's streams never interleave and their times
    /// never run backwards.
    private let lock = Mutex(())

    /// Stamps `event` with the time since the runner started and writes it.
    func send(_ event: RunnerEvent) {
        lock.withLock { _ in
            var stamped = event
            stamped.time = RunnerEvent.seconds(of: ContinuousClock.now - start)
            guard let line = try? stamped.line() else { return }
            try? FileHandle.standardOutput.write(contentsOf: Data((line + "\n").utf8))
        }
    }
}

/// A request to stop the container, from a `stop` line or the end of standard input.
///
/// While the run is still being set up (kernel, image, boot) the request ends the runner at once; once the container runs,
/// ``requested()`` returns so the flow can stop the container and end normally.
final class StopSignal: Sendable {
    private struct State {
        var requested = false
        var settingUp = true
    }

    private let state = Mutex(State())
    private let stream: AsyncStream<Void>
    private let continuation: AsyncStream<Void>.Continuation
    private let whileSettingUp: @Sendable () -> Void

    /// `whileSettingUp` runs on a request that arrives before ``settingUpDone()``; it is expected not to return.
    init(whileSettingUp: @escaping @Sendable () -> Void) {
        (stream, continuation) = AsyncStream.makeStream()
        self.whileSettingUp = whileSettingUp
    }

    var isRequested: Bool { state.withLock { $0.requested } }

    /// The container runs: from now on a request is answered by ``requested()``.
    func settingUpDone() {
        state.withLock { $0.settingUp = false }
    }

    func request() {
        let early = state.withLock { state in
            state.requested = true
            return state.settingUp
        }
        if early { whileSettingUp() }
        continuation.finish()
    }

    /// Returns once a stop was requested, or when the calling task is cancelled. Only one task may wait.
    func requested() async {
        for await _ in stream {}
    }
}

/// Watches standard input for a `stop` line or its end, and asks `signal` to stop. The launcher closing the pipe, or dying,
/// ends the container the same way.
final class StdinWatcher: Sendable {
    private let detector = Mutex(StopLineDetector())
    private let signal: StopSignal

    init(signal: StopSignal) {
        self.signal = signal
    }

    func start() {
        FileHandle.standardInput.readabilityHandler = { [self] handle in
            let data = handle.availableData
            guard data.isEmpty || detector.withLock({ $0.feed(data) }) else { return }
            handle.readabilityHandler = nil
            signal.request()
        }
    }
}

/// Collects what the container writes, keeping at most `limit` bytes: the verdict needs the start of the output, not all of it.
final class OutputCapture: Sendable {
    private let bytes = Mutex(Data())
    private let limit: Int

    init(limit: Int = .max) {
        self.limit = limit
    }

    /// What was kept, decoded leniently: a limit may cut a character in two.
    var text: String { bytes.withLock { String(decoding: $0, as: UTF8.self) } }  // swiftlint:disable:this optional_data_string_conversion

    func append(_ data: Data) {
        bytes.withLock { $0.append(data.prefix(max(0, limit - $0.count))) }
    }
}

/// Receives what the container writes to one stream: it goes into the capture and, with an emitter, out as `output` events.
struct StreamWriter: Writer {
    let capture: OutputCapture
    let stream: RunnerEvent.Stream
    let emitter: EventEmitter?

    func write(_ data: Data) throws {
        capture.append(data)
        for chunk in RunnerEvent.chunks(of: data) {
            let text = String(decoding: chunk, as: UTF8.self)  // swiftlint:disable:this optional_data_string_conversion
            emitter?.send(RunnerEvent(.output, stream: stream, text: text))
        }
    }

    func close() throws {}
}

import Containerization

import Foundation

/// The run flow of the package's sample command-line tool, in the order that tool uses: a kernel, a container manager that
/// pulls the init image, a pull of the image, create, start, wait, stop. It runs the requested command, `uname -a` unless
/// one was given, and records what each step cost. With a loopback network the baseline container also pings its gateway,
/// and the frames its network device delivered are reported. With an emitter every phase and every chunk of output is also
/// sent as an event, and a stop request ends the wait by stopping the container.
struct ContainerFlow {
    /// The init image, matched to the pinned package release: the guest and the host speak a versioned protocol.
    static let initReference = "ghcr.io/apple/containerization/vminit:0.52.1"
    /// How much of the container's output the runner keeps for the verdict when it streams events.
    static let capturedBytes = 64 * 1024

    let layout: RunnerLayout
    let arguments: RunnerArguments
    let events: EventEmitter?
    let stop: StopSignal

    private var network: RunnerArguments.Network { arguments.network }

    /// The container's command: the one asked for, else the baseline. The baseline ping has no reply to get, so the shell
    /// ends with status 0 regardless; the frames the ping caused are the result, not the ping's own status.
    private var command: [String] {
        guard arguments.command.isEmpty else { return arguments.command }
        switch network {
        case .none: return ["uname", "-a"]
        case .loopback: return ["/bin/sh", "-c", "uname -a; ping -c 1 -W 2 \(NetworkVerdict.gatewayAddress) >/dev/null 2>&1; exit 0"]
        }
    }

    /// Appends each completed phase to `report`, so a failure half way still tells how far the run got.
    func run(kernel: URL, report: inout RunnerReport) async throws {
        guard network == .loopback else {
            try await execute(kernel: kernel, link: nil, report: &report)
            return
        }
        let link = try LoopbackLink()
        let collector = Task {
            var seen: [FrameSummary] = []
            for await bytes in link.frames {
                if let frame = FrameSummary.parse(bytes) { seen.append(frame) }
            }
            return seen
        }
        var failure: (any Error)?
        do {
            try await execute(kernel: kernel, link: link, report: &report)
        } catch {
            failure = error
        }
        link.stop()
        report.frames = await collector.value
        report.networkVerdict = NetworkVerdict.evaluate(frames: report.frames)
        if let failure { throw failure }
    }

    private func execute(kernel: URL, link: LoopbackLink?, report: inout RunnerReport) async throws {
        let clock = ContinuousClock()
        var started = clock.now
        func finishPhase(_ name: String) {
            let duration = started.duration(to: clock.now)
            report.phases.append(.init(name: name, duration: duration))
            events?.send(RunnerEvent(.phase, name: name, seconds: RunnerEvent.seconds(of: duration)))
            started = clock.now
        }

        var manager = try await ContainerManager(
            kernel: Kernel(path: kernel, platform: .linuxArm),
            initfsReference: Self.initReference,
            root: layout.imageStore,
            network: link.map { LoopbackNetwork(link: $0) }
        )
        finishPhase("init-image")

        let image = try await manager.imageStore.pull(reference: arguments.image, platform: Platform(from: "linux/arm64"))
        finishPhase("pull")

        let id = "runner-\(UUID().uuidString.prefix(8).lowercased())"
        let capture = OutputCapture(limit: events == nil ? .max : Self.capturedBytes)
        let memory = UInt64(arguments.memoryMiB) << 20
        let command = command
        let container = try await manager.create(
            id, image: image, rootfsSizeInBytes: UInt64(1) << 30, networking: link != nil,
            vm: VMResources(cpus: arguments.cpus, memoryInBytes: memory + VMResources.guestMemoryOverhead)
        ) { config in
            config.cpus = arguments.cpus
            config.memoryInBytes = memory
            config.process.arguments = command
            config.process.environmentVariables = arguments.environment(merging: config.process.environmentVariables)
            config.process.stdout = StreamWriter(capture: capture, stream: .stdout, emitter: events)
            config.process.stderr = StreamWriter(capture: capture, stream: .stderr, emitter: events)
        }
        finishPhase("create")

        do {
            try await container.create()
            finishPhase("boot")
            try await container.start()
            stop.settingUpDone()
            let status = try await waitForExit(of: container)
            finishPhase("run")
            try await container.stop()
            report.stopped = stop.isRequested
            if let status {
                events?.send(RunnerEvent(.exited, status: status.exitCode))
                report.output = capture.text
                report.exitCode = status.exitCode
                report.verdict = RunnerVerdict.evaluate(
                    exitCode: status.exitCode, output: capture.text, checksArchitecture: arguments.command.isEmpty)
            }
        } catch {
            try? await container.stop()
            Console.line("boot log kept at \(layout.imageStore.path)/containers/\(id)/bootlog.log", to: .error)
            throw error
        }
        do {
            try manager.delete(id)
        } catch {
            Console.line("warning: could not remove container \(id): \(error)", to: .error)
        }
    }

    /// Waits for the container's process. A stop request stops the container meanwhile, so the wait then ends with the
    /// status the process left, or with nil when it left none.
    private func waitForExit(of container: LinuxContainer) async throws -> ExitStatus? {
        let stop = stop
        return try await withThrowingTaskGroup(of: ExitStatus?.self) { group in
            group.addTask {
                do {
                    return try await container.wait()
                } catch  where stop.isRequested {
                    return nil
                }
            }
            group.addTask {
                await stop.requested()
                if !Task.isCancelled { try? await container.stop() }
                return nil
            }
            defer { group.cancelAll() }
            for try await status in group where status != nil { return status }
            return nil
        }
    }
}
