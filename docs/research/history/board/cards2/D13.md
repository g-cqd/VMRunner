Anchors: locate every edit by its text (`grep -n`), never by a line number. Verified on `main` at `73bceb0`.

Why (read from the code): `SystemNameResolver.lookup` starts a new thread per lookup (`Thread.detachNewThread`), at most 32 at
once, and answers SERVFAIL at once to the 33rd. A burst therefore creates and destroys a thread per lookup and fails lookups that
a moment's wait would serve. The repository bans `DispatchQueue` and lock types other than `Mutex` (CONTRIBUTING, "Code rules"),
and a parked worker would need a condition variable, so the pool is built without idle threads: at most 32 workers exist; each
one, once its lookup is done, takes the next queued lookup, and ends when none is left. Up to 256 lookups wait for a worker
(the DNS proxy itself keeps at most 256 pending); past that a lookup is answered SERVFAIL at once, as before. The admission
logic is a plain value type (`ResolverQueue`) tested deterministically; the thread part is tested end to end.

```swift
import Foundation
import Synchronization

/// Who runs a blocking lookup, decided under ``ResolverWorkers``'s lock: a new worker while fewer than `maximumWorkers` run, a
/// running worker later while fewer than `maximumQueued` wait, nobody otherwise.
struct ResolverQueue<Job> {
    enum Admission {
        /// Start a worker for this job; it takes queued jobs once it is done with it.
        case startWorker
        /// A running worker takes this job when it is free.
        case queued
        /// Every worker is busy and the queue is full: the job is not run.
        case refused
    }

    let maximumWorkers: Int
    let maximumQueued: Int
    private(set) var running = 0
    private var waiting: [Job] = []

    init(maximumWorkers: Int, maximumQueued: Int) {
        self.maximumWorkers = maximumWorkers
        self.maximumQueued = maximumQueued
    }

    /// Admits `job`. On `.startWorker` the caller hands `job` to the new worker; on `.queued` it is kept here.
    mutating func admit(_ job: Job) -> Admission {
        if running < maximumWorkers {
            running += 1
            return .startWorker
        }
        guard waiting.count < maximumQueued else { return .refused }
        waiting.append(job)
        return .queued
    }

    /// The oldest waiting job, for a worker that finished one; nil when none waits, and the worker then stops.
    mutating func next() -> Job? {
        guard !waiting.isEmpty else {
            running -= 1
            return nil
        }
        return waiting.removeFirst()
    }
}

/// A few threads of their own for lookups that block (`getaddrinfo`, `DNSServiceProcessResult`), so no cooperative-pool thread
/// ever waits on the system resolver. A burst starts at most `maximumWorkers` threads; each runs lookups one after another until
/// none waits, then ends, so a quiet stack holds no thread.
final class ResolverWorkers: Sendable {
    typealias Job = @Sendable () -> Void

    private let queue: Mutex<ResolverQueue<Job>>

    init(maximumWorkers: Int, maximumQueued: Int) {
        queue = Mutex(ResolverQueue(maximumWorkers: maximumWorkers, maximumQueued: maximumQueued))
    }

    /// Runs `job` on a worker thread soon. Returns false, and never runs it, when every worker is busy and the queue is full.
    func submit(_ job: @escaping Job) -> Bool {
        switch queue.withLock({ $0.admit(job) }) {
        case .startWorker:
            Thread.detachNewThread { [self] in
                job()
                drain()
            }
            return true
        case .queued:
            return true
        case .refused:
            return false
        }
    }

    private func drain() {
        while let job = queue.withLock({ $0.next() }) { job() }
    }
}
```

```swift
import Foundation

@testable import NetStack

struct ResolverQueueTests {
    @Test func workersStartUpToTheLimitThenLookupsWaitThenTheyAreRefused() {
        var queue = ResolverQueue<Int>(maximumWorkers: 2, maximumQueued: 1)
        // `#expect` cannot hold a mutating call, so each result is taken first.
        let admissions = [queue.admit(1), queue.admit(2), queue.admit(3), queue.admit(4)]
        #expect(admissions == [.startWorker, .startWorker, .queued, .refused])
        let next = queue.next()
        let last = queue.next()
        #expect(next == 3, "a worker that finished takes the oldest waiting lookup")
        #expect(last == nil, "and stops when none is left")
        #expect(queue.running == 1)
        let again = queue.admit(5)
        #expect(again == .startWorker, "a stopped worker's place is free again")
    }
}

struct ResolverWorkersTests {
    @Test func everyLookupOfABurstLargerThanThePoolRuns() async throws {
        let workers = ResolverWorkers(maximumWorkers: 4, maximumQueued: 200)
        let done = AsyncEventProbe<Int>()
        for index in 0..<100 { #expect(workers.submit { done.record(index) }) }
        let events = try await done.wait(forAtLeast: 100)
        #expect(Set(events) == Set(0..<100))
    }
}
```
