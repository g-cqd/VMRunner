Why: ADR-0056 decisions 2, 4, 5 and 7. The four methods `machines.list`, `machines.state`, `machines.start` and
`machines.stop` run only when, in this order: the method exists and its params are well formed; a token is presented and
matches a stored one; the token has the method's capability; the machine (when one is named) is one of the token's; the
machine exists and its agent access is on. Each refusal is reported (the app logs it in J10) as a `ControlDenial`, a value
that has no field for a token. The machines are behind a protocol, `ControlMachines`: the app's library in J10, a fake here.

```swift
public import Foundation

/// What a machine is doing, as the control service reports it.
public enum ControlState: String, Codable, CaseIterable, Sendable {
    case stopped, starting, running, shuttingDown, pausing, paused, resuming, suspending, stopping
}

/// A machine as an agent sees it.
public struct ControlMachine: Codable, Equatable, Sendable {
    public let id: UUID
    public let name: String
    /// `macOS` or `linux`.
    public let guest: String
    public let state: ControlState
    /// Only machines whose agent access is on are listed or acted on.
    public let agentAccess: Bool

    public init(id: UUID, name: String, guest: String, state: ControlState, agentAccess: Bool) {
        self.id = id
        self.name = name
        self.guest = guest
        self.state = state
        self.agentAccess = agentAccess
    }
}

/// What a start or a stop came to.
public enum ControlOutcome: Equatable, Sendable {
    case done(ControlState)
    /// Not done, with a sentence that says why.
    case refused(String)
}

/// The machines the service acts on: the app's library, or a fake in tests.
public protocol ControlMachines: Sendable {
    func machines() async -> [ControlMachine]
    func start(_ id: UUID) async -> ControlOutcome
    func stop(_ id: UUID, force: Bool) async -> ControlOutcome
}

/// The methods of the control service, each needing one capability.
public enum ControlMethod: String, CaseIterable, Sendable {
    case list = "machines.list"
    case state = "machines.state"
    case start = "machines.start"
    case stop = "machines.stop"

    public var capability: ControlCapability {
        switch self {
        case .list: .list
        case .state: .state
        case .start: .start
        case .stop: .stop
        }
    }

    /// The members its `params` may hold.
    var parameters: Set<String> {
        switch self {
        case .list: ["token"]
        case .state, .start: ["token", "machine"]
        case .stop: ["token", "machine", "force"]
        }
    }
}

/// A refused call, as it is logged. It has no field for a token, so what is logged can never hold one.
public struct ControlDenial: Error, Equatable, Sendable {
    public enum Reason: String, Sendable {
        case otherUser = "the client runs as another user"
        case noToken = "no token"
        case unknownToken = "an unknown or revoked token"
        case missingCapability = "the token lacks the capability"
        case outsideScope = "the machine is not one of the token's"
        case agentAccessOff = "agent access is off for the machine"
    }

    public let reason: Reason
    public let method: ControlMethod?
    public let tokenID: UUID?
    public let machine: UUID?
    /// The user id of a client that runs as another user.
    public let peerUser: UInt32?

    public init(
        reason: Reason, method: ControlMethod? = nil, tokenID: UUID? = nil, machine: UUID? = nil, peerUser: UInt32? = nil
    ) {
        self.reason = reason
        self.method = method
        self.tokenID = tokenID
        self.machine = machine
        self.peerUser = peerUser
    }

    /// What was refused and why, without any identifier: safe to log in the clear.
    public var summary: String {
        let what = method.map { "\($0.rawValue) refused" } ?? "connection refused"
        return peerUser.map { "\(what): \(reason.rawValue) (user \($0))" } ?? "\(what): \(reason.rawValue)"
    }
}
```

```swift
import Foundation

/// Checks each request against its token and the machine's agent access, then calls the machines (ADR-0056). Every refusal
/// goes to `report`, which the app logs.
public struct ControlDispatcher: Sendable {
    private let tokens: ControlTokenStore
    private let machines: any ControlMachines
    private let reportDenial: @Sendable (ControlDenial) -> Void

    public init(tokens: ControlTokenStore, machines: any ControlMachines, report: @escaping @Sendable (ControlDenial) -> Void) {
        self.tokens = tokens
        self.machines = machines
        reportDenial = report
    }

    /// Reports a refusal decided before a request is read (a client of another user).
    public func report(_ denial: ControlDenial) { reportDenial(denial) }

    /// The answer to `request`.
    public func answer(_ request: RPCRequest) async -> RPCResponse {
        guard let method = ControlMethod(rawValue: request.method) else {
            return RPCResponse(id: request.id, error: .methodNotFound)
        }
        let call: ControlCall
        switch ControlCall.read(method, request.params) {
        case .success(let read): call = read
        case .failure(let error): return RPCResponse(id: request.id, error: error)
        }
        switch authorize(call) {
        case .failure(let denial):
            report(denial)
            return RPCResponse(id: request.id, error: .notPermitted)
        case .success(let token):
            return await perform(call, token: token, id: request.id)
        }
    }

    private func authorize(_ call: ControlCall) -> Result<ControlToken, ControlDenial> {
        func denied(_ reason: ControlDenial.Reason, _ token: ControlToken? = nil) -> Result<ControlToken, ControlDenial> {
            .failure(ControlDenial(reason: reason, method: call.method, tokenID: token?.id, machine: call.machine))
        }
        guard let secret = call.token else { return denied(.noToken) }
        guard let token = tokens.authenticate(secret) else { return denied(.unknownToken) }
        guard token.capabilities.contains(call.method.capability) else { return denied(.missingCapability, token) }
        if let machine = call.machine, let allowed = token.machines, !allowed.contains(machine) {
            return denied(.outsideScope, token)
        }
        return .success(token)
    }

    private func perform(_ call: ControlCall, token: ControlToken, id: RPCID) async -> RPCResponse {
        let all = await machines.machines()
        guard let machineID = call.machine else {
            let visible = all.filter { $0.agentAccess && (token.machines?.contains($0.id) ?? true) }
            return Self.result(ControlMachineList(machines: visible), id: id)
        }
        guard let machine = all.first(where: { $0.id == machineID }) else {
            return RPCResponse(id: id, error: .noSuchMachine)
        }
        guard machine.agentAccess else {
            report(ControlDenial(reason: .agentAccessOff, method: call.method, tokenID: token.id, machine: machineID))
            return RPCResponse(id: id, error: .agentAccessOff)
        }
        switch call.method {
        case .start: return Self.outcome(await machines.start(machineID), of: machineID, id: id)
        case .stop: return Self.outcome(await machines.stop(machineID, force: call.force), of: machineID, id: id)
        case .list, .state: return Self.result(machine, id: id)
        }
    }

    private static func outcome(_ outcome: ControlOutcome, of machine: UUID, id: RPCID) -> RPCResponse {
        switch outcome {
        case .done(let state): result(ControlStateChange(id: machine, state: state), id: id)
        case .refused(let reason): RPCResponse(id: id, error: .refused.with(reason))
        }
    }

    private static func result(_ value: some Encodable, id: RPCID) -> RPCResponse {
        let encoder = JSONEncoder()
        encoder.outputFormatting = [.sortedKeys]
        guard let json = try? encoder.encode(value) else { return RPCResponse(id: id, error: .internalError) }
        return RPCResponse(id: id, result: json)
    }
}

/// The answer of `machines.list`.
struct ControlMachineList: Encodable {
    let machines: [ControlMachine]
}

/// The answer of `machines.start` and `machines.stop`.
struct ControlStateChange: Encodable {
    let id: UUID
    let state: ControlState
}

/// A request's params, checked against what its method takes.
struct ControlCall: Equatable {
    let method: ControlMethod
    var token: String?
    var machine: UUID?
    var force = false

    static func read(_ method: ControlMethod, _ params: [String: RPCValue]) -> Result<Self, RPCError> {
        guard Set(params.keys).isSubset(of: method.parameters) else {
            let allowed = method.parameters.sorted().joined(separator: ", ")
            return .failure(.invalidParams.with("\(method.rawValue) takes only \(allowed)"))
        }
        var call = Self(method: method)
        switch params["token"] {
        case .string(let secret): call.token = secret
        case .bool: return .failure(.invalidParams.with("token must be a string"))
        case nil: break
        }
        if method != .list {
            guard case .string(let text) = params["machine"], let machine = UUID(uuidString: text) else {
                return .failure(.invalidParams.with("machine must be a machine's id"))
            }
            call.machine = machine
        }
        switch params["force"] {
        case .bool(let force): call.force = force
        case .string: return .failure(.invalidParams.with("force must be true or false"))
        case nil: break
        }
        return .success(call)
    }
}
```

```swift
import ControlProtocol
import Foundation
import Synchronization

/// Machines for the tests: a list to report, every start and stop recorded, and a start that waits at `startGate` when set.
final class FakeMachines: ControlMachines {
    private let list: Mutex<[ControlMachine]>
    let calls = AsyncEventProbe<String>()
    let startGate: AsyncGate?
    let startOutcome: ControlOutcome

    init(_ machines: [ControlMachine], startGate: AsyncGate? = nil, startOutcome: ControlOutcome = .done(.running)) {
        list = Mutex(machines)
        self.startGate = startGate
        self.startOutcome = startOutcome
    }

    func machines() async -> [ControlMachine] { list.withLock { $0 } }

    func start(_ id: UUID) async -> ControlOutcome {
        calls.record("start \(id.uuidString)")
        if let startGate { try? await startGate.waitUntilOpen() }
        return startOutcome
    }

    func stop(_ id: UUID, force: Bool) async -> ControlOutcome {
        calls.record("stop \(id.uuidString) force=\(force)")
        return .done(force ? .stopped : .shuttingDown)
    }
}

extension ControlMachine {
    static func fake(_ name: String, access: Bool = true, state: ControlState = .stopped) -> ControlMachine {
        Self(id: UUID(), name: name, guest: "linux", state: state, agentAccess: access)
    }
}
```

```swift
import Foundation
import TestKit
import Testing

@testable import ControlProtocol

/// The checks every request passes (a token, its capability, its machines, the machine's agent access) and what each method
/// answers. Refusals are reported, and never with a token.
struct ControlDispatcherTests {
    /// A dispatcher over fake machines and a store in a scratch folder, with the refusals it reported.
    struct Rig {
        let folder: ScratchFolder
        let store: ControlTokenStore
        let machines: FakeMachines
        let denials: AsyncEventProbe<ControlDenial>
        let dispatcher: ControlDispatcher

        init(_ list: [ControlMachine], startOutcome: ControlOutcome = .done(.running)) throws {
            folder = try ScratchFolder()
            store = try ControlTokenStore(file: folder.url.appending(path: "tokens.json"))
            machines = FakeMachines(list, startOutcome: startOutcome)
            let probe = AsyncEventProbe<ControlDenial>()
            denials = probe
            dispatcher = ControlDispatcher(tokens: store, machines: machines, report: { probe.record($0) })
        }

        func secret(_ capabilities: Set<ControlCapability> = Set(ControlCapability.allCases), machines: [UUID]? = nil) throws
            -> String
        {
            try store.create(name: "test", capabilities: capabilities, machines: machines).secret
        }

        func call(_ method: String, _ params: [String: RPCValue]) async -> RPCResponse {
            await dispatcher.answer(RPCRequest(id: .number(1), method: method, params: params))
        }
    }

    private func object(_ response: RPCResponse) throws -> [String: Any] {
        guard case .result(let json) = response.body else {
            Issue.record("expected a result, got \(response)")
            return [:]
        }
        return try #require(try JSONSerialization.jsonObject(with: json) as? [String: Any])
    }

    private func names(_ response: RPCResponse) throws -> [String] {
        (try object(response)["machines"] as? [[String: Any]] ?? []).compactMap { $0["name"] as? String }
    }

    @Test
    func eachMethodNeedsItsOwnCapability() async throws {
        let machine = ControlMachine.fake("m")
        let rig = try Rig([machine])
        defer { rig.folder.cleanUp() }
        for method in ControlMethod.allCases {
            let lacking = try rig.secret(Set(ControlCapability.allCases).subtracting([method.capability]))
            let holding = try rig.secret([method.capability])
            var params: [String: RPCValue] = method == .list ? [:] : ["machine": .string(machine.id.uuidString)]
            params["token"] = .string(lacking)
            #expect(await rig.call(method.rawValue, params).error?.code == -32001, "\(method) without its capability")
            params["token"] = .string(holding)
            #expect(await rig.call(method.rawValue, params).error == nil, "\(method) with its capability")
        }
        #expect(rig.denials.events.count == ControlMethod.allCases.count)
        #expect(rig.denials.events.allSatisfy { $0.reason == .missingCapability && $0.tokenID != nil })
    }

    @Test
    func noTokenAnUnknownOneAndARevokedOneAreRefusedAndReported() async throws {
        let rig = try Rig([.fake("m")])
        defer { rig.folder.cleanUp() }
        let made = try rig.store.create(name: "gone", capabilities: [.list])
        try rig.store.revoke(made.token.id)

        #expect(await rig.call("machines.list", [:]).error?.code == -32001)
        #expect(await rig.call("machines.list", ["token": .string(ControlSecret.make())]).error?.code == -32001)
        #expect(await rig.call("machines.list", ["token": .string(made.secret)]).error?.code == -32001)
        #expect(rig.denials.events.map(\.reason) == [.noToken, .unknownToken, .unknownToken])
        #expect(rig.denials.events.allSatisfy { $0.tokenID == nil && $0.method == .list })
    }

    @Test
    func aTokenLimitedToSomeMachinesSeesAndActsOnThoseOnly() async throws {
        let mine = ControlMachine.fake("mine")
        let other = ControlMachine.fake("other")
        let rig = try Rig([mine, other])
        defer { rig.folder.cleanUp() }
        let secret = try rig.secret(machines: [mine.id])

        let listed = await rig.call("machines.list", ["token": .string(secret)])
        #expect(try names(listed) == ["mine"])
        #expect(await rig.call("machines.state", ["token": .string(secret), "machine": .string(mine.id.uuidString)]).error == nil)
        let refused = await rig.call("machines.start", ["token": .string(secret), "machine": .string(other.id.uuidString)])
        #expect(refused.error?.code == -32001)
        #expect(rig.denials.events.map(\.reason) == [.outsideScope])
        #expect(rig.machines.calls.events.isEmpty, "nothing was started")
    }

    @Test
    func aMachineWithAgentAccessOffIsLeftOutOfTheListAndRefused() async throws {
        let closed = ControlMachine.fake("closed", access: false)
        let open = ControlMachine.fake("open")
        let rig = try Rig([closed, open])
        defer { rig.folder.cleanUp() }
        let secret = try rig.secret()

        let listed = await rig.call("machines.list", ["token": .string(secret)])
        #expect(try names(listed) == ["open"])
        let refused = await rig.call("machines.start", ["token": .string(secret), "machine": .string(closed.id.uuidString)])
        #expect(refused.error?.code == -32003)
        #expect(rig.denials.events.map(\.reason) == [.agentAccessOff])
        #expect(rig.machines.calls.events.isEmpty, "nothing was started")
        let unknown = await rig.call("machines.state", ["token": .string(secret), "machine": .string(UUID().uuidString)])
        #expect(unknown.error?.code == -32002)
    }

    @Test
    func malformedParamsAndUnknownMethodsAreRefusedBeforeAnyTokenIsLookedAt() async throws {
        let machine = ControlMachine.fake("m")
        let rig = try Rig([machine])
        defer { rig.folder.cleanUp() }
        let secret = try rig.secret()
        let id = machine.id.uuidString
        #expect(await rig.call("machines.state", ["token": .string(secret)]).error?.code == -32602, "no machine")
        #expect(await rig.call("machines.state", ["token": .string(secret), "machine": .string("nope")]).error?.code == -32602)
        #expect(await rig.call("machines.list", ["token": .string(secret), "verbose": .bool(true)]).error?.code == -32602)
        #expect(
            await rig.call("machines.stop", ["token": .string(secret), "machine": .string(id), "force": .string("yes")]).error?
                .code == -32602)
        #expect(await rig.call("machines.list", ["token": .bool(true)]).error?.code == -32602)
        #expect(await rig.call("machines.delete", ["token": .string(secret)]).error?.code == -32601)
        #expect(await rig.call("rpc.discover", [:]).error?.code == -32601)
        #expect(rig.denials.events.isEmpty)
    }

    @Test
    func startAndStopAnswerTheNewStateAndAStopPassesForce() async throws {
        let machine = ControlMachine.fake("m")
        let rig = try Rig([machine])
        defer { rig.folder.cleanUp() }
        let secret = try rig.secret()
        let id = machine.id.uuidString

        let startAnswer = await rig.call("machines.start", ["token": .string(secret), "machine": .string(id)])
        let started = try object(startAnswer)
        #expect(started["state"] as? String == "running")
        #expect(started["id"] as? String == id)
        let forceAnswer = await rig.call("machines.stop", ["token": .string(secret), "machine": .string(id), "force": .bool(true)])
        #expect(try object(forceAnswer)["state"] as? String == "stopped")
        let askAnswer = await rig.call("machines.stop", ["token": .string(secret), "machine": .string(id)])
        #expect(try object(askAnswer)["state"] as? String == "shuttingDown")
        #expect(rig.machines.calls.events == ["start \(id)", "stop \(id) force=true", "stop \(id) force=false"])

        let stateAnswer = await rig.call("machines.state", ["token": .string(secret), "machine": .string(id)])
        let state = try object(stateAnswer)
        #expect(state["name"] as? String == "m")
        #expect(state["state"] as? String == "stopped")
    }

    @Test
    func aRefusedStartIsAnErrorThatSaysWhy() async throws {
        let machine = ControlMachine.fake("m")
        let rig = try Rig([machine], startOutcome: .refused("“m” is being set up."))
        defer { rig.folder.cleanUp() }
        let secret = try rig.secret()
        let answer = await rig.call("machines.start", ["token": .string(secret), "machine": .string(machine.id.uuidString)])
        #expect(answer.error == RPCError.refused.with("“m” is being set up."))
    }

    @Test
    func noRefusalCarriesTheTokenThatWasPresented() async throws {
        let closed = ControlMachine.fake("closed", access: false)
        let rig = try Rig([closed])
        defer { rig.folder.cleanUp() }
        let weak = try rig.secret([.list])
        let strong = try rig.secret()
        let unknown = ControlSecret.make()
        _ = await rig.call("machines.start", ["token": .string(weak), "machine": .string(closed.id.uuidString)])
        _ = await rig.call("machines.start", ["token": .string(strong), "machine": .string(closed.id.uuidString)])
        _ = await rig.call("machines.list", ["token": .string(unknown)])
        #expect(rig.denials.events.count == 3)
        for denial in rig.denials.events {
            for secret in [weak, strong, unknown] {
                #expect(!String(describing: denial).contains(secret))
                #expect(!denial.summary.contains(secret))
                #expect(!denial.summary.contains(ControlSecret.prefix))
            }
        }
        let otherUser = ControlDenial(reason: .otherUser, peerUser: 501)
        #expect(otherUser.summary == "connection refused: the client runs as another user (user 501)")
    }
}
```

carries a token.

## Stop conditions

- A test fails on the order of the checks (for example an unknown method reported as a denial): report it; do not reorder the
  checks or change the test, the order is ADR-0056's.
- `FakeMachines.swift` fails to compile on `AsyncEventProbe()` with `no macro named _sourceLocation`: it already imports Testing
  for that; if it still fails, report the error.
- `switch params["token"]` with `case nil` does not compile: replace it by `case .none` (the same meaning) and report it.
- SwiftLint reports `cyclomatic_complexity` or `function_body_length` on `ControlCall.read` or `answer`: split out a private
  helper with the same checks in the same order; report it.
