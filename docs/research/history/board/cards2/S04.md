Why: the app will run a container as a machine of its own kind (a "container machine"), next to the virtual machines. Before any engine or screen exists, the configuration must be able to say what a machine runs and, for a container, which image, command, environment, CPUs, memory and network, and the library must keep such a machine: a bundle that holds `config.json` and nothing else (no disk, no firmware state, no display). The limits are the container runner's own, so a configuration the app writes is one the runner accepts (image: 1 to 255 UTF-8 bytes, no whitespace or control character; command: at most 64 elements of at most 4096 bytes, no NUL; environment: at most 64 `NAME=value` entries, NAME matching `[A-Za-z_][A-Za-z0-9_]*`, value without NUL or newline, no name twice; CPUs 1 to 16; memory 128 to 16384 MiB). Decisions: `workload` defaults to `vm`, so every stored configuration is a virtual machine; a container machine's spec is required by `validate()`, and `VMConfig.decode(from:)` supplies the default spec when a stored container configuration has none (a partial spec gets the default of each missing key); a virtual machine ignores `container`, even a bad one; the environment is an ordered list of `{name, value}` objects, so the order is kept and a duplicate can be refused; a container machine is a `.linux` guest as far as the settings go and is created `isProvisioned` (there is nothing to set up); duplicating one keeps the workload and the spec and makes no machine identifier. Nothing starts a container yet: the two ways to start a machine (`startMachine`, `startAndWait`) refuse a container machine with a clear message, and `beginSetup` ignores it; the card that adds the engine replaces those guards.

import Foundation

/// What a machine runs: a virtual machine with its own kernel and disk, or one container started from an image.
enum WorkloadKind: String, Codable, CaseIterable, Identifiable, Sendable {
    case vm, container

    var id: Self { self }
    var label: String { self == .vm ? "Virtual machine" : "Container" }
}

/// The network a container sees. A container never reaches beyond this Mac.
enum ContainerNetwork: String, Codable, CaseIterable, Identifiable, Sendable {
    case none, loopback

    var id: Self { self }
    var label: String { self == .none ? "None" : "Loopback only" }
}

/// What the container runner accepts, so the editor, `VMConfig.validate()` and the runner's own checks cannot disagree.
enum ContainerLimits {
    static let maxImageBytes = 255
    static let maxCommandElements = 64
    static let maxCommandElementBytes = 4096
    static let maxEnvironmentEntries = 64
    static let cpus = 1...16
    static let memoryMiB = 128...16384
}

/// One `NAME=value` setting of a container's environment. The spec keeps them as a list, so the order the user chose is the
/// order the container sees, and `VMConfig.validate()` refuses a name that appears twice.
struct ContainerEnvironmentEntry: Codable, Hashable, Sendable {
    var name: String
    var value: String

    /// A letter or an underscore, then letters, digits and underscores: ASCII only.
    var hasValidName: Bool {
        func isLetterOrUnderscore(_ byte: UInt8) -> Bool {
            (UInt8(ascii: "A")...UInt8(ascii: "Z")).contains(byte) || (UInt8(ascii: "a")...UInt8(ascii: "z")).contains(byte)
                || byte == UInt8(ascii: "_")
        }
        guard let first = name.utf8.first, isLetterOrUnderscore(first) else { return false }
        return name.utf8.dropFirst().allSatisfy { isLetterOrUnderscore($0) || (UInt8(ascii: "0")...UInt8(ascii: "9")).contains($0) }
    }

    /// A value holds no NUL byte and no newline.
    var hasValidValue: Bool { !value.utf8.contains { $0 == 0 || $0 == UInt8(ascii: "\n") } }
}

/// How a container machine runs. A stored spec that lacks a key gets that key's default.
struct ContainerSpec: Codable, Hashable, Sendable {
    static let defaultImage = "docker.io/library/alpine:3.24.2"
    static let defaultCPUs = 2
    static let defaultMemoryMiB = 512

    var image = Self.defaultImage
    /// The program and its arguments, one element each, run without a shell. Empty means the runner's baseline command.
    var command: [String] = []
    var environment: [ContainerEnvironmentEntry] = []
    var cpus = Self.defaultCPUs
    var memoryMiB = Self.defaultMemoryMiB
    var network = ContainerNetwork.none

    init(
        image: String = Self.defaultImage, command: [String] = [], environment: [ContainerEnvironmentEntry] = [],
        cpus: Int = Self.defaultCPUs,
        memoryMiB: Int = Self.defaultMemoryMiB, network: ContainerNetwork = .none
    ) {
        self.image = image
        self.command = command
        self.environment = environment
        self.cpus = cpus
        self.memoryMiB = memoryMiB
        self.network = network
    }

    init(from decoder: any Decoder) throws {
        let values = try decoder.container(keyedBy: CodingKeys.self)
        image = try values.decodeIfPresent(String.self, forKey: .image) ?? Self.defaultImage
        command = try values.decodeIfPresent([String].self, forKey: .command) ?? []
        environment = try values.decodeIfPresent([ContainerEnvironmentEntry].self, forKey: .environment) ?? []
        cpus = try values.decodeIfPresent(Int.self, forKey: .cpus) ?? Self.defaultCPUs
        memoryMiB = try values.decodeIfPresent(Int.self, forKey: .memoryMiB) ?? Self.defaultMemoryMiB
        network = try values.decodeIfPresent(ContainerNetwork.self, forKey: .network) ?? .none
    }
}

    /// A container machine's spec, within what the container runner accepts. A virtual machine ignores its `container`.
    private func validateContainer() throws {
        guard workload == .container else { return }
        guard let spec = container else { throw Self.corrupted("container", "is required for a container machine") }
        try Self.requireImage(spec.image)
        guard spec.command.count <= ContainerLimits.maxCommandElements else {
            throw Self.corrupted("container.command", "must hold at most \(ContainerLimits.maxCommandElements) elements")
        }
        for element in spec.command {
            guard element.utf8.count <= ContainerLimits.maxCommandElementBytes, !element.utf8.contains(0) else {
                throw Self.corrupted(
                    "container.command", "must hold elements of at most \(ContainerLimits.maxCommandElementBytes) bytes without NUL")
            }
        }
        try Self.requireEnvironment(spec.environment)
        try Self.require(spec.cpus, in: ContainerLimits.cpus, "container.cpus")
        try Self.require(spec.memoryMiB, in: ContainerLimits.memoryMiB, "container.memoryMiB")
    }

    private static func requireImage(_ image: String) throws {
        guard !image.isEmpty, image.utf8.count <= ContainerLimits.maxImageBytes else {
            throw corrupted("container.image", "must be from 1 to \(ContainerLimits.maxImageBytes) bytes")
        }
        guard !image.unicodeScalars.contains(where: { $0.properties.isWhitespace || $0.properties.generalCategory == .control }) else {
            throw corrupted("container.image", "must not hold whitespace or control characters")
        }
    }

    private static func requireEnvironment(_ environment: [ContainerEnvironmentEntry]) throws {
        guard environment.count <= ContainerLimits.maxEnvironmentEntries else {
            throw corrupted("container.environment", "must hold at most \(ContainerLimits.maxEnvironmentEntries) entries")
        }
        var seen = Set<String>()
        for entry in environment {
            guard entry.hasValidName else {
                throw corrupted(
                    "container.environment", "must name variables of letters, digits and underscores, not starting with a digit")
            }
            guard entry.hasValidValue else { throw corrupted("container.environment", "must hold values without NUL or newline") }
            guard seen.insert(entry.name).inserted else { throw corrupted("container.environment", "must not name \(entry.name) twice") }
        }
    }

    private static func require(_ value: Int, in range: ClosedRange<Int>, _ key: String) throws {
