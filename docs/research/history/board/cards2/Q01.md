## Why

Roadmap 0014 and ADR-0014 item 3 leave one question open: can a virtio GPU implemented by the host, through the custom
virtio devices of the macOS 27 framework, drive a Linux guest's display? This chain builds the smallest thing that answers
it: a runner that boots one Linux guest with that device and shows (or dumps) its two-dimensional scanout. It is a
feasibility study, so it follows the container runner's pattern (roadmap 0024): a package of its own under `Packages/`,
which the root package does not depend on, so the app never links it and `vmr build-app` (which builds the root's
`VMRunner` product only) never bundles it; the gate still formats, lints, builds, tests and analyzes it (the `packages`
and `lintPaths` keys of `.vmr.json`, `PackageLayoutTests` already require both and a committed `Package.resolved`). The
names follow ADR-0042: the package is named for the capability (`VirtioGPU`), the executable for its role
(`VirtioGPURunner`), its testable logic is `VirtioGPURunnerCore`, its tests `VirtioGPURunnerTests`.

This card adds the package with the device's identity and configuration space (virtio specification 1.3, section 5.7: device
ID 16, PCI class 0x03 display, subclass 0x80 "other" since the device is not VGA compatible, a control and a cursor queue,
`VIRTIO_F_VERSION_1` = feature bit 32 = bit 0 of the upper half, `struct virtio_gpu_config` = four little-endian 32-bit
fields), an executable that prints them (`--describe`), and the layering tests: the root manifest never names the package,
no root source imports its modules, the library imports no interface or virtualization framework, and the package depends
on no other package.

3. Create `Packages/VirtioGPU/Sources/VirtioGPURunnerCore/GPUWire.swift` with exactly:

```swift
/// What the guest sees of the device before any command: its virtio identity, its queues and the feature bits it offers
/// (virtio specification 1.3, section 5.7). Two-dimensional only: none of the 3D, EDID, blob or context feature bits.
package enum GPUWire {
    /// The virtio device ID of a GPU device.
    package static let deviceID: UInt16 = 16
    /// PCI class 0x03, display controller.
    package static let pciClassID: UInt8 = 0x03
    /// PCI subclass 0x80, "other display controller": the device is not VGA compatible.
    package static let pciSubclassID: UInt8 = 0x80
    /// The control queue and the cursor queue.
    package static let queueCount: UInt16 = 2
    package static let controlQueue: UInt16 = 0
    package static let cursorQueue: UInt16 = 1
    /// `VIRTIO_F_VERSION_1` is feature bit 32: bit 0 of the upper 32-bit half of the feature set.
    package static let versionOneFeatureSubset1: UInt32 = 1
    /// `VIRTIO_GPU_EVENT_DISPLAY`, set in `events_read` when the display configuration changed.
    package static let eventDisplay: UInt32 = 1
}

/// The device-specific configuration space of a virtio GPU (`struct virtio_gpu_config`): four little-endian 32-bit fields.
package struct GPUConfigSpace: Equatable, Sendable {
    package var eventsRead: UInt32
    package var eventsClear: UInt32
    package var scanoutCount: UInt32
    package var capsetCount: UInt32

    package init(eventsRead: UInt32, eventsClear: UInt32, scanoutCount: UInt32, capsetCount: UInt32) {
        self.eventsRead = eventsRead
        self.eventsClear = eventsClear
        self.scanoutCount = scanoutCount
        self.capsetCount = capsetCount
    }

    /// The 16 bytes the guest reads, in field order, each field little-endian.
    package var bytes: [UInt8] {
        [eventsRead, eventsClear, scanoutCount, capsetCount].flatMap { field in
            (0..<4).map { UInt8(truncatingIfNeeded: field >> ($0 * 8)) }
        }
    }
}
```

import Foundation
import VirtioGPURunnerCore

/// The lines the runner writes go straight to the standard streams.
func line(_ text: String, error: Bool = false) {
    let handle = error ? FileHandle.standardError : FileHandle.standardOutput
    try? handle.write(contentsOf: Data((text + "\n").utf8))
}

let usage = """
    usage: VirtioGPURunner --describe
      --describe   print the virtio identity and the configuration space the guest sees
    The device needs macOS 27 or later.
    """

func hex(_ bytes: [UInt8]) -> String {
    bytes.map { byte in
        let digits = String(byte, radix: 16)
        return digits.count == 1 ? "0" + digits : digits
    }.joined(separator: " ")
}

let arguments = Array(CommandLine.arguments.dropFirst())
guard arguments == ["--describe"] else {
    line(usage, error: !(arguments == ["--help"] || arguments == ["-h"]))
    exit(arguments == ["--help"] || arguments == ["-h"] ? 0 : 64)
}
guard #available(macOS 27.0, *) else {
    line("error: the custom virtio device needs macOS 27 or later", error: true)
    exit(69)
}
let space = GPUConfigSpace(eventsRead: 0, eventsClear: 0, scanoutCount: 1, capsetCount: 0)
line("device ID \(GPUWire.deviceID), PCI class \(GPUWire.pciClassID) subclass \(GPUWire.pciSubclassID)")
line("queues \(GPUWire.queueCount): control \(GPUWire.controlQueue), cursor \(GPUWire.cursorQueue)")
line("mandatory features: subset1 \(GPUWire.versionOneFeatureSubset1) (VIRTIO_F_VERSION_1); optional: none")
line("display event bit \(GPUWire.eventDisplay)")
line("configuration space: \(hex(space.bytes))")
exit(0)
