**The queue rules (`VirtioInputEndpoint`), over any queue of elements, tested with fakes.** Event queue: each buffer the guest posts is handed to the library's `InputEventQueue.hold` and **not** given back; the library writes one 8-byte event into the oldest held buffer for each event of a complete frame and gives that buffer back, exactly once; a buffer over the library's limit of 1024 held is given back at once, unwritten, as the library says; a buffer too small, or one whose write fails, is given back unwritten and its event waits. import VirtioGPU
````swift
import VirtioInput

/// A queue element of the event queue as one of the library's buffers: the room the guest left for an event, a write, and the
/// one way back.
struct EventSlot<Element: QueueElement>: GuestEventBuffer {
    let element: Element

    var writeCapacity: Int { element.writableByteCount }

    func put(_ bytes: [UInt8]) throws {
        try element.write(bytes)
    }

    func giveBack() {
        element.giveBack()
    }
}

/// A queue element of the status queue as one of the library's buffers.
struct StatusSlot<Element: QueueElement>: GuestStatusBuffer {
    let element: Element

    var byteCount: Int { element.readableByteCount }

    func take(count: Int) throws -> [UInt8] {
        try element.read(count: count)
    }

    func giveBack() {
        element.giveBack()
    }
}

/// What a virtio input device does with its two queues, over any queue of elements, so that it is tested without a machine.
///
/// **The event queue.** The guest posts empty buffers; the device keeps them (`takeBuffers`) and gives each one back exactly
/// once, with one event in it, when there is an event for it (`send`). The library holds at most 1024 and gives one over the
/// limit back at once. A buffer given back with no event would show the guest the event it held last, so none is given back
/// without one but the ones the library refuses.
///
/// **A reset or a stop** forgets every buffer held (`forgetHeld`) and every event not yet delivered, and gives none back: the
/// guest's queue is reset with the device, and the framework raises an exception when an element is returned to a queue that no
/// longer has it.
///
/// **The status queue.** Each element is read once, decoded and given back at once, whatever it holds.
@MainActor
final class VirtioInputEndpoint<Queue: ElementQueue> {
    private var events = InputEventQueue<EventSlot<Queue.Element>>()
    /// The status elements that were not one event of 8 bytes, since the endpoint was made.
    private(set) var malformedStatusCount = 0

    /// How many frames the event queue dropped since the last `forgetHeld`, because the guest holds no buffers for them.
    var droppedFrameCount: Int { events.droppedFrameCount }

    /// Takes up to `budget` buffers (at least one) the guest posted. Returns `true` when the budget ended the turn: more may
    /// wait, and the caller must come back for them, because the guest notifies once for a batch and not again.
    func takeBuffers(from queue: Queue, budget: Int) -> Bool {
        let budget = max(budget, 1)
        var taken = 0
        while taken < budget, let element = queue.nextElement() {
            events.hold(EventSlot(element: element))
            taken += 1
        }
        return taken == budget
    }

    /// Serves up to `budget` elements (at least one) of the status queue, with the same return value as `takeBuffers`.
    func serveStatus(from queue: Queue, budget: Int) -> Bool {
        let budget = max(budget, 1)
        var served = 0
        while served < budget, let element = queue.nextElement() {
            if case .malformed = InputStatus.serve(StatusSlot(element: element)) { malformedStatusCount += 1 }
            served += 1
        }
        return served == budget
    }

    /// Frames of events for the guest, each ending in a report, written to the buffers held as they allow.
    func send(_ frames: [InputEvent]) {
        events.enqueue(frames)
    }

    /// Lets go of the buffers held and the events waiting, without giving anything back.
    func forgetHeld() {
        events = InputEventQueue()
    }
}
````

````swift
import AppKit
import VirtioInput

/// The keys of a Mac keyboard as the guest's virtio keyboard sees them: what a key press, a release or a change of the
/// modifiers sends, and which keys the guest holds down, so that none stays down when the window loses the keyboard.
struct KeyboardState {
    /// The modifier keys by the flag that says one of them is down, with the virtual key code of each (left, then right).
    private static let modifierKeys: [(flag: NSEvent.ModifierFlags, keyCodes: [UInt16])] = [
        (.shift, [0x38, 0x3C]), (.control, [0x3B, 0x3E]), (.option, [0x3A, 0x3D]), (.command, [0x37, 0x36]), (.function, [0x3F]),
    ]
    private static let capsLockKeyCode: UInt16 = 0x39

    /// The Linux keys the guest holds down, as far as this state has sent them.
    private var held: Set<UInt16> = []

    /// A key went down. A key the table does not map, and the repeat of a held key (the guest repeats by itself), send nothing.
    mutating func press(keyCode: UInt16, isRepeat: Bool) -> [InputEvent] {
        guard !isRepeat, let key = VirtioKeyMap.linuxKey(forMacKeyCode: keyCode) else { return [] }
        held.insert(key)
        return InputEvents.key(key, .press)
    }

    /// A key went up. Only a key this state sent down is released: a key pressed before the window had the keyboard, or one
    /// taken by a menu shortcut, has nothing to release.
    mutating func release(keyCode: UInt16) -> [InputEvent] {
        guard let key = VirtioKeyMap.linuxKey(forMacKeyCode: keyCode), held.remove(key) != nil else { return [] }
        return InputEvents.key(key, .release)
    }

    /// A modifier key changed; `keyCode` says which, `flags` the modifiers now. A flag that is clear means none of its keys is
    /// down (every one held is released); a flag that is set means this key went down, or went up while the other key of its
    /// pair still holds the flag. Caps lock reports once for each change of its state, which the guest's own caps lock toggles
    /// on a tap.
    mutating func modifiersChanged(keyCode: UInt16, flags: NSEvent.ModifierFlags) -> [InputEvent] {
        guard let key = VirtioKeyMap.linuxKey(forMacKeyCode: keyCode) else { return [] }
        if keyCode == Self.capsLockKeyCode { return InputEvents.key(key, .press) + InputEvents.key(key, .release) }
        guard let modifier = Self.modifierKeys.first(where: { $0.keyCodes.contains(keyCode) }) else { return [] }
        if flags.contains(modifier.flag) {
            return held.contains(key) ? release(keyCode: keyCode) : press(keyCode: keyCode, isRepeat: false)
        }
        return modifier.keyCodes.flatMap { release(keyCode: $0) }
    }

    /// A key-up for every key held, for a window that lost the keyboard: it would never see the keys come up.
    mutating func releaseAll() -> [InputEvent] {
        let keys = held.sorted()
        held = []
        return keys.flatMap { InputEvents.key($0, .release) }
    }
}

/// The pointer as the guest's virtio tablet sees it: where it is, which buttons are down, and the wheel.
struct PointerState {
    /// How many points of a trackpad's scroll make one notch of the wheel; a mouse wheel reports lines, one notch each. The
    /// real run shows whether the guest scrolls too fast or too slowly with these.
    static let pointsPerNotch = 16.0

    private var held: Set<PointerButton> = []
    /// The part of a notch scrolled and not yet sent.
    private var carry = 0.0

    /// The pointer at `point` of a view of `view` points, origin at the bottom left as AppKit's, over a picture that occupies
    /// `picture` (origin at the top left, `ScreenPlacement.pictureRect`). A point outside the picture lands on its nearest edge.
    /// Nothing is sent while there is no picture.
    static func position(of point: CGPoint, picture: CGRect, in view: CGSize) -> [InputEvent] {
        guard picture != .zero else { return [] }
        return InputEvents.pointer(
            x: point.x - picture.minX, y: view.height - point.y - picture.minY, viewWidth: picture.width, viewHeight: picture.height,
            range: VirtioInputStaticConfig.tabletRange)
    }

    /// The tablet's button for a mouse button of the window: the left and right ones, and the third (`buttonNumber` 2) as the
    /// middle one. The others have no button on the tablet.
    static func button(_ button: GuestMouseButton, number: Int) -> PointerButton? {
        switch button {
        case .left: .left
        case .right: .right
        case .other: number == 2 ? .middle : nil
        }
    }

    mutating func press(_ button: PointerButton) -> [InputEvent] {
        held.insert(button)
        return InputEvents.button(button, pressed: true)
    }

    /// Only a button this state sent down is released.
    mutating func release(_ button: PointerButton) -> [InputEvent] {
        guard held.remove(button) != nil else { return [] }
        return InputEvents.button(button, pressed: false)
    }

    /// The wheel turned by `delta` (a scroll's vertical delta, positive scrolling up): the whole notches it adds up to, with the
    /// rest carried to the next. Something that is not a number sends nothing.
    mutating func scroll(delta: Double, precise: Bool) -> [InputEvent] {
        guard delta.isFinite else { return [] }
        carry += precise ? delta / Self.pointsPerNotch : delta
        let notches = carry.rounded(.towardZero)
        carry -= notches
        // A wheel does not turn by a billion notches: the clamp only keeps the conversion from trapping.
        return InputEvents.scroll(notches: Int32(min(max(notches, -1e9), 1e9)))
    }

    /// A release for every button held, for a window that lost the pointer.
    mutating func releaseAll() -> [InputEvent] {
        let buttons = held.sorted { $0.rawValue < $1.rawValue }
        held = []
        carry = 0
        return buttons.flatMap { InputEvents.button($0, pressed: false) }
    }
}
````

````swift
import Foundation
import VirtioInput
import Virtualization

/// The host side of a virtio input device for a Linux machine, on the framework's custom virtio devices (macOS 27): a keyboard
/// or an absolute pointer, each a device of its own with the fixed configuration space of the library
/// (`VirtioInputStaticConfig`). The framework calls it on the main queue, so its state is main-actor state. The rules of the two
/// queues are `VirtioInputEndpoint`'s; this class gives it the framework's queues, tells it about the device's life, and makes
/// the configuration. The configuration conformance is `@preconcurrency` for the reason `CustomGPUDevice`'s is.
@available(macOS 27.0, *)
@MainActor
final class VirtioInputDevice: NSObject, @preconcurrency VZCustomVirtioDeviceConfigurationDelegate, VZCustomVirtioDeviceDelegate,
    InputEventSink
{
    /// What can go wrong, each logged once so that a guest cannot make the log grow.
    private enum Trouble {
        case dropped
        case malformed
    }

    private let layout: VirtioInputStaticConfig
    private let label: String
    private let endpoint = VirtioInputEndpoint<GuestQueue>()
    /// The queues with a later turn already scheduled, so a busy guest does not pile up tasks.
    private var resuming: Set<UInt16> = []
    private var troubles = TroubleLog<Trouble>(Log.input)
    /// While the machine is paused the guest runs nothing, and events are not delivered.
    private var isPaused = false

    /// `label` names the device in the log.
    init(layout: VirtioInputStaticConfig, label: String) {
        self.layout = layout
        self.label = label
    }

    /// The device as the machine's configuration takes it (`customVirtioDevices`): device ID 18, PCI class 0x09 subclass 0x80,
    /// an event and a status queue, only `VIRTIO_F_VERSION_1`, the 136 bytes of the layout, and no save and restore.
    nonisolated func makeConfiguration() -> VZCustomVirtioDeviceConfiguration {
        VZCustomVirtioDeviceConfiguration(
            deviceID: InputWire.deviceID, pciClass: (InputWire.pciClassID, InputWire.pciSubclassID), queueCount: InputWire.queueCount,
            versionOneSubset: InputWire.versionOneFeatureSubset1, space: layout.bytes, delegate: self)
    }

    /// Frames of events for the guest (`VirtioInputEndpoint.send`); none while the machine is paused.
    func send(_ frames: [InputEvent]) {
        guard !isPaused else { return }
        endpoint.send(frames)
        if endpoint.droppedFrameCount > 0 {
            troubles.notice(.dropped, "The guest's \(label) isn't reading events fast enough: input was dropped")
        }
    }

    func customVirtioConfiguration(_ configuration: VZCustomVirtioDeviceConfiguration, didCreateDevice device: VZCustomVirtioDevice) {
        device.delegate = self
    }

    func customVirtioDevice(_ device: VZCustomVirtioDevice, didReceiveNotificationFor queue: VZVirtioQueue) {
        serveTurn(of: queue)
    }

    /// Serves up to `CustomGPUDevice.turnBudget` elements of a queue. A guest that notifies once for a batch does not notify
    /// again for the rest of it, so when the budget ended the turn another one is scheduled on the main actor.
    private func serveTurn(of queue: VZVirtioQueue) {
        let elements = GuestQueue(queue: queue)
        let more: Bool
        switch queue.queueIndex {
        case InputWire.eventQueue:
            more = endpoint.takeBuffers(from: elements, budget: CustomGPUDevice.turnBudget)
        case InputWire.statusQueue:
            more = endpoint.serveStatus(from: elements, budget: CustomGPUDevice.turnBudget)
            if endpoint.malformedStatusCount > 0 {
                troubles.notice(.malformed, "The guest's \(label) wrote a status message of the wrong length")
            }
        default:
            return
        }
        guard more, resuming.insert(queue.queueIndex).inserted else { return }
        Task { [weak self] in
            guard let self else { return }
            resuming.remove(queue.queueIndex)
            serveTurn(of: queue)
        }
    }

    /// exception in the framework, so they are let go. Events not yet delivered are stale and go with them.
````

````swift
import Foundation

@testable import VirtioGPU

/// A queue element of the input device that records what is done to it.
final class FakeQueueElement: QueueElement {
    struct Failure: Error {}

    let readableByteCount: Int
    let writableByteCount: Int
    private let content: [UInt8]
    private let failsToWrite: Bool
    private let failsToRead: Bool
    private(set) var written: [[UInt8]] = []
    private(set) var reads: [Int] = []
    private(set) var givenBackCount = 0

    /// A buffer the guest posted for events: `room` bytes to write, nothing to read.
    init(room: Int = 8, failsToWrite: Bool = false) {
        (readableByteCount, writableByteCount, content, self.failsToWrite, failsToRead) = (0, room, [], failsToWrite, false)
    }

    /// A message the guest wrote to the status queue: `length` bytes claimed, `content` read back.
    init(wrote length: Int, content: [UInt8] = [], failsToRead: Bool = false) {
        (readableByteCount, writableByteCount, self.content, failsToWrite, self.failsToRead) = (length, 0, content, false, failsToRead)
    }

    func read(count: Int) throws -> [UInt8] {
        reads.append(count)
        if failsToRead { throw Failure() }
        return Array((content + Array(repeating: 0, count: max(0, count - content.count))).prefix(count))
    }

    func write(_ bytes: [UInt8]) throws {
        if failsToWrite { throw Failure() }
        written.append(bytes)
    }

    func giveBack() {
        givenBackCount += 1
    }
}

/// A queue that hands out the elements it is given, once each, in order.
final class FakeInputQueue: ElementQueue {
    private var waiting: [FakeQueueElement]

    init(_ elements: [FakeQueueElement]) {
        waiting = elements
    }

    func nextElement() -> FakeQueueElement? {
        waiting.isEmpty ? nil : waiting.removeFirst()
    }
}
````
