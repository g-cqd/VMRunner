```swift
import Foundation

/// A container's settings while they are typed: the command and the environment are text until they are read back into a
/// `ContainerSpec`. The New Machine sheet and the editor of a stored machine both go through it, so the same text gives the
/// same settings and the same refusals.
struct ContainerDraft: Equatable {
    var spec: ContainerSpec
    var commandLine: String
    var environmentText: String
    /// Whether the command and the environment can be edited as text. It is false when the stored command or environment does not
    /// survive being written as text and read back (an argument with a double quote, a value with a carriage return): editing
    /// the text would change what is stored, so the draft then keeps the stored command and environment as they are.
    let textIsEditable: Bool

    /// A draft of text the user typed, which `applied(to:)` reads.
    init(spec: ContainerSpec, commandLine: String, environmentText: String) {
        self.spec = spec
        self.commandLine = commandLine
        self.environmentText = environmentText
        textIsEditable = true
    }

    /// A draft of a stored spec, showing its command and environment as text.
    init(_ spec: ContainerSpec) {
        self.spec = spec
        commandLine = ContainerInput.line(from: spec.command)
        environmentText = ContainerInput.text(from: spec.environment)
        textIsEditable =
            (try? ContainerInput.arguments(from: commandLine)) == spec.command
            && (try? ContainerInput.environment(from: environmentText)) == spec.environment
    }

    /// `config` with the draft's settings, checked by the same rules a stored configuration is. Throws the reason the text or
    /// the settings cannot be used; read it with `VMConfig.reason(for:)`.
    func applied(to config: VMConfig) throws -> VMConfig {
        var settings = spec
        settings.image = settings.image.trimmingCharacters(in: .whitespacesAndNewlines)
        if textIsEditable {
            settings.command = try ContainerInput.arguments(from: commandLine)
            settings.environment = try ContainerInput.environment(from: environmentText)
        }
        var machine = config
        machine.container = settings
        try machine.validate()
        return machine
    }
}

extension VMConfig {
    /// A configuration check's reason, without the framework's generic wording around it.
    static func reason(for error: any Error) -> String {
        if case DecodingError.dataCorrupted(let context) = error { return context.debugDescription }
        return error.localizedDescription
    }
}
```

```swift
import Foundation

/// The words alerts and tooltips use for a machine, by what it runs: a virtual machine keeps the wording it always had, and a
/// container is called a container.
enum MachineWording {
    /// The start of the report the machine state makes when a machine stops by itself with an error.
    static let virtualMachineStoppedPrefix = "The virtual machine stopped: "

    static func errorTitle(for workload: WorkloadKind) -> String {
        workload == .container ? "Container Error" : "Virtual Machine Error"
    }

    static func startHelp(for workload: WorkloadKind, hasSavedState: Bool) -> String {
        if workload == .container { return "Start the container" }
        return hasSavedState ? "Resume from suspended state" : "Start the virtual machine"
    }

    /// `message` as the user should read it for a machine that runs `workload`: the report of a machine that stopped by itself
    /// names a container as one; any other report is already neutral.
    static func report(_ message: String, for workload: WorkloadKind) -> String {
        guard workload == .container, message.hasPrefix(virtualMachineStoppedPrefix) else { return message }
        return "The container stopped: " + message.dropFirst(virtualMachineStoppedPrefix.count)
    }

    /// What the delete dialog says is removed.
    static func deleteDetail(for workload: WorkloadKind) -> String {
        workload == .container
            ? "The machine's folder, with the container's working files and image cache, is permanently removed."
            : "The virtual disk and all its contents are permanently removed."
    }

    /// The title of the alert shown when the app quits while machines run.
    static func quitTitle(virtualMachines: Int, containers: Int) -> String {
        let total = virtualMachines + containers
        let noun =
            containers == 0 ? "virtual machine" : virtualMachines == 0 ? "container" : "machine"
        return "\(total) \(noun)\(total == 1 ? " is" : "s are") still running."
    }

    /// What quitting does to the machines that run, which differs for a container: it is asked to stop, not cut off.
    static func quitDetail(virtualMachines: Int, containers: Int) -> String {
        let virtualMachineText =
            "Quitting force-stops them, like pulling the power cord. Suspend or shut them down first to avoid data loss."
        guard containers > 0 else { return virtualMachineText }
        let containerText = "Quitting asks each container to stop, and kills it if it has not ended after 5 seconds."
        guard virtualMachines > 0 else { return containerText }
        return "Quitting force-stops the virtual machines, like pulling the power cord. Suspend or shut them down first to avoid "
            + "data loss. " + containerText
    }
}
```

```swift
import Foundation

/// The sections of a stopped machine's settings form, in the order they are shown.
enum EditorSection: CaseIterable, Identifiable {
    case general, container, security, compute, storage, installer, provisioning, display, input, audio, network, networkPolicy
    case identity, sharing, hostInfo, boot, devices, agentAccess, snapshots

    var id: Self { self }

    /// The sections that apply to `config`, in order. A container machine has a general section, its container settings and
    /// agent access (the control service starts and stops it); everything else is about a virtual machine's hardware.
    static func visible(for config: VMConfig) -> [Self] {
        allCases.filter { $0.isShown(for: config) }
    }

    private func isShown(for config: VMConfig) -> Bool {
        if config.workload == .container { return Self.containerSections.contains(self) }
        switch self {
        case .container: return false
        case .installer, .provisioning: return config.guest == .linux
        case .networkPolicy: return config.networkMode == .filtered
        case .boot: return config.guest == .macOS
        default: return true
        }
    }

    private static let containerSections: Set<Self> = [.general, .container, .agentAccess]
}
```

```swift
import SwiftUI

/// The container section over a stored machine. The settings are edited as a draft, and each change that is valid is saved
/// through the configuration binding like any other editor change; one that is not valid is explained and not saved.
struct ContainerEditorSection: View {
    @Binding var config: VMConfig
    private let policy: ManagedPreferences
    @State private var draft: ContainerDraft
    @State private var problem: String?

    init(config: Binding<VMConfig>, policy: ManagedPreferences = .current) {
        _config = config
        self.policy = policy
        _draft = State(initialValue: ContainerDraft(config.wrappedValue.container ?? ContainerSpec()))
    }

    var body: some View {
        ContainerSection(
            container: $draft.spec, commandLine: $draft.commandLine, environmentText: $draft.environmentText,
            problem: problem, textIsEditable: draft.textIsEditable, isLocked: policy.isLocked(.allowContainers)
        )
        .onChange(of: draft) { _, draft in save(draft) }
    }

    private func save(_ draft: ContainerDraft) {
        do {
            config = try draft.applied(to: config)
            problem = nil
        } catch {
            problem = VMConfig.reason(for: error)
        }
    }
}
```
