- **Design (per ADR-0013, which stays `proposed`):** `ManagedPreferences` is a `Sendable` value read once from a
  `UserDefaults` (by default `.standard`, the app's own domain, which a configuration profile targets; a `VMRUNNER_DATA_DIR`
  run reads the same policy). Only a value the system reports as forced (`objectIsForced(forKey:)`) is policy; a value the
  user wrote never is. A forced value of the wrong type fails closed (it restricts). The first set of keys:
  `AllowSharedFolders`, `AllowClipboardSharing`, `AllowMicrophone`, `AllowSuspend`, `AllowHostInformation`,
  `RequireDisposableMachines`, `MaxRuntimeMinutes`. `VMLibrary` clamps every configuration it loads, adds or changes, so a
  start (which reads the library) never sees a configuration outside the policy. Tests use an injected `forcedValue`
  closure and a suite-name `UserDefaults` (a suite cannot force a key, which is exactly what that test shows).

     ```swift
     import Foundation

     /// A machine setting an organisation can restrict through managed preferences (ADR-0013). The raw value is the key a
     /// configuration profile forces in the app's domain.
     enum ManagedKey: String, CaseIterable, Sendable {
         case allowSharedFolders = "AllowSharedFolders"
         case allowClipboardSharing = "AllowClipboardSharing"
         case allowMicrophone = "AllowMicrophone"
         case allowSuspend = "AllowSuspend"
         case allowHostInformation = "AllowHostInformation"
         case requireDisposableMachines = "RequireDisposableMachines"
         case maxRuntimeMinutes = "MaxRuntimeMinutes"
     }

     /// What an organisation forced in the app's managed preferences, read once. Only a forced value is policy
     /// (`UserDefaults.objectIsForced(forKey:)`); a value the user wrote never is. A forced value of the wrong type restricts.
     struct ManagedPreferences: Equatable, Sendable {
         /// The `Allow…` keys forced to anything but `true`.
         private(set) var denied: Set<ManagedKey> = []
         /// `RequireDisposableMachines` forced to anything but `false`.
         private(set) var requiresDisposable = false
         /// The longest run `MaxRuntimeMinutes` allows, when it is forced.
         private(set) var maxRuntimeMinutes: Int?

         /// Reads the keys forced in `defaults`: by default the app's own domain, which a configuration profile targets, so a
         /// run that keeps its settings in a domain of its own (`VMRUNNER_DATA_DIR`) obeys the same policy.
         init(reading defaults: UserDefaults = .standard) {
             self.init(forcedValue: { defaults.objectIsForced(forKey: $0) ? defaults.object(forKey: $0) : nil })
         }

         /// `forcedValue` returns a key's forced value, or nil when the key is not forced.
         init(forcedValue: (String) -> Any?) {
             for key in ManagedKey.allCases {
                 guard let value = forcedValue(key.rawValue) else { continue }
                 switch key {
                 case .requireDisposableMachines:
                     requiresDisposable = (value as? Bool) != false
                 case .maxRuntimeMinutes:
                     // A value that is not a whole number, or lies outside what the editor offers, becomes the shortest limit.
                     let offered = EditorLimits.runtimeStepMinutes...EditorLimits.runtimeMinutes.upperBound
                     maxRuntimeMinutes = (value as? Int).flatMap { offered.contains($0) ? $0 : nil } ?? EditorLimits.runtimeStepMinutes
                 default:
                     if (value as? Bool) != true { denied.insert(key) }
                 }
             }
         }

         /// `config` with every forced restriction applied; whatever the policy allows is left as it is.
         func clamp(_ config: VMConfig) -> VMConfig {
             var config = config
             if denied.contains(.allowSharedFolders) { config.sharedFolders = [] }
             if denied.contains(.allowClipboardSharing) { config.sharesClipboard = false }
             if denied.contains(.allowMicrophone) { config.audioInput = false }
             if denied.contains(.allowSuspend) { config.allowsSuspend = false }
             if denied.contains(.allowHostInformation) { config.hostInfo = HostInfoSharing() }
             if requiresDisposable { config.ephemeral = true }
             if let limit = maxRuntimeMinutes, config.maxRuntimeMinutes == 0 || config.maxRuntimeMinutes > limit {
                 config.maxRuntimeMinutes = limit
             }
             return config
         }
     }
     ```

     ```swift
     import Foundation
     import Testing

     @testable import VMRunner

     /// Managed preferences (ADR-0013): only a forced value is policy, a forced value of the wrong type restricts, and what the
     /// policy allows is left alone.
     struct ManagedPreferencesTests {
         private func forced(_ values: [String: Any]) -> ManagedPreferences {
             ManagedPreferences(forcedValue: { values[$0] })
         }

         private func openMachine() -> VMConfig {
             var config = VMConfig.new(name: "m", guest: .linux)
             config.sharedFolders = [SharedFolder(path: "/tmp/a")]
             config.sharesClipboard = true
             config.audioInput = true
             config.allowsSuspend = true
             config.hostInfo.battery = true
             config.maxRuntimeMinutes = 0
             return config
         }

         @Test func nothingForcedChangesNothing() {
             let config = openMachine()
             #expect(ManagedPreferences(forcedValue: { _ in nil }).clamp(config) == config)
         }

         @Test func everyForcedRestrictionIsApplied() {
             let policy = forced([
                 "AllowSharedFolders": false, "AllowClipboardSharing": false, "AllowMicrophone": false, "AllowSuspend": false,
                 "AllowHostInformation": false, "RequireDisposableMachines": true, "MaxRuntimeMinutes": 60,
             ])
             let clamped = policy.clamp(openMachine())
             #expect(clamped.sharedFolders.isEmpty && !clamped.sharesClipboard && !clamped.audioInput && !clamped.allowsSuspend)
             #expect(clamped.hostInfo.isEmpty && clamped.ephemeral && clamped.maxRuntimeMinutes == 60)
         }

         @Test func whatThePolicyAllowsIsKept() {
             var config = openMachine()
             config.maxRuntimeMinutes = 30
             let policy = forced(["AllowSharedFolders": true, "RequireDisposableMachines": false, "MaxRuntimeMinutes": 60])
             #expect(policy.clamp(config) == config, "allowed stays, and a shorter limit of the user's own is kept")
         }

         @Test func aForcedValueOfTheWrongTypeRestricts() {
             let policy = forced(["AllowMicrophone": "yes", "RequireDisposableMachines": "no", "MaxRuntimeMinutes": "lots"])
             let clamped = policy.clamp(openMachine())
             #expect(!clamped.audioInput && clamped.ephemeral)
             #expect(clamped.maxRuntimeMinutes == EditorLimits.runtimeStepMinutes)
         }

         @Test func aValueTheUserWroteIsNotPolicy() throws {
             let name = "ManagedPreferencesTests-\(UUID().uuidString)"
             let suite = try #require(UserDefaults(suiteName: name))
             defer { UserDefaults.standard.removePersistentDomain(forName: name) }
             for key in ManagedKey.allCases { suite.set(false, forKey: key.rawValue) }
             #expect(ManagedPreferences(reading: suite) == ManagedPreferences(forcedValue: { _ in nil }))
         }
     }

     /// The library brings every configuration it adds, changes or loads within the policy.
     @MainActor
     struct ManagedLibraryTests {
         @Test func theLibraryAppliesThePolicyToWhatItAddsChangesAndLoads() throws {
             let directory = try scratchDirectory()
             defer { try? FileManager.default.removeItem(at: directory) }
             let policy = ManagedPreferences(forcedValue: { $0 == ManagedKey.allowSharedFolders.rawValue ? false as Any : nil })
             let library = VMLibrary(machinesDirectory: directory, saveDelay: .seconds(3600), managed: policy)
             var config = VMConfig.new(name: "m", guest: .linux)
             config.sharedFolders = [SharedFolder(path: "/tmp/a")]
             try library.add(config)
             #expect(library.machine(config.id)?.sharedFolders.isEmpty == true)
             library.update(config)
             #expect(library.machine(config.id)?.sharedFolders.isEmpty == true)

             // A file written before the policy arrived is brought within it as it loads.
             try config.encoded().writePrivately(to: library.bundle(config.id).config)
             let reloaded = VMLibrary(machinesDirectory: directory, managed: policy)
             reloaded.load()
             #expect(reloaded.machine(config.id)?.sharedFolders.isEmpty == true)
         }
     }
     ```
