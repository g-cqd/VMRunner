/// Read access to the guest's RAM, as the device sees it. The guest may change its memory at any time, so an implementation
/// returns a copy, and the store uses only that copy.
public protocol GuestMemoryReader {
    /// A copy of `count` bytes of guest RAM starting at the guest physical `address`, or `nil` when that range is not guest
    /// RAM.
    func copyBytes(atPhysicalAddress address: UInt64, count: Int) -> [UInt8]?
}

/// A piece of guest RAM: where it starts and how many bytes it has.
public struct GuestRange: Equatable, Sendable {
    public var address: UInt64
    public var count: Int

    public init(address: UInt64, count: Int) {
        self.address = address
        self.count = count
    }
}

/// The guest memory a resource's pixels live in: its entries, end to end, as one run of bytes.
public struct BackingLayout: Equatable, Sendable {
    public let entries: [GPUMemoryEntry]
    /// Where each entry starts in the run.
    private let starts: [UInt64]
    /// The length of the run.
    public let totalLength: UInt64

    /// `nil` for an entry of zero length or one that wraps past the top of the address space (the decoder already refuses
    /// both), or a total that overflows.
    public init?(entries: [GPUMemoryEntry]) {
        var starts: [UInt64] = []
        starts.reserveCapacity(entries.count)
        var total: UInt64 = 0
        for entry in entries {
            guard entry.length > 0, !entry.address.addingReportingOverflow(UInt64(entry.length)).overflow else { return nil }
            starts.append(total)
            let (sum, overflow) = total.addingReportingOverflow(UInt64(entry.length))
            guard !overflow else { return nil }
            total = sum
        }
        self.entries = entries
        self.starts = starts
        self.totalLength = total
    }

    /// The pieces of guest memory that hold bytes `offset ..< offset + length` of the run, in order; `nil` when the range
    /// ends past the run or its end overflows. A zero length gives no pieces.
    public func pieces(offset: UInt64, length: UInt64) -> [GuestRange]? {
        let (end, overflow) = offset.addingReportingOverflow(length)
        guard !overflow, end <= totalLength else { return nil }
        guard length > 0 else { return [] }
        var pieces: [GuestRange] = []
        var index = entryIndex(containing: offset)
        var position = offset
        while position < end, index < entries.count {
            let skip = position - starts[index]
            let available = UInt64(entries[index].length) - skip
            let take = min(available, end - position)
            pieces.append(GuestRange(address: entries[index].address + skip, count: Int(take)))
            position += take
            index += 1
        }
        return pieces
    }

    /// The last entry that starts at or before `offset` (a binary search: a guest controls the number of entries).
    private func entryIndex(containing offset: UInt64) -> Int {
        var low = 0
        var high = starts.count
        while high - low > 1 {
            let middle = (low + high) / 2
            if starts[middle] <= offset { low = middle } else { high = middle }
        }
        return low
    }
}

/// The 32-bit pixel formats a 2D resource may have; any other value is refused when the resource is created.
public enum GPUPixelFormat: UInt32, Sendable, CaseIterable {
    case b8g8r8a8 = 1
    case b8g8r8x8 = 2
    case a8r8g8b8 = 3
    case x8r8g8b8 = 4
    case r8g8b8a8 = 67
    case x8b8g8r8 = 68
    case a8b8g8r8 = 121
    case r8g8b8x8 = 134

    public static let bytesPerPixel = 4
}

/// A 2D resource: its host copy of the pixels, and the guest memory they are transferred from.
public struct GPUResource: Equatable, Sendable {
    public let id: UInt32
    public let format: GPUPixelFormat
    public let width: UInt32
    public let height: UInt32
    /// `width × height × 4` bytes, rows `stride` apart.
    public internal(set) var pixels: [UInt8]
    public internal(set) var backing: BackingLayout?

    public var stride: Int { Int(width) * GPUPixelFormat.bytesPerPixel }
}

/// The device's 2D resources, under `GPULimits`. Every operation answers with the response type the guest gets: `okNoData`
/// when it was done, an error otherwise, and an operation that fails changes nothing.
public struct ResourceStore: Sendable {
    public let limits: GPULimits
    private var resources: [UInt32: GPUResource] = [:]
    /// The bytes of every resource's host copy together.
    public private(set) var totalBytes = 0

    public init(limits: GPULimits) {
        self.limits = limits
    }

    public var count: Int { resources.count }
    public var isEmpty: Bool { resources.isEmpty }

    public func resource(_ id: UInt32) -> GPUResource? {
        resources[id]
    }

    /// Creates a zero-filled resource. The size is computed with overflow checks and held against the limits before the
    /// host copy is allocated.
    public mutating func create2D(resourceID id: UInt32, format: UInt32, width: UInt32, height: UInt32) -> GPUResponseType {
        guard id != 0, resources[id] == nil else { return .errInvalidResourceID }
        guard let format = GPUPixelFormat(rawValue: format), width > 0, height > 0 else { return .errInvalidParameter }
        let (pixels, wide) = Int(width).multipliedReportingOverflow(by: Int(height))
        let (bytes, overflow) = pixels.multipliedReportingOverflow(by: GPUPixelFormat.bytesPerPixel)
        guard !wide, !overflow, bytes <= limits.maxResourceBytes else { return .errOutOfMemory }
        guard resources.count < limits.maxResources, bytes <= limits.maxTotalResourceBytes - totalBytes else { return .errOutOfMemory }
        resources[id] = GPUResource(
            id: id, format: format, width: width, height: height, pixels: [UInt8](repeating: 0, count: bytes), backing: nil)
        totalBytes += bytes
        return .okNoData
    }

    public mutating func unref(_ id: UInt32) -> GPUResponseType {
        guard let removed = resources.removeValue(forKey: id) else { return .errInvalidResourceID }
        totalBytes -= removed.pixels.count
        return .okNoData
    }

    public mutating func attachBacking(_ id: UInt32, entries: [GPUMemoryEntry]) -> GPUResponseType {
        guard var resource = resources[id] else { return .errInvalidResourceID }
        guard resource.backing == nil else { return .errUnspec }
        guard entries.count <= limits.maxBackingEntries, let layout = BackingLayout(entries: entries) else { return .errInvalidParameter }
        resource.backing = layout
        resources[id] = resource
        return .okNoData
    }

    public mutating func detachBacking(_ id: UInt32) -> GPUResponseType {
        guard var resource = resources[id] else { return .errInvalidResourceID }
        guard resource.backing != nil else { return .errUnspec }
        resource.backing = nil
        resources[id] = resource
        return .okNoData
    }

    /// Copies `rect` of the resource from its backing. As the guest's driver lays it out, `offset` is where the rectangle's
    /// first pixel is in the backing (it already counts `y × stride + x × 4`), and each next row is `stride` further on.
    /// Every row is read into a temporary copy first; the host copy is written only once all of them were read, so a read
    /// that fails leaves the resource as it was.
    public mutating func transferToHost2D(resourceID id: UInt32, rect: GPURect, offset: UInt64, memory: some GuestMemoryReader)
        -> GPUResponseType
    {
        guard id != 0, var resource = resources[id] else { return .errInvalidResourceID }
        guard rect.fits(width: resource.width, height: resource.height) else { return .errInvalidParameter }
        guard let backing = resource.backing else { return .errUnspec }
        guard rect.width > 0, rect.height > 0 else { return .okNoData }
        let rowBytes = Int(rect.width) * GPUPixelFormat.bytesPerPixel
        let stride = UInt64(resource.stride)
        var rows: [UInt8] = []
        rows.reserveCapacity(rowBytes * Int(rect.height))
        for row in 0..<UInt64(rect.height) {
            let (skip, wide) = row.multipliedReportingOverflow(by: stride)
            let (start, overflow) = offset.addingReportingOverflow(skip)
            guard !wide, !overflow, let pieces = backing.pieces(offset: start, length: UInt64(rowBytes)) else {
                return .errInvalidParameter
            }
            for piece in pieces {
                guard let bytes = memory.copyBytes(atPhysicalAddress: piece.address, count: piece.count), bytes.count == piece.count
                else { return .errUnspec }
                rows.append(contentsOf: bytes)
            }
        }
        resources[id] = nil
        for row in 0..<Int(rect.height) {
            let destination = (Int(rect.y) + row) * resource.stride + Int(rect.x) * GPUPixelFormat.bytesPerPixel
            resource.pixels.replaceSubrange(destination..<destination + rowBytes, with: rows[row * rowBytes..<(row + 1) * rowBytes])
        }
        resources[id] = resource
        return .okNoData
    }

    /// Drops every resource, as a device reset does.
    public mutating func removeAll() {
        resources.removeAll()
        totalBytes = 0
    }
}

struct SeededGenerator: RandomNumberGenerator {
    private var state: UInt64

    init(seed: UInt64) {
        state = seed
    }

    mutating func next() -> UInt64 {
        state &+= 0x9E37_79B9_7F4A_7C15
        var value = state
        value = (value ^ (value >> 30)) &* 0xBF58_476D_1CE4_E5B9
        value = (value ^ (value >> 27)) &* 0x94D0_49BB_1331_11EB
        return value ^ (value >> 31)
    }
}

/// The bytes of a request: a header, then each field as a little-endian 32-bit word, then `tail`.
func request(
    _ type: UInt32, flags: UInt32 = 0, fence: UInt64 = 0, context: UInt32 = 0, ring: UInt8 = 0, words: [UInt32] = [],
    tail: [UInt8] = []
) -> [UInt8] {
    var writer = ByteWriter(capacity: 64)
    writer.u32(type)
    writer.u32(flags)
    writer.u64(fence)
    writer.u32(context)
    writer.u8(ring)
    writer.zeros(3)
    for word in words { writer.u32(word) }
    return writer.bytes + tail
}

func request(_ type: GPUCommandType, words: [UInt32] = [], tail: [UInt8] = []) -> [UInt8] {
    request(type.rawValue, words: words, tail: tail)
}

/// Memory entries as an attach-backing request carries them.
func entryBytes(_ entries: [(address: UInt64, length: UInt32)]) -> [UInt8] {
    var writer = ByteWriter(capacity: entries.count * 16)
    for entry in entries {
        writer.u64(entry.address)
        writer.u32(entry.length)
        writer.u32(0)
    }
    return writer.bytes
}

/// A transfer request: rectangle, 64-bit offset, resource.
func transfer(_ rect: GPURect, offset: UInt64, resource: UInt32) -> [UInt8] {
    let words = [
        rect.x, rect.y, rect.width, rect.height, UInt32(truncatingIfNeeded: offset), UInt32(truncatingIfNeeded: offset >> 32), resource, 0,
    ]
    return request(.transferToHost2D, words: words)
}

/// Reads the little-endian word at `offset` of an answer.
func word(_ bytes: [UInt8], at offset: Int) -> UInt32 {
    (0..<4).reduce(0) { $0 | UInt32(bytes[offset + $1]) << (8 * $1) }
}

/// Guest RAM made of regions; a read is answered only when it lies inside one region. Each byte of a region is its offset
/// in the region plus the region's `seed`, so a test can tell which bytes were copied.
struct FakeGuestMemory: GuestMemoryReader {
    var regions: [(address: UInt64, bytes: [UInt8])] = []

    mutating func add(at address: UInt64, count: Int, seed: UInt8 = 0) {
        regions.append((address, (0..<count).map { UInt8(truncatingIfNeeded: $0) &+ seed }))
    }

    func copyBytes(atPhysicalAddress address: UInt64, count: Int) -> [UInt8]? {
        for region in regions where address >= region.address {
            let start = address - region.address
            guard start <= UInt64(region.bytes.count), UInt64(count) <= UInt64(region.bytes.count) - start else { continue }
            return Array(region.bytes[Int(start)..<Int(start) + count])
        }
        return nil
    }
}
