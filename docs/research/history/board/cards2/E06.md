     ```swift
     /// What this Mac reports, as the host-information share publishes it. A plain value, so the files can be checked without
     /// a battery or a network.
     struct HostFacts: Equatable, Sendable {
         enum PowerSource: String, Sendable { case ac, battery, unknown }
         enum Thermal: String, Sendable { case nominal, fair, serious, critical }

         /// The internal battery's charge in percent; nil on a Mac without one, or when it can't be read.
         var batteryPercent: Int?
         var powerSource = PowerSource.unknown
         var lowPowerMode = false
         var thermal = Thermal.nominal
         /// The Mac's network path; nil until the first sample.
         var network: HostNetworkPath?
         /// The time zone's identifier, such as `Europe/Paris`.
         var timeZone = ""

         /// The share's file names, one per category.
         static let fileNames = ["battery", "power", "thermal", "network", "timezone"]

         /// The text of the file of each category `sharing` turns on, one `key=value` per line. A category that is off has no
         /// file.
         func files(for sharing: HostInfoSharing) -> [String: String] {
             var files: [String: String] = [:]
             if sharing.battery { files["battery"] = "percent=\(batteryPercent.map { String($0) } ?? "none")\n" }
             if sharing.power { files["power"] = "source=\(powerSource.rawValue)\nlowPowerMode=\(lowPowerMode ? 1 : 0)\n" }
             if sharing.thermal { files["thermal"] = "state=\(thermal.rawValue)\n" }
             if sharing.networkPath { files["network"] = network.map(Self.networkText) ?? "status=unknown\n" }
             if sharing.timeZone { files["timezone"] = "identifier=\(timeZone)\n" }
             return files
         }

         private static func networkText(_ path: HostNetworkPath) -> String {
             "status=\(path.isSatisfied ? "satisfied" : "unsatisfied")\ninterface=\(path.interface.rawValue)\n"
                 + "expensive=\(path.isExpensive ? 1 : 0)\nconstrained=\(path.isConstrained ? 1 : 0)\n"
         }
     }
     ```
  3. Create `Sources/VMRunner/Engine/HostFacts+System.swift`:
     ```swift
     import Foundation

     extension HostFacts {
         /// What this Mac reports now. `network` is the latest path sample, which the caller keeps as the path changes.
         static func current(network: HostNetworkPath?) -> HostFacts {
             let info = ProcessInfo.processInfo
             return HostFacts(
                 lowPowerMode: info.isLowPowerModeEnabled, thermal: Thermal(info.thermalState), network: network,
                 timeZone: TimeZone.current.identifier)
         }
     }

     extension HostFacts.Thermal {
         /// The four levels macOS reports; a level added later reads as the most severe.
         init(_ state: ProcessInfo.ThermalState) {
             switch state {
             case .nominal: self = .nominal
             case .fair: self = .fair
             case .serious: self = .serious
             case .critical: self = .critical
             @unknown default: self = .critical
             }
         }
     }
     ```
  4. Create `Sources/VMRunner/Views/Editor/HostInfoSection.swift`:
     ```swift
     import SwiftUI

     /// The facts about this Mac the guest can read, one switch per category, how to read them, and what they say now.
     struct HostInfoSection: View {
         let guest: GuestOS
         @Binding var sharing: HostInfoSharing

         var body: some View {
             Section {
                 row("Battery level", $sharing.battery)
                 row("Power source and Low Power Mode", $sharing.power)
                 row("Thermal state", $sharing.thermal)
                 row("Network path (no addresses or names)", $sharing.networkPath)
                 row("Time zone", $sharing.timeZone)
                 if !sharing.isEmpty {
                     mountHint
                     preview
                 }
             } header: {
                 HelpHeader("Host information", topic: .hostInformation)
             }
         }

         private func row(_ title: String, _ isOn: Binding<Bool>) -> some View {
             Toggle(title, isOn: isOn).settingHelp(.hostInformation)
         }

         private var mountHint: some View {
             Text(
                 guest == .macOS
                     ? "In the guest: mkdir ~/host && mount_virtiofs \(ConfigurationBuilder.hostInfoTag) ~/host"
                     : "In the guest: sudo mkdir -p /mnt/host && sudo mount -t virtiofs \(ConfigurationBuilder.hostInfoTag) /mnt/host"
             )
             .font(.caption).foregroundStyle(.secondary).textSelection(.enabled)
         }

         /// The files as they would read now. The network path is followed only while the machine runs, so it reads unknown here.
         private var preview: some View {
             DisclosureGroup("What the guest would read now") {
                 let files = HostFacts.current(network: nil).files(for: sharing)
                 ForEach(HostFacts.fileNames.filter { files[$0] != nil }, id: \.self) { name in
                     LabeledContent(name, value: (files[name] ?? "").split(separator: "\n").joined(separator: ", "))
                 }
             }
             .font(.callout)
         }
     }
     ```
