Why: ADR-0012 item 4 passes the `cloudInit` section through, so a cloud-config written for another platform can be dropped
in unchanged. When a spec gives both its own `userData` and steps, the seed's `user-data` is a MIME multi-part document
(cloud-init's documented form for several parts): the user's part first, as given, then the steps' cloud-config with a
`Merge-Type` header that appends its lists (`bootcmd`, `runcmd`, `write_files`) to the user's instead of replacing them.
The user's part must start with `#cloud-config` or `#!` (other cloud-init part kinds are refused, not guessed); a part that
contains the fixed boundary is refused. `networkData` becomes `network-config` as given. `CloudInitSeed` is the only input
the image builder (next cards) takes, and only `SeedCompiler.compile` makes one, from checked scripts; a test pins that its
fields are all text, so no secret type can reach the seed. The merge has not been seen in a guest.

```swift
// The provisioning library has no caller outside its tests until a later change attaches the seed to a machine, so a
// declaration only the tests reach is expected here; every other dead-code rule still applies.
// @dw:disable referenced-only-by-tests

/// The text files of a cloud-init NoCloud seed. Only `SeedCompiler.compile` makes one, from a spec whose script files were
/// checked. Every field is text the guest may read: the seed holds no secret, and no field can hold one.
package struct CloudInitSeed: Equatable, Sendable {
    package let userData: String
    package let metaData: String
    /// The `network-config` file, when the spec gives `networkData`.
    package let networkConfig: String?
}

extension SeedCompiler {
    /// The MIME boundary between the parts of a `user-data` that holds both the spec's own data and the steps.
    static let boundary = "==vmrunner-seed-boundary=="

    /// The seed for `spec`, refusing with the spec's line anything this version cannot carry.
    package static func compile(_ spec: ProvisioningSpec, scripts: VerifiedScripts, instanceID: String) throws -> CloudInitSeed {
        let metaData = try Self.metaData(name: spec.name, instanceID: instanceID)
        var steps: String?
        if !spec.steps.isEmpty { steps = try cloudConfig(for: spec.steps, scripts: scripts) }
        let userData: String
        switch (spec.cloudInit.userData, steps) {
        case (let given?, let steps?):
            userData = try multipart(given, steps: steps)
        case (let given?, nil):
            _ = try partType(of: given)
            userData = given.text
        case (nil, let steps?):
            userData = steps
        case (nil, nil):
            userData = "#cloud-config\n"
        }
        return CloudInitSeed(userData: userData, metaData: metaData, networkConfig: spec.cloudInit.networkData?.text)
    }

    /// The MIME type cloud-init gives the spec's own user data, read from its first line; other kinds are refused.
    private static func partType(of given: SpecText) throws -> String {
        // `#cloud-config-archive` and `#cloud-config-jsonp` are other part kinds that begin with the same words.
        if given.text.hasPrefix("#cloud-config"), given.text.dropFirst("#cloud-config".count).first != "-" {
            return "text/cloud-config"
        }
        if given.text.hasPrefix("#!") { return "text/x-shellscript" }
        throw ProvisioningError(line: given.line, "cloudInit.userData must start with #cloud-config or #!")
    }

    /// The user's part, then the steps' cloud-config, which appends its lists to the user's instead of replacing them.
    private static func multipart(_ given: SpecText, steps: String) throws -> String {
        let type = try partType(of: given)
        guard !given.text.contains(boundary) else {
            throw ProvisioningError(line: given.line, "cloudInit.userData contains the seed's MIME boundary \(boundary)")
        }
        let userPart = given.text.hasSuffix("\n") ? given.text : given.text + "\n"
        var text = "Content-Type: multipart/mixed; boundary=\"\(boundary)\"\nMIME-Version: 1.0\n\n"
        text += "--\(boundary)\nContent-Type: \(type); charset=\"utf-8\"\n\n" + userPart
        text += "--\(boundary)\nContent-Type: text/cloud-config; charset=\"utf-8\"\n"
        text += "Merge-Type: list(append)+dict(no_replace,recurse_list)+str()\n\n" + steps
        text += "--\(boundary)--\n"
        return text
    }
}
```

```swift
import Foundation
import Testing

@testable import Provisioning

private let header = "apiVersion: vmrunner.dev/v1alpha1\nkind: Provisioning\nmetadata:\n  name: node\nspec:\n"
private let firstStep = "  provision:\n  - at: first-boot\n    script: |\n      #!/bin/sh\n      echo first\n"

/// The steps' cloud-config for `firstStep`, line by line (golden).
private let firstStepConfig = [
    "#cloud-config",
    "runcmd:",
    "- ['/bin/sh', '-c', 'mkdir -p /var/lib/cloud/vmrunner && printf %s IyEvYmluL3NoCmVjaG8gZmlyc3QK "
        + "| base64 -d > /var/lib/cloud/vmrunner/step-001 && chmod 700 /var/lib/cloud/vmrunner/step-001 && "
        + "exec env /var/lib/cloud/vmrunner/step-001']",
]

struct CloudInitSeedTests {
    private func seed(_ specBody: String, instanceID: String = "i-1") throws -> CloudInitSeed {
        let spec = try ProvisioningSpec(parsing: header + specBody)
        return try SeedCompiler.compile(
            spec, scripts: try ScriptFiles.verify(spec, in: FileManager.default.temporaryDirectory), instanceID: instanceID)
    }

    private func refusal(_ specBody: String) throws -> ProvisioningError {
        try #require(throws: ProvisioningError.self) { try seed(specBody) }
    }

    @Test
    func theUsersDataAndTheStepsMergeIntoAGoldenMultipartDocument() throws {
        let compiled = try seed(
            "  cloudInit:\n    userData: |\n      #cloud-config\n      package_update: true\n    networkData: |\n      version: 2\n"
                + firstStep)
        let head = [
            "Content-Type: multipart/mixed; boundary=\"==vmrunner-seed-boundary==\"",
            "MIME-Version: 1.0",
            "",
            "--==vmrunner-seed-boundary==",
            "Content-Type: text/cloud-config; charset=\"utf-8\"",
            "",
            "#cloud-config",
            "package_update: true",
            "--==vmrunner-seed-boundary==",
            "Content-Type: text/cloud-config; charset=\"utf-8\"",
            "Merge-Type: list(append)+dict(no_replace,recurse_list)+str()",
            "",
        ]
        let expected = (head + firstStepConfig + ["--==vmrunner-seed-boundary==--"]).joined(separator: "\n") + "\n"
        #expect(compiled.userData == expected)
        #expect(compiled.metaData == "instance-id: i-1\nlocal-hostname: node\n")
        #expect(compiled.networkConfig == "version: 2\n")
    }

    @Test
    func stepsAloneAreTheWholeUserData() throws {
        let compiled = try seed(firstStep)
        #expect(compiled.userData == firstStepConfig.joined(separator: "\n") + "\n")
        #expect(compiled.networkConfig == nil)
    }

    @Test
    func theUsersDataAloneIsPassedThroughAsGiven() throws {
        let script = try seed("  cloudInit:\n    userData: |\n      #!/bin/sh\n      echo hi\n")
        #expect(script.userData == "#!/bin/sh\necho hi\n")
        let config = try seed("  cloudInit:\n    userDataBase64: I2Nsb3VkLWNvbmZpZwpwYWNrYWdlX3VwZGF0ZTogdHJ1ZQo=\n")
        #expect(config.userData == "#cloud-config\npackage_update: true\n")
    }

    @Test
    func userDataOfAnotherKindIsRefusedWithItsLine() throws {
        let alone = try refusal("  cloudInit:\n    userData: |\n      #include https://example.invalid/x\n")
        #expect(alone.line == 7)
        #expect(alone.message == "cloudInit.userData must start with #cloud-config or #!")
        let archive = try refusal("  cloudInit:\n    userData: |\n      #cloud-config-archive\n      - x\n")
        #expect(archive.line == 7)
        let merged = try refusal("  cloudInit:\n    userData: plain words\n" + firstStep)
        #expect(merged.line == 7)
    }

    @Test
    func userDataHoldingTheBoundaryIsRefused() throws {
        let error = try refusal("  cloudInit:\n    userData: |\n      #cloud-config\n      # --==vmrunner-seed-boundary==\n" + firstStep)
        #expect(error.line == 7)
        #expect(error.message.contains("MIME boundary"), "\(error)")
    }

    @Test
    func theSameSpecGivesTheSameSeed() throws {
        let body = "  cloudInit:\n    userData: |\n      #cloud-config\n" + firstStep
        let first = try seed(body)
        let second = try seed(body)
        #expect(first == second)
    }

    /// The seed's type is the boundary of what the guest can read: all of its fields are text the compiler made, so a type
    /// that names a secret cannot be stored in it. Adding a field of another type fails here first.
    @Test
    func everyFieldOfTheSeedIsText() throws {
        let compiled = try seed(firstStep)
        let fields = Mirror(reflecting: compiled).children
        #expect(fields.compactMap(\.label) == ["userData", "metaData", "networkConfig"])
        #expect(fields.allSatisfy { ["String", "Optional<String>"].contains("\(type(of: $0.value))") })
    }
}
```
