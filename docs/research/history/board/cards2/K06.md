Why: the loop an agent talks to. `MCPStdio.serve(input:output:caller:)` reads the input descriptor in chunks into a
`LineBuffer` of at most 1 MiB, answers each line in order, skips empty lines, and ends with 0 at the end of the input. A
longer line is answered -32600 and ends the session with 1 (the stream cannot be resynchronised); a failed read, or an
output that does not take an answer for 10 s (a non-blocking one: a blocking pipe waits for its reader), ends it with 1. A tool call is one `ControlCaller` call: the service's result
becomes the text content; its error becomes a tool error `error <code>: <message>`; a service that cannot be reached is a
tool error that says so. The tests serve the control library's own connection over socket pairs (a new pair per call), with
a token store in a scratch folder and two machines held in memory; the session's input and output are pipes. The output
pipe's writing end is non-blocking, so a regression that outgrew the pipe would fail rather than hang the gate.

```swift
public import ControlProtocol
import Foundation
import GuestConsole

/// The MCP server over a pair of descriptors, standard input and output in `vmr control mcp`: one message per line in, one
/// answer per line out, in order. Each tool call is one call of the control service through `caller`, so a token, a socket
/// and the service's own checks decide what an agent may do.
public enum MCPStdio {
    /// Serves until the input ends (returns 0), or a line is longer than `lineBytes`, the input fails, or a non-blocking
    /// output stops taking answers for ten seconds (returns 1; a blocking output waits for its reader). A line longer than the
    /// limit is answered with an error before the end. A last line without its newline is dropped: a message ends with one.
    public static func serve(input: Int32, output: Int32, caller: ControlCaller, lineBytes: Int = MCPServer.lineBytes) -> Int32 {
        var buffer = LineBuffer(limit: lineBytes)
        var chunk = [UInt8](repeating: 0, count: 4_096)
        while true {
            switch buffer.next() {
            case .line(let bytes):
                guard !bytes.isEmpty, let answer = answer(to: Data(bytes), caller: caller) else { continue }
                guard write(answer, to: output) else { return 1 }
                continue
            case .tooLong:
                let error = RPCError.invalidRequest.with("the message is longer than \(lineBytes) bytes")
                _ = write(RPCResponse(id: .null, error: error).encoded(), to: output)
                return 1
            case .needMore:
                break
            }
            switch ConsoleSocket.read(input, into: &chunk) {
            case .transferred(let count): buffer.append(chunk[..<count])
            case .wouldBlock: _ = ConsoleSocket.wait(input, milliseconds: 1_000)
            case .endOfStream: return 0
            case .failed: return 1
            }
        }
    }

    /// The answer line to one message; nil when it gets none.
    static func answer(to line: Data, caller: ControlCaller) -> Data? {
        switch MCPServer.action(for: line) {
        case .reply(let data):
            return data
        case .ignore:
            return nil
        case .callTool(let id, let tool, let arguments):
            return run(tool, id: id, arguments: arguments, caller: caller)
        }
    }

    /// Calls the tool's control method and answers with its result as text: the service's JSON result, or its error as a tool
    /// error that names the error's code, or why the service could not be reached.
    static func run(_ tool: MCPTool, id: RPCID, arguments: [String: RPCValue], caller: ControlCaller) -> Data {
        do {
            let answer = try caller.call(tool.method, params: arguments)
            if let code = answer.errorCode {
                return MCPServer.toolResult(id: id, text: "error \(code): \(answer.errorMessage ?? "refused")", isError: true)
            }
            let text = answer.result.flatMap { String(bytes: $0, encoding: .utf8) } ?? "null"
            return MCPServer.toolResult(id: id, text: text, isError: false)
        } catch {
            return MCPServer.toolResult(id: id, text: "the control service could not be used: \(error)", isError: true)
        }
    }

    private static func write(_ data: Data, to output: Int32) -> Bool {
        do {
            try ConsoleSocket.writeAll(output, Array(data), stallMilliseconds: 10_000)
            return true
        } catch {
            return false
        }
    }
}
```

```swift
import Foundation
import Testing

@testable import AgentBridge
@testable import ControlProtocol

/// A whole MCP session over pipes, each tool call made through the control library against a served connection: the
/// service's checks, its token store in a scratch folder and two machines held in memory. No app, no machine.
struct MCPStdioTests {
    /// One machine with agent access and one without; starts and stops answer at once.
    private struct TwoMachines: ControlMachines {
        let open = ControlMachine(id: UUID(), name: "open", guest: "linux", state: .stopped, agentAccess: true)
        let closed = ControlMachine(id: UUID(), name: "closed", guest: "linux", state: .stopped, agentAccess: false)

        func machines() async -> [ControlMachine] { [open, closed] }
        func start(_ id: UUID) async -> ControlOutcome { .done(.running) }
        func stop(_ id: UUID, force: Bool) async -> ControlOutcome { .done(force ? .stopped : .shuttingDown) }
    }

    private let machines = TwoMachines()

    /// Serves `input` (a newline ends each line) through a caller whose token holds `capabilities`, and returns the exit
    /// status and the answers. The answers' end is non-blocking: should they ever outgrow the pipe, the session fails instead
    /// of waiting forever for a reader that reads only once it ends.
    private func session(
        _ input: [String], capabilities: Set<ControlCapability> = Set(ControlCapability.allCases), lineBytes: Int = MCPServer.lineBytes
    ) async throws -> (status: Int32, answers: [[String: Any]]) {
        let folder = FileManager.default.temporaryDirectory.appending(path: "mcp-\(UUID().uuidString)", directoryHint: .isDirectory)
        try FileManager.default.createDirectory(at: folder, withIntermediateDirectories: true, attributes: [.posixPermissions: 0o700])
        defer { try? FileManager.default.removeItem(at: folder) }
        let store = try ControlTokenStore(file: folder.appending(path: "tokens.json"))
        let secret = try store.create(name: "agent", capabilities: capabilities).secret
        let connection = ControlConnection(dispatcher: ControlDispatcher(tokens: store, machines: machines, report: { _ in }))
        let caller = ControlCaller(token: secret) {
            let pair = try ControlClient.connectedPair()
            let server = pair.server
            Task { await connection.serve(server) }
            return pair.client
        }
        let toServer = Pipe()
        let fromServer = Pipe()
        let answers = fromServer.fileHandleForWriting.fileDescriptor
        _ = fcntl(answers, F_SETFL, fcntl(answers, F_GETFL) | O_NONBLOCK)
        try toServer.fileHandleForWriting.write(contentsOf: Data(input.map { $0 + "\n" }.joined().utf8))
        try toServer.fileHandleForWriting.close()
        let reading = toServer.fileHandleForReading.fileDescriptor
        let status = await runOnThread { MCPStdio.serve(input: reading, output: answers, caller: caller, lineBytes: lineBytes) }
        try fromServer.fileHandleForWriting.close()
        let data = try fromServer.fileHandleForReading.readToEnd() ?? Data()
        let lines = data.split(separator: UInt8(ascii: "\n"))
        return (status, try lines.map { try #require(try JSONSerialization.jsonObject(with: Data($0)) as? [String: Any]) })
    }

    private func call(_ id: Int, _ tool: String, _ arguments: String = "{}") -> String {
        #"{"jsonrpc":"2.0","id":\#(id),"method":"tools/call","params":{"name":"\#(tool)","arguments":\#(arguments)}}"#
    }

    /// The text and the error flag of a tool result.
    private func toolOutcome(_ answer: [String: Any]) throws -> (text: String, isError: Bool) {
        let result = try #require(answer["result"] as? [String: Any])
        let text = try #require((result["content"] as? [[String: Any]])?.first?["text"] as? String)
        return (text, try #require(result["isError"] as? Bool))
    }

    @Test
    func aSessionInitializesListsItsToolsAndCallsEachControlMethod() async throws {
        let open = machines.open.id.uuidString
        let (status, answers) = try await session([
            #"{"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2025-11-25","capabilities":{}}}"#,
            #"{"jsonrpc":"2.0","method":"notifications/initialized"}"#,
            #"{"jsonrpc":"2.0","id":2,"method":"tools/list"}"#,
            call(3, "machines_list"),
            call(4, "machines_start", #"{"machine":"\#(open)"}"#),
            call(5, "machines_stop", #"{"machine":"\#(open)","force":true}"#),
        ])
        #expect(status == 0)
        #expect(answers.compactMap { $0["id"] as? Int } == [1, 2, 3, 4, 5], "the notification got no answer")
        let list = try toolOutcome(answers[2])
        #expect(!list.isError)
        let machines = try JSONDecoder().decode([String: [ControlMachine]].self, from: Data(list.text.utf8))
        #expect(machines["machines"] == [self.machines.open], "only the machine with agent access")
        #expect(try toolOutcome(answers[3]) == (#"{"id":"\#(open)","state":"running"}"#, false))
        #expect(try toolOutcome(answers[4]) == (#"{"id":"\#(open)","state":"stopped"}"#, false))
    }

    @Test
    func aRefusalOfTheServiceIsAToolErrorWithItsCode() async throws {
        let closed = machines.closed.id.uuidString
        let open = machines.open.id.uuidString
        let (status, answers) = try await session(
            [call(1, "machines_start", #"{"machine":"\#(closed)"}"#), call(2, "machines_stop", #"{"machine":"\#(open)"}"#)],
            capabilities: [.start])
        #expect(status == 0)
        #expect(try toolOutcome(answers[0]) == ("error -32003: Agent access is off for this machine", true))
        #expect(try toolOutcome(answers[1]) == ("error -32001: Not permitted", true))
    }

    @Test
    func aServiceThatCannotBeReachedIsAToolError() async throws {
        let caller = ControlCaller(token: "vmrc_x", socket: "/tmp/vmr-no-such-control-\(UUID().uuidString.prefix(8)).sock")
        let answer = try #require(MCPStdio.answer(to: Data(call(1, "machines_list").utf8), caller: caller))
        let object = try #require(try JSONSerialization.jsonObject(with: answer) as? [String: Any])
        let outcome = try toolOutcome(object)
        #expect(outcome.isError)
        #expect(outcome.text.contains("could not be used: nothing is serving"), "\(outcome.text)")
    }

    @Test
    func aLineLongerThanTheLimitIsRefusedAndEndsTheSession() async throws {
        let (status, answers) = try await session([String(repeating: "x", count: 100), call(2, "machines_list")], lineBytes: 64)
        #expect(status == 1)
        #expect(answers.count == 1, "nothing after the refusal")
        #expect((answers.first?["error"] as? [String: Any])?["code"] as? Int == -32600)
    }

    @Test
    func emptyLinesAreSkippedAndTheEndOfInputEndsTheSessionCleanly() async throws {
        let (status, answers) = try await session(["", #"{"jsonrpc":"2.0","id":1,"method":"ping"}"#, ""])
        #expect(status == 0)
        #expect(answers.count == 1)
    }
}
```

## Test

A whole session (initialize, a notification, tools/list, then list, start and a forced stop) gives five answers with ids
1 to 5; the list's text decodes to the one machine with agent access; start and stop answer `running` and `stopped`. A token
with only `start`: starting the machine whose access is off is `error -32003: ...`, stopping is `error -32001: Not
permitted`, both tool errors. A caller of a socket nobody serves gives a tool error that says the service could not be used
and why. A line over a 64-byte limit is refused and ends the session (status 1, nothing after it). Empty lines are skipped and
the end of the input ends the session with 0.
