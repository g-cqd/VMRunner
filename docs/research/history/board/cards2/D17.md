# D17 — F10, part 4: a machine made before machines kept their own installer gets its clone at launch

Model: haiku · Effort: low · Worktree: yes, `WT = ~/Developer/g-cqd/VMRunner-chains/d-security` (branch `chain/d-security`) · Chain: D, card 17 of 22 (after D16; next D18)

Anchors: locate every edit by its text (`grep -n`), never by a line number. D16 added a function to `ImageStore.swift`; this card
inserts after that one.

Why (read from the code and ADR-0031): before ADR-0031 a Linux machine stored the path of a file directly in the shared image
cache (`Images/<name>`, the "flat" layout); today's cache files are `Images/<sha256>/<name>` and a machine boots from its own
`Installer.iso` clone. ADR-0031 left the old machines as they were, so clearing the downloaded images removes their installer.
At launch, right after approvals are loaded, each such machine whose path the user approved for it gets a clone in its bundle
(`ImageStore.keepInstaller`, the same call setup uses), its configuration names the clone, and the clone is approved too
(the old approval stays). A path that is not approved (a bundle copied from elsewhere) is never migrated, so a copied bundle cannot get
a file of the cache cloned into it. A file that is gone, or a clone that fails, leaves the machine as it was and logs a notice.

## Change

1. `ImageStore.swift`: directly after the closing brace of `static func installCopy(of file: URL) throws -> URL`, insert:

```swift

    /// Whether `path` names a file directly in `imagesDirectory`, where the cache kept downloads before it filed them by
    /// checksum (`Images/<name>`, not `Images/<sha256>/<name>`, ADR-0031).
    static func isInFlatCache(_ path: String, imagesDirectory: URL = AppPaths.images) -> Bool {
        let parent = URL(filePath: path).deletingLastPathComponent().standardizedFileURL.pathComponents
        return parent == imagesDirectory.standardizedFileURL.pathComponents
    }
```

2. `VMLibrary.swift`, in `load()`: directly after the line `        recordApprovals { try $0.grandfather(machines) }` insert
   `        migrateFlatCacheInstallers()`.

3. `VMLibrary.swift`: directly before the line `/// Why the library refused to do something to a machine.` (the doc comment of `enum LibraryError`),

   class's private `approvals`):

```swift
extension VMLibrary {
    /// Gives each machine made before machines kept their own installer (ADR-0031) a clone of it in its bundle. Such a machine
    /// stores the path of a file directly in the old flat image cache, which clearing the downloaded images removes. Only a path
    /// the user approved for that machine is migrated, so a copied bundle cannot get a file of the cache cloned into it, and
    /// the clone is approved too. A machine whose file is gone, or whose clone fails, is left as it is.
    func migrateFlatCacheInstallers(imagesDirectory: URL = AppPaths.images) {
        for config in machines {
            guard let path = config.installerISO, ImageStore.isInFlatCache(path, imagesDirectory: imagesDirectory),
                approvals.missing([HostPathGrant(.installer, path, writable: false)], for: config.id).isEmpty
            else { continue }
            do {
                let kept = try ImageStore.keepInstaller(URL(filePath: path), in: bundle(config.id))
                approveHostPaths([HostPathGrant(.installer, kept.path, writable: false)], for: config.id)
                var migrated = config
                migrated.installerISO = kept.path
                update(migrated)
            } catch {
                Log.storage.notice("Kept a machine's old installer path: \(error.localizedDescription, privacy: .private)")
            }
        }
    }
}

```

    /// A file in the old flat cache, as a machine made before ADR-0031 names it.
    private func flatCacheInstaller(in directory: URL) throws -> (images: URL, file: URL) {
        let images = directory.appending(path: "Images", directoryHint: .isDirectory)
        try FileManager.default.createDirectory(at: images, withIntermediateDirectories: true)
        let file = images.appending(path: "debian-13.7.0-arm64-netinst.iso")
        try Data("installer".utf8).write(to: file)
        return (images, file)
    }

    private func machine(installer path: String, named name: String) -> VMConfig {
        var config = VMConfig.new(name: name, guest: .linux)
        config.isProvisioned = true
        config.installerISO = path
        return config
    }

    @Test func aMachineFromBeforeTheContentCacheGetsItsOwnCloneOfItsApprovedInstaller() throws {
        let directory = try scratchDirectory()
        defer { try? FileManager.default.removeItem(at: directory) }
        let (images, old) = try flatCacheInstaller(in: directory)
        let config = machine(installer: old.path, named: "old")
        try plant(config, in: directory)

        let library = makeLibrary(in: directory)
        library.load()  // approves the paths of machines from before approvals, as an upgrade does
        library.migrateFlatCacheInstallers(imagesDirectory: images)

        let bundle = library.bundle(config.id)
        #expect(library.machine(config.id)?.installerISO == bundle.installer.path)
        #expect(library.unapprovedHostPaths(for: config.id).isEmpty, "the clone is approved")
        try FileManager.default.removeItem(at: old)
        #expect(try Data(contentsOf: bundle.installer) == Data("installer".utf8), "clearing the images no longer removes it")
    }

    @Test func aCopiedBundleNamingTheOldCacheIsNotGivenAClone() throws {
        let directory = try scratchDirectory()
        defer { try? FileManager.default.removeItem(at: directory) }
        let (images, old) = try flatCacheInstaller(in: directory)
        makeLibrary(in: directory).load()  // from now on approvals are recorded, so a later bundle is not grandfathered
        let imported = machine(installer: old.path, named: "imported")
        try plant(imported, in: directory)

        let library = makeLibrary(in: directory)
        library.load()
        library.migrateFlatCacheInstallers(imagesDirectory: images)

        #expect(library.machine(imported.id)?.installerISO == old.path)
        #expect(!FileManager.default.fileExists(atPath: library.bundle(imported.id).installer.path))
    }

    @Test func onlyAFileDirectlyInTheImagesFolderIsInTheFlatCache() {
        let images = URL(filePath: "/data/Images", directoryHint: .isDirectory)
        #expect(ImageStore.isInFlatCache("/data/Images/a.iso", imagesDirectory: images))
        #expect(!ImageStore.isInFlatCache("/data/Images/\(String(repeating: "a", count: 64))/a.iso", imagesDirectory: images))
        #expect(!ImageStore.isInFlatCache("/data/Images2/a.iso", imagesDirectory: images))
        #expect(!ImageStore.isInFlatCache("/data/Machines/x.vm/Installer.iso", imagesDirectory: images))
    }
