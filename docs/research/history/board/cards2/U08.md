## Why

A Windows machine needs a driver disc (a virtio driver ISO several hundred MiB large) that the owner downloads from a source they trust. The owner wants a way to check that file against SHA-256 values they have pinned in the repository. Which discs to trust is the owner's decision, so the list ships EMPTY; the command and its tests are the whole card.

- Command: `vmr windows drivers verify FILE [--root DIR]`.
- The file is opened read-only (`O_RDONLY | O_NONBLOCK | O_CLOEXEC`), checked as a regular file through the descriptor it was opened with, and hashed in 1 MiB chunks with CryptoKit `SHA256` (never loaded whole). A directory, a device, a pipe, a missing file and a symlink loop (`ELOOP`) are refused. A symbolic link to a regular file is followed.
- The list is `Resources/WindowsDriverPins.json` under the package root (DevTools ships no data of its own: it reads `Resources/` from the package root, as `vmr build-app` reads the entitlements; so `Package.swift` is untouched). `--root DIR` names another root, which is how the tests inject a list.
- Exit status: 0 match (prints the entry and its notes); 1 "not in the pinned list" (prints the computed value); 64 usage error (argument parser); 65 the list is missing, unreadable or malformed; 66 the file is missing, unreadable, a directory or not a regular file.
- Schema: `{"schema": 1, "about": "...", "entries": [{"sha256", "name", "version", "notes"}]}`. A malformed list is: not that shape, another schema number, a `sha256` that is not 64 hexadecimal digits, a value listed twice, or an entry with an empty `name`, `version` or `notes` (notes carry the source and the date of the file the owner checked). `about` is the in-file schema comment: it says how an entry is added, by the owner.

```swift
import Foundation

/// One driver disc the owner has checked and pinned.
struct DriverPin: Codable, Equatable, Sendable {
    /// The file's SHA-256, as 64 hexadecimal digits.
    let sha256: String
    let name: String
    let version: String
    /// Where the owner got the file and the date they checked it.
    let notes: String
}

/// The SHA-256 values pinned in `Resources/WindowsDriverPins.json`.
///
/// The file is `{"schema": 1, "about": "...", "entries": [{"sha256", "name", "version", "notes"}]}`. The list ships empty:
/// which discs to trust is the owner's decision, and an entry is added by the owner, with the source and the date of
/// the file they checked in `notes`.
struct DriverPins: Equatable, Sendable {
    static let schema = 1
    /// Where the list lives, relative to the package root.
    static let relativePath = "Resources/WindowsDriverPins.json"

    let entries: [DriverPin]

    private struct Document: Decodable {
        let schema: Int
        let entries: [DriverPin]
    }

    /// Reads and checks the list at `url`.
    /// - Throws: a `ToolError` with status `DriverVerifyStatus.listUnusable` if the file cannot be read or is not a valid list.
    static func load(from url: URL) throws -> Self {
        let data: Data
        do {
            data = try Data(contentsOf: url)
        } catch {
            throw ToolError("cannot read the pinned list \(url.path)", status: DriverVerifyStatus.listUnusable)
        }
        return try Self(parsing: data, source: url.path)
    }

    /// Checks `data` as a list: the schema number, a well-formed value in every entry, and no value twice.
    init(parsing data: Data, source: String) throws {
        func refuse(_ reason: String) -> ToolError {
            ToolError("the pinned list \(source) is malformed: \(reason)", status: DriverVerifyStatus.listUnusable)
        }
        let document: Document
        do {
            document = try JSONDecoder().decode(Document.self, from: data)
        } catch {
            throw refuse("it is not JSON of the documented shape")
        }
        guard document.schema == Self.schema else { throw refuse("schema \(document.schema), expected \(Self.schema)") }
        var seen: Set<String> = []
        for entry in document.entries {
            guard Self.isDigest(entry.sha256) else { throw refuse("\"\(entry.sha256)\" is not 64 hexadecimal digits") }
            guard seen.insert(entry.sha256.lowercased()).inserted else { throw refuse("\(entry.sha256) is listed twice") }
            guard !entry.name.isEmpty, !entry.version.isEmpty, !entry.notes.isEmpty else {
                throw refuse("the entry \(entry.sha256) needs a name, a version and notes (the source and the date checked)")
            }
        }
        entries = document.entries
    }

    /// The entry for `digest` (hexadecimal, any case), or nil.
    func entry(forDigest digest: String) -> DriverPin? {
        entries.first { $0.sha256.lowercased() == digest.lowercased() }
    }

    private static func isDigest(_ text: String) -> Bool {
        text.utf8.count == 64
            && text.utf8.allSatisfy { ($0 >= 0x30 && $0 <= 0x39) || ($0 >= 0x61 && $0 <= 0x66) || ($0 >= 0x41 && $0 <= 0x46) }
    }
}

/// The exit statuses of `vmr windows drivers verify`: 0 matches, and one distinct status for each way it does not.
enum DriverVerifyStatus {
    /// The file is not in the pinned list.
    static let noMatch: Int32 = 1
    /// `sysexits.h` EX_DATAERR: the pinned list cannot be read or is malformed.
    static let listUnusable: Int32 = 65
    /// `sysexits.h` EX_NOINPUT: the file cannot be read, is a directory, or is not a regular file.
    static let fileUnreadable: Int32 = 66
}
```

3. Create `Sources/DevTools/Windows/FileDigest.swift` with exactly:

```swift
import CryptoKit
import Foundation

/// The SHA-256 of a file the user named, read once, in chunks, and never modified.
enum FileDigest {
    /// How much of the file is in memory at a time: disc images are several hundred MiB.
    static let chunkBytes = 1 << 20

    /// The digest of the regular file at `url` as 64 lowercase hexadecimal digits.
    ///
    /// The file is opened read-only and checked through the descriptor it was opened with. A link is followed (a loop
    /// fails with `ELOOP`); a directory, a device or a pipe is refused.
    /// - Throws: a `ToolError` with status `DriverVerifyStatus.fileUnreadable`.
    static func sha256(ofFileAt url: URL) throws -> String {
        // `open` gets a NUL-terminated copy of the path that outlives the call; O_NONBLOCK keeps a pipe from blocking it.
        let descriptor = unsafe open(url.path, O_RDONLY | O_NONBLOCK | O_CLOEXEC)
        guard descriptor >= 0 else {
            // `strerror` returns a static, NUL-terminated message that is copied at once.
            let reason = unsafe String(cString: strerror(errno))
            throw ToolError("cannot read \(url.path): \(reason)", status: DriverVerifyStatus.fileUnreadable)
        }
        let handle = FileHandle(fileDescriptor: descriptor, closeOnDealloc: true)
        defer { try? handle.close() }
        var info = stat()
        // `info` is a local that outlives the call, which fills it.
        guard unsafe fstat(descriptor, &info) == 0, info.st_mode & S_IFMT == S_IFREG else {
            throw ToolError("\(url.path) is not a regular file", status: DriverVerifyStatus.fileUnreadable)
        }
        var hasher = SHA256()
        do {
            while true {
                let finished = try autoreleasepool {
                    guard let chunk = try handle.read(upToCount: chunkBytes), !chunk.isEmpty else { return true }
                    hasher.update(data: chunk)
                    return false
                }
                if finished { break }
            }
        } catch {
            throw ToolError("cannot read \(url.path): \(error.localizedDescription)", status: DriverVerifyStatus.fileUnreadable)
        }
        return hasher.finalize().map { ($0 < 16 ? "0" : "") + String($0, radix: 16) }.joined()
    }
}
```

import ArgumentParser
import Foundation

struct WindowsCommand: AsyncParsableCommand {
    static let configuration = CommandConfiguration(
        commandName: "windows",
        abstract: "Helpers for Windows machines.",
        subcommands: [Drivers.self]
    )

    struct Drivers: AsyncParsableCommand {
        static let configuration = CommandConfiguration(
            commandName: "drivers",
            abstract: "Driver discs for a Windows machine.",
            subcommands: [Verify.self]
        )

        struct Verify: AsyncParsableCommand {
            static let configuration = CommandConfiguration(
                commandName: "verify",
                abstract: "Checks a driver disc you downloaded against the SHA-256 values pinned in this repository.",
                discussion: """
                    Hashes FILE (read-only, in chunks) and looks its SHA-256 up in \(DriverPins.relativePath) under the package root. \
                    It prints the matching entry, or "not in the pinned list" with the value it computed. It downloads nothing, \
                    uses no network and changes no file.

                    The list ships empty: which discs to trust is the owner's decision. The owner adds an entry (sha256, name, \
                    version, notes with the source and the date of the file they checked) to that file; a SHA-256 comes from \
                    `shasum -a 256 FILE`.

                    Exit status: 0 the file matches an entry; 1 it is not in the pinned list; 64 a usage error; 65 the pinned list \
                    cannot be read or is malformed; 66 FILE cannot be read, is a directory, or is not a regular file.
                    """
            )

            @OptionGroup var project: ProjectOptions

            @Argument(help: ArgumentHelp("The driver disc image to check.", valueName: "file"))
            var file: String

            func run() async throws {
                try await perform {
                    let cwd = URL(fileURLWithPath: FileManager.default.currentDirectoryPath, isDirectory: true)
                    let pins = try DriverPins.load(from: try project.packageRoot().appending(path: DriverPins.relativePath))
                    let digest = try FileDigest.sha256(ofFileAt: URL(fileURLWithPath: file, relativeTo: cwd).standardizedFileURL)
                    guard let entry = pins.entry(forDigest: digest) else {
                        Output.out("not in the pinned list")
                        Output.out("sha256: \(digest)")
                        return DriverVerifyStatus.noMatch
                    }
                    Output.out("matches the pinned entry: \(entry.name) \(entry.version)")
                    Output.out("sha256: \(digest)")
                    Output.out("notes: \(entry.notes)")
                    return 0
                }
            }
        }
    }
}

@testable import DevTools

/// `vmr windows drivers verify`: the digest of a scratch file against a list the test writes, never the user's files.
struct WindowsDriversTests {
    /// SHA-256 of the three bytes "abc" (FIPS 180-4 example).
    private static let abc = "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad"

    private static let repositoryRoot = URL(fileURLWithPath: #filePath).deletingLastPathComponent().deletingLastPathComponent()
        .deletingLastPathComponent()

    private static func list(entries: String) -> String {
        #"{"schema": 1, "about": "test", "entries": [\#(entries)]}"#
    }

    private static let abcEntry =
        #"{"sha256": "\#(abc.uppercased())", "name": "Test disc", "version": "1.2", "notes": "scratch, 2026-10-25"}"#

    /// A package root with the given list and a file `disc.iso` holding "abc".
    private func withFixture(list: String?, _ body: (URL) async throws -> Void) async throws {
        try await withTemporaryDirectory { directory in
            var files = ["disc.iso": "abc"]
            files[DriverPins.relativePath] = list
            try write(files, to: directory)
            try await body(directory)
        }
    }

    private func verify(_ file: String, in directory: URL) async throws -> Invocation {
        try await VMR.vmr(["windows", "drivers", "verify", file, "--root", directory.path], in: directory)
    }

    @Test
    func hashesAKnownFileInChunks() throws {
        let url = FileManager.default.temporaryDirectory.appending(path: "vmr-digest-\(UUID().uuidString)")
        defer { try? FileManager.default.removeItem(at: url) }
        try Data("abc".utf8).write(to: url)
        #expect(try FileDigest.sha256(ofFileAt: url) == Self.abc)
        // Larger than three chunks, and not a multiple of the chunk size.
        let big = Data((0..<(FileDigest.chunkBytes * 3 + 17)).map { UInt8(truncatingIfNeeded: $0 &* 31) })
        try big.write(to: url)
        let expected = SHA256.hash(data: big).map { ($0 < 16 ? "0" : "") + String($0, radix: 16) }.joined()
        #expect(try FileDigest.sha256(ofFileAt: url) == expected)
    }

    @Test
    func aMatchPrintsTheEntryAndExitsZero() async throws {
        try await withFixture(list: Self.list(entries: Self.abcEntry)) { directory in
            let run = try await verify("disc.iso", in: directory)
            #expect(run.status == 0, "\(run.combined)")
            #expect(run.output.contains("matches the pinned entry: Test disc 1.2"), "\(run.output)")
            #expect(run.output.contains("notes: scratch, 2026-10-25"), "\(run.output)")
        }
    }

    @Test
    func aFileNotInTheListSaysSoAndExitsOne() async throws {
        let other = #"{"sha256": "\#(String(repeating: "0", count: 64))", "name": "Other", "version": "9", "notes": "n"}"#
        try await withFixture(list: Self.list(entries: other)) { directory in
            let run = try await verify("disc.iso", in: directory)
            #expect(run.status == DriverVerifyStatus.noMatch, "\(run.combined)")
            #expect(run.output.contains("not in the pinned list"), "\(run.output)")
            #expect(run.output.contains(Self.abc), "\(run.output)")
        }
    }

    @Test
    func theShippedListParsesAndMatchesNothing() throws {
        let pins = try DriverPins.load(from: Self.repositoryRoot.appending(path: DriverPins.relativePath))
        #expect(pins.entries.isEmpty, "the owner adds the pins; the repository ships none")
        #expect(pins.entry(forDigest: Self.abc) == nil)
    }

    @Test
    func aMalformedListExitsSixtyFive() async throws {
        let bad = [
            "not json", Self.list(entries: #"{"sha256": "abc", "name": "n", "version": "v", "notes": "n"}"#),
            Self.list(entries: Self.abcEntry + "," + Self.abcEntry),
            Self.list(entries: #"{"sha256": "\#(Self.abc)", "name": "n", "version": "v", "notes": ""}"#),
            #"{"schema": 2, "entries": []}"#, #"{"schema": 1}"#,
        ]
        for text in bad {
            try await withFixture(list: text) { directory in
                let run = try await verify("disc.iso", in: directory)
                #expect(run.status == DriverVerifyStatus.listUnusable, "\(text): \(run.combined)")
                #expect(run.error.contains("malformed"), "\(run.error)")
            }
        }
        try await withFixture(list: nil) { directory in
            let run = try await verify("disc.iso", in: directory)
            #expect(run.status == DriverVerifyStatus.listUnusable, "a missing list: \(run.combined)")
        }
    }

    @Test
    func aMissingFileExitsSixtySix() async throws {
        try await withFixture(list: Self.list(entries: Self.abcEntry)) { directory in
            let run = try await verify("missing.iso", in: directory)
            #expect(run.status == DriverVerifyStatus.fileUnreadable, "\(run.combined)")
            #expect(run.error.contains("cannot read"), "\(run.error)")
        }
    }

    @Test
    func aDirectoryAndASymlinkLoopAreRefused() async throws {
        try await withFixture(list: Self.list(entries: Self.abcEntry)) { directory in
            try FileManager.default.createDirectory(at: directory.appending(path: "folder"), withIntermediateDirectories: false)
            try FileManager.default.createSymbolicLink(atPath: directory.appending(path: "loop").path, withDestinationPath: "loop")

            #expect(folder.error.contains("not a regular file"), "\(folder.error)")

            #expect(loop.status == DriverVerifyStatus.fileUnreadable, "\(loop.combined)")
        }
    }

            let after = try FileManager.default.attributesOfItem(atPath: directory.appending(path: "disc.iso").path)
            #expect(before[.modificationDate] as? Date == after[.modificationDate] as? Date)
        }
