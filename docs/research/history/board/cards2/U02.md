## Why

ADR-0059 (item 2) and ADR-0019: one target, `PrivateInterfaces`, holds a registry of capabilities. Each has an id, a range of macOS builds (`kern.osversion`), a resolver (the Objective-C runtime, asked by `NSClassFromString` and `NSSelectorFromString`; `Selector("…")` is rejected by the compiler settings), a validator (the method's type encoding with its digits removed must equal the declared one: `v@:B` for the setter, `B@:` for the getter), a fallback (unavailable, with a reason) and a kill switch. The first capability is performance-monitoring emulation on `VZGenericPlatformConfiguration`: setter `_setPerformanceMonitoringUnitEmulationEnabled:` (framework encoding `v20@0:8B16`), getter `_performanceMonitoringUnitEmulationEnabled` (`B16@0:8`). The setter is called through the method's implementation (`unsafeBitCast` to a `@convention(c)` function type); the getter is read back; a value that was not kept puts the old value back and makes the capability unavailable. A test in `DevToolsTests` keeps the runtime's lookup functions out of every other target. The app target links the module (nothing in the app calls it yet); no other target does except the module's own tests.

The registered builds are `26A0` to `26Z99999` (Darwin 26 is macOS 27). The probe below was run on macOS 27.0.1 (build 26A434) only; a macOS 26 host reports the capability unavailable with the build named, until someone probes it and widens the range.

```swift
package import Foundation

extension CapabilityEntry {
    private typealias Setter = @convention(c) (AnyObject, Selector, Bool) -> Void
    private typealias Getter = @convention(c) (AnyObject, Selector) -> Bool

    /// Sets the switch on `object` and reads it back. The object must be an instance of the setter's class, and the contract
    /// must hold now (`status(in:)`); otherwise nothing is called. If the value read back is not the value set, the value the
    /// switch had is put back and the capability is reported unavailable.
    package func apply(_ value: Bool, to object: NSObject, in context: PrivateCapabilityContext) -> PrivateCapabilityStatus {
        let checked = status(in: context)
        guard checked.isAvailable else { return checked }
        guard let receiver = NSClassFromString(setter.className), object.isKind(of: receiver) else {
            return .unavailable(.receiverMismatch(class: setter.className))
        }
        let setSelector = NSSelectorFromString(setter.selector)
        let getSelector = NSSelectorFromString(getter.selector)
        guard let setMethod = unsafe class_getInstanceMethod(type(of: object), setSelector),
            let getMethod = unsafe class_getInstanceMethod(type(of: object), getSelector)
        else { return .unavailable(.selectorMissing(class: setter.className, selector: setter.selector)) }
        let set = unsafe unsafeBitCast(method_getImplementation(setMethod), to: Setter.self)
        let get = unsafe unsafeBitCast(method_getImplementation(getMethod), to: Getter.self)

        let before = get(object, getSelector)
        set(object, setSelector, value)
        guard get(object, getSelector) == value else {
            set(object, setSelector, before)
            return .unavailable(.readBackMismatch(selector: setter.selector))
        }
        return .available
    }
}
```

```swift
import Foundation

/// One method the capability calls: the class, the selector, and the type encoding with its digits removed (`v@:B` for a
/// setter of a boolean), because the digits are offsets that change with the architecture and the SDK.
package struct MethodContract: Equatable, Sendable {
    package let className: String
    package let selector: String
    package let encoding: String

    package init(className: String, selector: String, encoding: String) {
        self.className = className
        self.selector = selector
        self.encoding = encoding
    }
}

/// What the runtime says about one instance method.
package enum MethodLookup: Equatable, Sendable {
    case classMissing
    case methodMissing
    case found(encoding: String)
}

/// Finds an instance method by class and selector name. The real one asks the Objective-C runtime; tests use a fake.
package protocol MethodResolver: Sendable {
    func lookup(className: String, selector: String) -> MethodLookup
}

/// The Objective-C runtime, asked by name. Nothing is called and no instance is made.
package struct RuntimeMethodResolver: MethodResolver {
    package init() {}

    package func lookup(className: String, selector: String) -> MethodLookup {
        guard let candidate = NSClassFromString(className) else { return .classMissing }
        guard let method = unsafe class_getInstanceMethod(candidate, NSSelectorFromString(selector)),
            let raw = unsafe method_getTypeEncoding(method)
        else { return .methodMissing }
        return .found(encoding: unsafe String(cString: raw))
    }
}

/// Everything a capability's check depends on, so a test can state it.
package struct PrivateCapabilityContext: Sendable {
    /// `kern.osversion`, or `nil` when it could not be read.
    package var osBuild: String?
    /// The capabilities turned off by the kill switch, by the user's settings or by an organisation.
    package var killed: Set<PrivateCapabilityID>
    package var resolver: any MethodResolver

    package init(osBuild: String?, killed: Set<PrivateCapabilityID>, resolver: any MethodResolver) {
        self.osBuild = osBuild
        self.killed = killed
        self.resolver = resolver
    }

    /// This Mac, with the given capabilities turned off.
    package static func current(killed: Set<PrivateCapabilityID> = []) -> Self {
        Self(osBuild: OSBuild.running, killed: killed, resolver: RuntimeMethodResolver())
    }
}

/// A capability that is a boolean switch on an object of the framework: a setter, a getter that reads the value back, and the
/// builds of macOS the pair was verified on.
package struct CapabilityEntry: Sendable {
    package let id: PrivateCapabilityID
    package let builds: ClosedRange<OSBuild>
    package let setter: MethodContract
    package let getter: MethodContract

    package init(id: PrivateCapabilityID, builds: ClosedRange<OSBuild>, setter: MethodContract, getter: MethodContract) {
        self.id = id
        self.builds = builds
        self.setter = setter
        self.getter = getter
    }

    /// Whether the capability can be used, without touching any object: the kill switch, the build of macOS, then each method
    /// by class, selector and type encoding. The first failure is the answer; nothing is called.
    package func status(in context: PrivateCapabilityContext) -> PrivateCapabilityStatus {
        if context.killed.contains(id) { return .unavailable(.killSwitch) }
        guard let text = context.osBuild, let build = OSBuild(text) else { return .unavailable(.osBuildUnreadable) }
        guard builds.contains(build) else {
            return .unavailable(.osBuildOutsideRange(build: text, verified: "\(builds.lowerBound) to \(builds.upperBound)"))
        }
        for method in [setter, getter] {
            switch context.resolver.lookup(className: method.className, selector: method.selector) {
            case .classMissing:
                return .unavailable(.classMissing(method.className))
            case .methodMissing:
                return .unavailable(.selectorMissing(class: method.className, selector: method.selector))
            case .found(let encoding):
                let stripped = String(encoding.filter { !$0.isNumber })
                if stripped != method.encoding {
                    return .unavailable(.encodingMismatch(selector: method.selector, expected: method.encoding, found: stripped))
                }
            }
        }
        return .available
    }
}

/// The capabilities the app knows, by id. A test builds its own to state a case the real one cannot be put in.
package struct CapabilityRegistry: Sendable {
    package var entries: [PrivateCapabilityID: CapabilityEntry]

    package init(entries: [CapabilityEntry]) {
        self.entries = Dictionary(entries.map { ($0.id, $0) }, uniquingKeysWith: { first, _ in first })
    }

    /// The capabilities of this app: the performance-monitoring switch of the generic platform configuration.
    package static let standard = Self(entries: [PMUEmulation.entry])
}
```

```swift
import Foundation

/// A macOS build as `kern.osversion` reports it, such as `26A434`: the Darwin major version, a capital letter, the build
/// count, and on a beta a trailing small letter that does not take part in the order.
package struct OSBuild: Comparable, Sendable, CustomStringConvertible {
    let major: Int
    let letter: Character
    let count: Int

    package init(major: Int, letter: Character, count: Int) {
        self.major = major
        self.letter = letter
        self.count = count
    }

    /// `nil` unless `text` is digits, one capital letter, digits and at most one small letter, in that order.
    package init?(_ text: String) {
        let characters = Array(text)
        guard let letterIndex = characters.firstIndex(where: { !$0.isASCIIDigit }), letterIndex > 0 else { return nil }
        let letter = characters[letterIndex]
        guard letter.isASCII, letter.isUppercase else { return nil }
        var end = letterIndex + 1
        while end < characters.count, characters[end].isASCIIDigit { end += 1 }
        guard end > letterIndex + 1 else { return nil }
        let suffix = characters[end...]
        guard suffix.isEmpty || (suffix.count == 1 && suffix[end].isASCII && suffix[end].isLowercase) else { return nil }
        guard let major = Int(String(characters[..<letterIndex])), let count = Int(String(characters[(letterIndex + 1)..<end]))
        else { return nil }
        self.init(major: major, letter: letter, count: count)
    }

    package var description: String { "\(major)\(letter)\(count)" }

    package static func < (lhs: Self, rhs: Self) -> Bool {
        (lhs.major, lhs.letter, lhs.count) < (rhs.major, rhs.letter, rhs.count)
    }

    /// The build this Mac is running, or `nil` when the system does not say.
    package static var running: String? {
        var size = 0
        guard unsafe sysctlbyname("kern.osversion", nil, &size, nil, 0) == 0, size > 1 else { return nil }
        var buffer = [UInt8](repeating: 0, count: size)
        guard unsafe sysctlbyname("kern.osversion", &buffer, &size, nil, 0) == 0 else { return nil }
        return String(bytes: buffer.prefix { $0 != 0 }, encoding: .utf8)
    }
}

extension Character {
    fileprivate var isASCIIDigit: Bool { isASCII && isNumber }
}
```

```swift
package import Virtualization

/// Performance-monitoring emulation for a guest: the generic platform configuration's switch, which Windows on ARM needs to
/// boot and the framework does not document. Every use goes through the capability's checks (`CapabilityEntry.status`) and
/// fails closed: when a check fails nothing is called and the status says why.
package enum PMUEmulation {
    private static let className = "VZGenericPlatformConfiguration"

    /// Verified on macOS 27.0.1 (build 26A434): the setter's encoding is `v20@0:8B16`, the getter's `B16@0:8`. The builds are
    /// those of Darwin 26, which is macOS 27; a build outside them is off until it is probed and this range is widened.
    package static let entry = CapabilityEntry(
        id: .performanceMonitoringEmulation,
        builds: OSBuild(major: 26, letter: "A", count: 0)...OSBuild(major: 26, letter: "Z", count: 99_999),
        setter: MethodContract(
            className: className, selector: "_setPerformanceMonitoringUnitEmulationEnabled:", encoding: "v@:B"),
        getter: MethodContract(className: className, selector: "_performanceMonitoringUnitEmulationEnabled", encoding: "B@:"))

    /// Whether the switch can be used on this Mac with these settings. Nothing is called and no configuration is touched.
    package static func status(
        in context: PrivateCapabilityContext = .current(), registry: CapabilityRegistry = .standard
    ) -> PrivateCapabilityStatus {
        guard let registered = registry.entries[entry.id] else { return .unavailable(.notRegistered) }
        return registered.status(in: context)
    }

    /// Turns the emulation on for `configuration` and reads it back. Any failed check, or a value that was not kept, leaves the
    /// configuration as it was and returns the reason.
    package static func apply(
        to configuration: VZGenericPlatformConfiguration, in context: PrivateCapabilityContext = .current(),
        registry: CapabilityRegistry = .standard
    ) -> PrivateCapabilityStatus {
        guard let registered = registry.entries[entry.id] else { return .unavailable(.notRegistered) }
        return registered.apply(true, to: configuration, in: context)
    }
}
```
