## Why

A custom device is not one of the framework's graphics devices, so `VZVirtualMachineView` cannot show it (the research
note): the host presents the scanout itself. A real-guest run must not capture the screen or drive a window
(CONTRIBUTING), so the primary output is a file: `--headless-dump` writes the newest picture to `scanout.png` in the
scratch directory, at most once a second and once at the end, through a private temporary file (`scanout.png.partial`,
0600) renamed over the last one, so a reader never sees half a file. `--window` shows the same pictures in the runner's own
window (a layer-backed view whose layer's contents is the picture, at most 30 times a second); the owner may watch it, an
agent never opens it. Both are `ScanoutSink`s fed by `GPUDevice` on each flush (Q05's `ScanoutUpdate`), ticked every
50 ms so a picture kept back by `FrameThrottle` (Core, tested) is still shown, and finished when the run ends.

The library converts each of the eight virtio formats (memory byte order: `b8g8r8x8` is blue, green, red, padding, the
format a Linux guest's console uses) to red, green, blue and an opaque padding byte (`ScanoutImage`, table-tested). The
flags join the parser (`--headless-dump`, `--window`, combinable); Q02's test row that used `--window` as its example of an
unknown option now uses `--bogus`.

```swift
import Foundation
import VirtioGPURunnerCore
import Virtualization

/// The host side of the virtio GPU device. The framework calls it on the device queue, which is the main queue, so its
/// state is main-actor state. Each request is copied out of guest memory once, decoded and carried out by
/// `GPUDeviceState`, answered when the guest left room for the answer, and given back exactly once. Every request goes to
/// the bounded command record; only milestones are printed.
@available(macOS 27.0, *)
@MainActor
final class GPUDevice: NSObject, VZCustomVirtioDeviceConfigurationDelegate, VZCustomVirtioDeviceDelegate {
    private let started = ContinuousClock.now
    private let probeConfigChangeSeconds: Int?
    private var device: VZCustomVirtioDevice?
    private var state = GPUDeviceState()
    private var recorder = CommandRecorder()
    private var notifications = 0
    private var oversized = 0
    private var unanswered = 0
    private var flushes = 0
    private var failures = 0
    private var probeArmed = false
    /// Where each new picture of the scanout goes (the window, the PNG file); none in a run without either.
    var sinks: [any ScanoutSink] = []

    init(probeConfigChangeSeconds: Int?) {
        self.probeConfigChangeSeconds = probeConfigChangeSeconds
    }

    /// Milliseconds since the device object was made: the clock of the log, the record and the pictures.
    var milliseconds: UInt64 {
        let elapsed = ContinuousClock.now - started
        return UInt64(elapsed.components.seconds) * 1000 + UInt64(elapsed.components.attoseconds / 1_000_000_000_000_000)
    }

    /// One line, prefixed with the seconds since the device object was made.
    func log(_ text: String) {
        let now = milliseconds
        Output.line("[\(now / 1000).\(String(now % 1000 + 1000).dropFirst())] \(text)")
    }

    /// The text of `commands.log`.
    func report() -> String {
        recorder.report(extra: [
            "queue notifications: \(notifications)", "oversized requests refused: \(oversized)",
            "answers that did not fit the guest's buffer: \(unanswered)", "scanout flushes: \(flushes)",
            "failed reads and writes of a request or an answer: \(failures)",
        ])
    }

    func customVirtioConfiguration(
        _ deviceConfiguration: VZCustomVirtioDeviceConfiguration, didCreateDevice device: VZCustomVirtioDevice
    ) {
        device.delegate = self
        self.device = device
        log("device created")
    }

    func customVirtioDeviceDidAcceptDriverOk(_ device: VZCustomVirtioDevice) {
        let features = device.negotiatedFeatures
        log("DRIVER_OK, negotiated features subset0 \(features?.subset0 ?? 0) subset1 \(features?.subset1 ?? 0)")
        if let seconds = probeConfigChangeSeconds, !probeArmed {
            probeArmed = true
            Task { @MainActor in
                try? await Task.sleep(for: .seconds(seconds))
                await self.signalDisplayChange()
            }
        }
    }

    /// Sets the display event in the configuration space, as a host does when the display changed. A guest that is told of
    /// the change asks for the display information again; the record shows whether and when it did.
    private func signalDisplayChange() async {
        guard let device else { return }
        let space = GPUConfigSpace(eventsRead: GPUWire.eventDisplay, eventsClear: 0, scanoutCount: 1, capsetCount: 0)
        recorder.noteConfigurationUpdate(atMilliseconds: milliseconds)
        log("configuration update: display event set")
        do {
            try await device.update(VZVirtioDeviceSpecificConfiguration(configurationData: Data(space.bytes)))
            log("configuration update completed")
        } catch {
            log("configuration update failed: \(error.localizedDescription)")
        }
    }

    func customVirtioDevice(_ device: VZCustomVirtioDevice, didReceiveNotificationFor queue: VZVirtioQueue) {
        notifications += 1
        let kind: GPUQueueKind = queue.queueIndex == GPUWire.cursorQueue ? .cursor : .control
        let memory = MappedGuestMemory(device: device)
        while let element = queue.nextElement() {
            service(element, on: kind, memory: memory)
        }
    }

    /// Reads the request once, handles the copy, writes the answer when it fits, and gives the element back exactly once.
    private func service(_ element: VZVirtioQueueElement, on queue: GPUQueueKind, memory: MappedGuestMemory) {
        defer { element.returnToQueue() }
        let length = element.readBuffersAvailableByteCount
        guard length <= state.store.limits.maxCommandBytes else {
            refuse(element, length: length, on: queue)
            return
        }
        let bytes: [UInt8]
        do {
            bytes = length == 0 ? [] : [UInt8](try element.readBytes(withExactLength: length))
        } catch {
            noteFailure("could not read a request of \(length) bytes: \(error.localizedDescription)")
            return
        }
        let handled = state.handle(bytes, queue: queue, memory: memory)
        for line in recorder.record(handled.logEntry, atMilliseconds: milliseconds) {
            log(line)
        }
        if let update = handled.scanoutUpdate {
            flushes += 1
            if flushes == 1 { log("first scanout flush: \(update.width) x \(update.height), format \(update.format)") }
            for sink in sinks { sink.show(update, atMilliseconds: milliseconds) }
        }
        if let response = handled.response { answer(element, with: response) }
    }

    /// A request longer than the device reads: only its header is read, and it is answered as an invalid parameter.
    private func refuse(_ element: VZVirtioQueueElement, length: Int, on queue: GPUQueueKind) {
        oversized += 1
        if oversized == 1 { log("refused a request of \(length) bytes, more than the device reads") }
        let header = (try? element.readBytes(withExactLength: GPUWireSize.header)).map { [UInt8]($0) } ?? []
        guard queue == .control else { return }
        let request = GPUDecoder.decode(header, limits: state.store.limits)
        answer(element, with: GPUResponse.error(.errInvalidParameter).encoded(answering: request.header))
    }

    private func answer(_ element: VZVirtioQueueElement, with response: [UInt8]) {
        guard element.writeBuffersAvailableByteCount >= response.count else {
            unanswered += 1
            if unanswered == 1 { log("an answer of \(response.count) bytes did not fit the guest's buffer; nothing was written") }
            return
        }
        do {
            try element.write(Data(response))
        } catch {
            noteFailure("could not write an answer: \(error.localizedDescription)")
        }
    }

    /// Counts a failed read or write and prints only the first, so a guest cannot make the log grow.
    private func noteFailure(_ text: String) {
        failures += 1
        if failures == 1 { log(text) }
    }

    func customVirtioDeviceWillStop(_ device: VZCustomVirtioDevice) { log("device stopping") }
    func customVirtioDeviceWillPause(_ device: VZCustomVirtioDevice) { log("device pausing") }
    func customVirtioDeviceWillResume(_ device: VZCustomVirtioDevice) { log("device resuming") }

    func customVirtioDeviceWillReset(_ device: VZCustomVirtioDevice) {
        state.reset()
        log("device reset: resources and scanout forgotten")
    }
}
```

```swift
import Foundation
import VirtioGPURunnerCore
import Virtualization

/// One run: the scratch directory, the configuration, the machine, and the hard time limit after which the machine is
/// stopped whatever it does. Everything runs on the main queue (the framework's, the device's and the runner's).
@available(macOS 27.0, *)
@MainActor
final class RunnerMain: NSObject, VZVirtualMachineDelegate {
    private let options: RunnerOptions
    /// Held here because the device provider keeps its delegate weakly.
    private let device: GPUDevice
    private var machine: VZVirtualMachine?

    init(options: RunnerOptions) {
        self.options = options
        let device = GPUDevice(probeConfigChangeSeconds: options.probeConfigChangeSeconds)
        var sinks: [any ScanoutSink] = []
        if options.headlessDump { sinks.append(ScanoutDump(path: options.scanoutPNGPath) { device.log($0) }) }
        if options.window { sinks.append(ScanoutWindow()) }
        device.sinks = sinks
        self.device = device
    }

    /// Builds and validates the configuration without writing a file or starting anything.
    func validate() throws {
        let configuration = try MachineBuilder.build(options, device: device, creatingFiles: false)
        try configuration.validate()
    }

    /// Creates the scratch directory, builds, validates and starts the machine, and arms the time limit.
    func start() throws {
        try FileManager.default.createDirectory(
            atPath: options.directory, withIntermediateDirectories: true, attributes: [.posixPermissions: 0o700])
        let pid = Data("\(ProcessInfo.processInfo.processIdentifier)\n".utf8)
        guard FileManager.default.createFile(atPath: options.pidPath, contents: pid, attributes: [.posixPermissions: 0o600]) else {
            throw RunnerFailure("cannot write \(options.pidPath)")
        }
        let configuration = try MachineBuilder.build(options, device: device, creatingFiles: true)
        try configuration.validate()
        let machine = VZVirtualMachine(configuration: configuration)
        machine.delegate = self
        self.machine = machine
        device.log("starting: \(options.cpuCount) CPUs, \(options.memoryMiB) MiB, limit \(options.maxSeconds) s")
        machine.start { result in
            MainActor.assumeIsolated {
                switch result {
                case .success: self.device.log("machine running")
                case .failure(let error): self.finish("the machine did not start: \(error.localizedDescription)", status: 1)
                }
            }
        }
        let sinks = device.sinks
        if !sinks.isEmpty {
            // Ends with the runner; the process ends with it in any case.
            Task { @MainActor [weak self, device] in
                while self != nil {
                    try? await Task.sleep(for: .milliseconds(50))
                    for sink in sinks { sink.tick(atMilliseconds: device.milliseconds) }
                }
            }
        }
        let limit = options.maxSeconds
        Task { @MainActor in
            try? await Task.sleep(for: .seconds(limit))
            self.stopAtLimit()
        }
    }

    private func stopAtLimit() {
        guard let machine else { return }
        device.log("time limit reached, stopping the machine")
        machine.stop { error in
            MainActor.assumeIsolated {
                self.finish("stopped at the time limit" + (error.map { ": \($0.localizedDescription)" } ?? ""), status: 0)
            }
        }
    }

    /// Writes the newest picture and the command record to `commands.log` (private), then ends the process.
    private func finish(_ text: String, status: Int32) {
        device.log(text)
        for sink in device.sinks { sink.finish() }
        let report = Data(device.report().utf8)
        if FileManager.default.createFile(atPath: options.commandsLogPath, contents: report, attributes: [.posixPermissions: 0o600]) {
            device.log("command record written to \(options.commandsLogPath)")
        } else {
            device.log("could not write \(options.commandsLogPath)")
        }
        exit(status)
    }

    nonisolated func guestDidStop(_ virtualMachine: VZVirtualMachine) {
        MainActor.assumeIsolated { finish("the guest stopped", status: 0) }
    }

    nonisolated func virtualMachine(_ virtualMachine: VZVirtualMachine, didStopWithError error: any Error) {
        MainActor.assumeIsolated { finish("the machine stopped with an error: \(error.localizedDescription)", status: 1) }
    }
}
```

```swift
import Foundation
import ImageIO
import UniformTypeIdentifiers
import VirtioGPURunnerCore

/// Writes the newest scanout picture to `scanout.png` in the scratch directory, at most once a second and once at the end,
/// so a run can be checked without a window. Each write goes to a private temporary file that is then renamed over the
/// last one, so a reader never sees half a file.
@MainActor
final class ScanoutDump: ScanoutSink {
    private let path: String
    private var throttle = FrameThrottle(intervalMilliseconds: 1000)
    private var written = 0
    private let log: (String) -> Void

    init(path: String, log: @escaping (String) -> Void) {
        self.path = path
        self.log = log
    }

    func show(_ update: ScanoutUpdate, atMilliseconds now: UInt64) {
        if let frame = throttle.offer(update, atMilliseconds: now) { write(frame) }
    }

    func tick(atMilliseconds now: UInt64) {
        if let frame = throttle.tick(atMilliseconds: now) { write(frame) }
    }

    func finish() {
        if let frame = throttle.drain() { write(frame) }
        log("scanout pictures written: \(written)")
    }

    private func write(_ update: ScanoutUpdate) {
        let temporary = path + ".partial"
        guard let image = ScanoutPicture.cgImage(update),
            let destination = CGImageDestinationCreateWithURL(
                URL(filePath: temporary) as CFURL, UTType.png.identifier as CFString, 1, nil)
        else {
            log("could not make a picture of the \(update.width) x \(update.height) scanout")
            return
        }
        CGImageDestinationAddImage(destination, image, nil)
        guard CGImageDestinationFinalize(destination) else {
            log("could not write \(temporary)")
            return
        }
        do {
            try FileManager.default.setAttributes([.posixPermissions: 0o600], ofItemAtPath: temporary)
            if FileManager.default.fileExists(atPath: path) {
                _ = try FileManager.default.replaceItemAt(URL(filePath: path), withItemAt: URL(filePath: temporary))
            } else {
                try FileManager.default.moveItem(atPath: temporary, toPath: path)
            }
        } catch {
            log("could not put \(path) in place: \(error.localizedDescription)")
            return
        }
        written += 1
        if written == 1 { log("first scanout picture written to \(path): \(update.width) x \(update.height)") }
    }
}
```

```swift
import CoreGraphics
import Foundation
import VirtioGPURunnerCore

/// Where the scanout's pictures go: the runner's window, the PNG file, or both.
@MainActor
protocol ScanoutSink: AnyObject {
    /// A new picture of the scanout, from a flush.
    func show(_ update: ScanoutUpdate, atMilliseconds now: UInt64)
    /// Called a few times a second, so a picture kept back by the throttle is shown.
    func tick(atMilliseconds now: UInt64)
    /// The end of the run: the newest picture is shown or written whatever the time.
    func finish()
}

/// The scanout's pixels as a CGImage in the sRGB space, red, green, blue and a padding byte.
enum ScanoutPicture {
    static func cgImage(_ update: ScanoutUpdate) -> CGImage? {
        guard update.width > 0, update.height > 0, let bytes = ScanoutImage.rgbx(update),
            let provider = CGDataProvider(data: Data(bytes) as CFData), let space = CGColorSpace(name: CGColorSpace.sRGB)
        else { return nil }
        // Unsafe only through the type of `decode`, which is nil here: no pointer is passed.
        return unsafe CGImage(
            width: Int(update.width), height: Int(update.height), bitsPerComponent: 8, bitsPerPixel: 32,
            bytesPerRow: Int(update.width) * GPUPixelFormat.bytesPerPixel, space: space,
            bitmapInfo: CGBitmapInfo(rawValue: CGImageAlphaInfo.noneSkipLast.rawValue), provider: provider, decode: nil,
            shouldInterpolate: false, intent: .defaultIntent)
    }
}
```

```swift
/// The runner's own window, showing the scanout as the contents of a layer, at most 30 times a second. It is made at the
/// first picture, sized to it, and never takes keys or clicks to the guest (the machine has no keyboard or pointer).
@MainActor
final class ScanoutWindow: ScanoutSink {
    private var window: NSWindow?
    private var throttle = FrameThrottle(intervalMilliseconds: 33)

    func show(_ update: ScanoutUpdate, atMilliseconds now: UInt64) {
        if let frame = throttle.offer(update, atMilliseconds: now) { present(frame) }
    }

    func tick(atMilliseconds now: UInt64) {
        if let frame = throttle.tick(atMilliseconds: now) { present(frame) }
    }

    func finish() {
        if let frame = throttle.drain() { present(frame) }
    }

    private func present(_ update: ScanoutUpdate) {
        guard let image = ScanoutPicture.cgImage(update) else { return }
        let view = window?.contentView ?? makeWindow(width: Int(update.width), height: Int(update.height))
        view.layer?.contents = image
    }

    private func makeWindow(width: Int, height: Int) -> NSView {
        let window = NSWindow(
            contentRect: NSRect(x: 0, y: 0, width: width, height: height), styleMask: [.titled, .resizable, .miniaturizable],
            backing: .buffered, defer: false)
        window.title = "Virtio GPU scanout"
        window.isReleasedWhenClosed = false
        let view = NSView(frame: NSRect(x: 0, y: 0, width: width, height: height))
        view.wantsLayer = true
        view.layer?.contentsGravity = .resizeAspect
        window.contentView = view
        window.center()
        window.orderFront(nil)
        self.window = window
        return view
    }
}
```

```swift
/// Why a command line was refused.
package enum RunnerArgumentError: Error, Equatable, CustomStringConvertible {
    case unknownOption(String)
    case missingValue(option: String)
    case notANumber(option: String, value: String)
    case outOfRange(option: String, value: Int, allowed: ClosedRange<Int>)
    case missing(option: String)
    case notAbsolute(option: String, path: String)
    case refusedDirectory(String)

    package var description: String {
        switch self {
        case .unknownOption(let option): "unknown option '\(option)'"
        case .missingValue(let option): "\(option) needs a value"
        case .notANumber(let option, let value): "\(option) takes a whole number, not '\(value)'"
        case .outOfRange(let option, let value, let allowed):
            "\(option) \(value) is outside \(allowed.lowerBound)...\(allowed.upperBound)"
        case .missing(let option): "\(option) is required"
        case .notAbsolute(let option, let path): "\(option) needs an absolute path without '..', not '\(path)'"

          --memory-mib <n>     1024 to 2048 (default 2048)
          --max-seconds <n>    10 to 3600 (default 900); the machine is stopped then, whatever it does
          --probe-config-change <s>  1 to 600: that many seconds after the driver is ready, signal a display change through
                               the configuration space and record whether the guest asks for the display information again
          --headless-dump      write the newest scanout picture to scanout.png in the directory (at most once a second)
          --window             show the scanout in a window of the runner's own
          --dry-run            build and validate the configuration, start nothing
          --describe           print the virtio identity and the configuration space the guest sees
        """

    /// Reads `arguments` (the program name already removed). `home` is the user's home folder, refused as a scratch
    /// directory with everything in its `Library`.
    package static func parse(_ arguments: [String], home: String) throws(RunnerArgumentError) -> RunnerRequest {
        var values: [String: String] = [:]
        var dryRun = false
        var headlessDump = false
        var window = false
        var index = 0
        while index < arguments.count {
            let argument = arguments[index]
            index += 1
            switch argument {
            case "-h", "--help": return .help
            case "--describe": return .describe
            case "--dry-run": dryRun = true
            case "--headless-dump": headlessDump = true
            case "--window": window = true
            case "--directory", "--disk", "--seed", "--cpus", "--memory-mib", "--max-seconds", "--probe-config-change":
                guard index < arguments.count else { throw .missingValue(option: argument) }
                values[argument] = arguments[index]
                index += 1
            default: throw .unknownOption(argument)
            }
        }
        guard let directory = values["--directory"] else { throw .missing(option: "--directory") }
        guard let disk = values["--disk"] else { throw .missing(option: "--disk") }
        var options = RunnerOptions(
            directory: try scratchDirectory(directory, home: home), disk: try absolute(disk, option: "--disk"),
            seed: try values["--seed"].map { path throws(RunnerArgumentError) in try absolute(path, option: "--seed") })
        options.cpuCount = try number(values["--cpus"], option: "--cpus", in: RunnerOptions.cpuRange) ?? options.cpuCount
        options.memoryMiB = try number(values["--memory-mib"], option: "--memory-mib", in: RunnerOptions.memoryRange) ?? options.memoryMiB
        options.maxSeconds =
            try number(values["--max-seconds"], option: "--max-seconds", in: RunnerOptions.secondsRange) ?? options.maxSeconds
        options.probeConfigChangeSeconds = try number(
            values["--probe-config-change"], option: "--probe-config-change", in: RunnerOptions.probeRange)
        options.dryRun = dryRun
        options.headlessDump = headlessDump
        options.window = window
        return .run(options)
    }

    private static func number(_ text: String?, option: String, in range: ClosedRange<Int>) throws(RunnerArgumentError) -> Int? {
        guard let text else { return nil }
        guard let value = Int(text) else { throw .notANumber(option: option, value: text) }
        guard range.contains(value) else { throw .outOfRange(option: option, value: value, allowed: range) }
        return value
    }

    /// `path` without a trailing slash, when it is absolute and has no `.` or `..` component.
    private static func absolute(_ path: String, option: String) throws(RunnerArgumentError) -> String {
        let components = path.split(separator: "/", omittingEmptySubsequences: true)
        guard path.hasPrefix("/"), !components.contains(".."), !components.contains(".") else {
            throw .notAbsolute(option: option, path: path)
        }
        return "/" + components.joined(separator: "/")
    }

    /// Refuses the root of the disk, the home folder and anything in its `Library`, by their names (the run creates the
    /// directory itself, so a link planted there in advance is the caller's own).
    private static func scratchDirectory(_ path: String, home: String) throws(RunnerArgumentError) -> String {
        let directory = try absolute(path, option: "--directory")
        let homeFolder = (try? absolute(home, option: "home")) ?? home
        if directory == "/" { throw .refusedDirectory("the root of the disk is not a scratch directory") }
        if directory == homeFolder { throw .refusedDirectory("your home folder is not a scratch directory") }
        let library = homeFolder + "/Library"
        if directory == library || directory.hasPrefix(library + "/") {
            throw .refusedDirectory("\(directory) is inside your Library; give a scratch directory under /tmp")
        }
        return directory
    }
}
```

```swift
/// The scanout's pixels as the host shows them: four bytes per pixel, red, green, blue, then a padding byte of 255, rows
/// packed. A scanout has no meaningful alpha, so the guest's alpha or padding byte is never shown.
public enum ScanoutImage {
    /// Where red, green and blue sit in one pixel, counted in bytes from its first.
    public struct ChannelOffsets: Equatable, Sendable {
        public let red: Int
        public let green: Int
        public let blue: Int
    }

    /// Where red, green and blue sit in one pixel of `format`. A virtio format names its bytes in memory order:
    /// `b8g8r8x8` is blue, green, red, padding.
    public static func channelOffsets(_ format: GPUPixelFormat) -> ChannelOffsets {
        switch format {
        case .b8g8r8a8, .b8g8r8x8: ChannelOffsets(red: 2, green: 1, blue: 0)
        case .a8r8g8b8, .x8r8g8b8: ChannelOffsets(red: 1, green: 2, blue: 3)
        case .r8g8b8a8, .r8g8b8x8: ChannelOffsets(red: 0, green: 1, blue: 2)
        case .a8b8g8r8, .x8b8g8r8: ChannelOffsets(red: 3, green: 2, blue: 1)
        }
    }

    /// `update`'s pixels in red, green, blue, padding order. The store guarantees `width × height × 4` bytes; anything shorter
    /// gives no image rather than a partial one.
    public static func rgbx(_ update: ScanoutUpdate) -> [UInt8]? {
        let count = Int(update.width) * Int(update.height) * GPUPixelFormat.bytesPerPixel
        guard update.pixels.count == count else { return nil }
        let offsets = channelOffsets(update.format)
        var out = [UInt8](repeating: 255, count: count)
        var index = 0
        while index < count {
            out[index] = update.pixels[index + offsets.red]
            out[index + 1] = update.pixels[index + offsets.green]
            out[index + 2] = update.pixels[index + offsets.blue]
            index += GPUPixelFormat.bytesPerPixel
        }
        return out
    }
}

/// Shows at most one frame per `interval` milliseconds: a frame offered sooner is kept and shown by a later tick, and the
/// last one kept is handed out at the end, so the newest picture is never lost and the host's work stays bounded.
public struct FrameThrottle: Sendable {
    public let interval: UInt64
    private var lastShownAt: UInt64?
    private var pending: ScanoutUpdate?

    public init(intervalMilliseconds: UInt64) {
        interval = intervalMilliseconds
    }

    /// The frame to show now, or `nil` when it was kept for later.
    public mutating func offer(_ update: ScanoutUpdate, atMilliseconds now: UInt64) -> ScanoutUpdate? {
        pending = update
        return tick(atMilliseconds: now)
    }

    /// The kept frame, when the interval since the last one shown has passed.
    public mutating func tick(atMilliseconds now: UInt64) -> ScanoutUpdate? {
        guard let frame = pending else { return nil }
        if let last = lastShownAt, now < last + interval { return nil }
        lastShownAt = now
        pending = nil
        return frame
    }

    /// The kept frame, whatever the time: at the end of the run.
    public mutating func drain() -> ScanoutUpdate? {
        defer { pending = nil }
        return pending
    }
}
```
