```swift
import ControlProtocol
import Foundation
import Observation

/// The token list in Settings (ADR-0056): make a token and show its secret once, list the tokens, revoke one.
@MainActor @Observable
final class ControlTokensModel {
    private(set) var tokens: [ControlToken] = []
    /// The secret of the token just made, shown until the user dismisses it; then it is gone, only its digest is stored.
    var newSecret: String?
    var errorMessage: String?
    var draftName = ""
    var draftCapabilities: Set<ControlCapability> = [.list, .state]
    @ObservationIgnored private let store: @MainActor () throws -> ControlTokenStore

    /// `store` is the service's token store; a test gives one in a scratch folder.
    init(store: @escaping @MainActor () throws -> ControlTokenStore = { try ControlService.shared.tokenStore() }) {
        self.store = store
    }

    func refresh() {
        do {
            tokens = try store().all.sorted { $0.created < $1.created }
        } catch {
            errorMessage = error.localizedDescription
        }
    }

    func create() {
        do {
            let made = try store().create(name: draftName, capabilities: draftCapabilities)
            newSecret = made.secret
            draftName = ""
            errorMessage = nil
            refresh()
        } catch {
            errorMessage = error.localizedDescription
        }
    }

    func revoke(_ id: UUID) {
        do {
            try store().revoke(id)
            errorMessage = nil
            refresh()
        } catch {
            errorMessage = error.localizedDescription
        }
    }

    func setCapability(_ capability: ControlCapability, _ isOn: Bool) {
        if isOn { draftCapabilities.insert(capability) } else { draftCapabilities.remove(capability) }
    }

    /// What a token allows, in a fixed order, for its row.
    static func summary(of token: ControlToken) -> String {
        ControlCapability.allCases.filter(token.capabilities.contains).map(\.label).joined(separator: ", ")
    }
}
```

```swift
import AppKit
import ControlProtocol
import SwiftUI

/// Settings' list of control tokens: make one (its secret is shown once), see what each allows, revoke one.
struct ControlTokensSection: View {
    @State private var model = ControlTokensModel()

    var body: some View {
        Section {
            ForEach(model.tokens) { token in
                LabeledContent {
                    Button("Revoke", role: .destructive) { model.revoke(token.id) }
                } label: {
                    Text(token.name)
                    Text(ControlTokensModel.summary(of: token))
                }
            }
            if let secret = model.newSecret {
                VStack(alignment: .leading, spacing: 6) {
                    Text("Copy this token now. VMRunner keeps only its fingerprint and cannot show it again.")
                        .font(.caption)
                    Text(secret).font(.body.monospaced()).textSelection(.enabled)
                    HStack {
                        Button("Copy") {
                            NSPasteboard.general.clearContents()
                            NSPasteboard.general.setString(secret, forType: .string)
                        }
                        Button("Done") { model.newSecret = nil }
                    }
                }
            }
            TextField("Name of a new token", text: $model.draftName)
            ForEach(ControlCapability.allCases, id: \.self) { capability in
                Toggle(
                    capability.label,
                    isOn: Binding(
                        get: { model.draftCapabilities.contains(capability) }, set: { model.setCapability(capability, $0) }))
            }
            Button("Create Token", action: model.create)
                .disabled(model.draftName.trimmingCharacters(in: .whitespaces).isEmpty || model.draftCapabilities.isEmpty)
            if let error = model.errorMessage {
                Text(error).font(.caption).foregroundStyle(.red)
            }
        } header: {
            HelpHeader("Control tokens", topic: .controlTokens)
        }
        .task { model.refresh() }
    }
}
```

```swift
import ControlProtocol
import Foundation
import Testing

@testable import VMRunner

/// The token list of Settings, on a store in a scratch folder.
@MainActor
struct ControlTokensModelTests {
    @Test
    func aNewTokenIsShownOnceListedWithoutItsSecretAndWorks() throws {
        let folder = try scratchDirectory()
        defer { try? FileManager.default.removeItem(at: folder) }
        let file = folder.appending(path: "tokens.json")
        let store = try ControlTokenStore(file: file)
        let model = ControlTokensModel(store: { store })
        model.draftName = "agent"
        model.draftCapabilities = [.list, .start]
        model.create()

        let secret = try #require(model.newSecret)
        #expect(model.errorMessage == nil)
        #expect(model.draftName.isEmpty)
        let token = try #require(model.tokens.first)
        #expect(model.tokens.map(\.name) == ["agent"])
        #expect(ControlTokensModel.summary(of: token) == "List machines, Start")
        #expect(store.authenticate(secret)?.capabilities == [.list, .start])
        #expect(try !String(contentsOf: file, encoding: .utf8).contains(secret))

        model.newSecret = nil
        model.refresh()
        #expect(model.newSecret == nil, "the secret is not shown again")
    }

    @Test
    func revokingRemovesTheTokenAndItsSecretFails() throws {
        let folder = try scratchDirectory()
        defer { try? FileManager.default.removeItem(at: folder) }
        let store = try ControlTokenStore(file: folder.appending(path: "tokens.json"))
        let model = ControlTokensModel(store: { store })
        model.draftName = "agent"
        model.create()
        let secret = try #require(model.newSecret)
        let token = try #require(model.tokens.first)

        model.revoke(token.id)
        #expect(model.tokens.isEmpty)
        #expect(store.authenticate(secret) == nil)
        model.revoke(token.id)
        #expect(model.errorMessage == "There is no such token.")
    }

    @Test
    func aTokenWithoutANameOrACapabilityIsRefusedWithAMessage() throws {
        let folder = try scratchDirectory()
        defer { try? FileManager.default.removeItem(at: folder) }
        let store = try ControlTokenStore(file: folder.appending(path: "tokens.json"))
        let model = ControlTokensModel(store: { store })
        model.draftName = "  "
        model.create()
        #expect(model.errorMessage == "A token needs a name.")
        model.draftName = "agent"
        model.setCapability(.list, false)
        model.setCapability(.state, false)
        #expect(model.draftCapabilities.isEmpty)
        model.create()
        #expect(model.errorMessage == "A token needs at least one capability.")
        #expect(model.newSecret == nil && model.tokens.isEmpty)
    }

    @Test
    func aTokenListThatCannotBeReadSaysWhy() {
        let model = ControlTokensModel(store: { throw ControlTokenError.unreadable(path: "/nowhere/tokens.json") })
        model.refresh()
        #expect(model.errorMessage == "The token list at /nowhere/tokens.json cannot be read.")
    }
}
```
