Why (read from the code): refusals are published once per destination per five seconds with the rest counted
(`DenialSuppression`), but every answered DNS query publishes its own `FlowEvent` (`UserNetCore.deliver`), so a guest asking the
same name in a loop floods the event stream, the helper's log and the app's flow log. A new `AnswerFolding` does for answers what
`DenialSuppression` does for refusals, keyed by the question (name and record type), the response code and the server. The key
includes the record type so that an A and an AAAA question stay two events (`dnsGoesThroughTheInjectedResolverNotTheInternet`
counts four distinct answers). `DNSAnswer` therefore carries the question's type.

```swift
import Foundation

/// Decides which answered DNS queries become flow events, the way ``DenialSuppression`` does for refusals.
///
/// A guest that asks the same question again and again (a retry loop, a short TTL, a polling client) published one event, and
/// gave the app one log line, per answer. Here an answer publishes once per ``DenialSuppression/interval`` for each question,
/// response code and server; the ones held back are counted and go out as one more event (its `count` says how many) when the
/// interval ends. At most ``DenialSuppression/capacity`` keys are followed at once; past that an answer is published as it is,
/// so nothing is lost but the folding. Time is the stack's own clock, passed in.
struct AnswerFolding {
    /// What an answer is about, from the guest's side.
    struct Key: Hashable {
        let name: String
        let type: UInt16
        let responseCode: UInt8
        let server: UInt32
    }

    /// Answers that were held back, to be published as one event standing for `count` of them.
    struct Held: Equatable {
        let key: Key
        let count: Int
    }

    enum Admission: Equatable {
        /// Publish this answer. `held`, when set, is the count of the previous interval for the same key, not published yet.
        case publish(held: Held?)
        case suppress
    }

    private struct Entry {
        var until: TimeInterval
        var held: Int
    }

    private var entries: [Key: Entry] = [:]

    /// Judges one answer.
    mutating func admit(_ key: Key, now: TimeInterval) -> Admission {
        guard let entry = entries[key] else {
            guard entries.count < DenialSuppression.capacity else { return .publish(held: nil) }
            entries[key] = Entry(until: now + DenialSuppression.interval, held: 0)
            return .publish(held: nil)
        }
        if now < entry.until {
            entries[key] = Entry(until: entry.until, held: entry.held + 1)
            return .suppress
        }
        entries[key] = Entry(until: now + DenialSuppression.interval, held: 0)
        return .publish(held: entry.held > 0 ? Held(key: key, count: entry.held) : nil)
    }

    /// Forgets the keys whose interval is over and returns what was held back for them.
    mutating func expire(now: TimeInterval) -> [Held] {
        guard !entries.isEmpty else { return [] }
        let over = entries.filter { now >= $0.value.until }
        for key in over.keys { entries[key] = nil }
        return over.compactMap { $0.value.held > 0 ? Held(key: $0.key, count: $0.value.held) : nil }
    }
}
```

```swift
    /// Publishes the answer unless an equal one went out within the interval; then it is only counted (``AnswerFolding``).
    private func publishAnswered(_ answer: DNSAnswer) {
        let key = AnswerFolding.Key(name: answer.name, type: answer.type, responseCode: answer.responseCode, server: answer.server)
        guard case .publish(let held) = answers.admit(key, now: ctx.now) else { return }
        if let held { publishHeldAnswer(held) }
        ctx.host.publish(
            FlowEvent(
                flowID: ctx.makeToken(), transport: .dns, phase: .answered(responseCode: answer.responseCode),
                remote: (answer.server, 53), name: answer.name, time: ctx.now, bytesToGuest: UInt64(answer.message.count)))
    }

    /// Publishes what was held back for answers whose interval is over, or all of it when the stack stops (`everything`).
    func publishHeldAnswers(everything: Bool = false) {
        for held in answers.expire(now: everything ? .infinity : ctx.now) { publishHeldAnswer(held) }
    }

    private func publishHeldAnswer(_ held: AnswerFolding.Held) {
        ctx.host.publish(
            FlowEvent(
                flowID: ctx.makeToken(), transport: .dns, phase: .answered(responseCode: held.key.responseCode),
                remote: (held.key.server, 53), name: held.key.name, time: ctx.now, count: held.count))
    }
```

```swift
import Foundation
import Testing

@testable import NetStack

struct AnswerFoldingTests {
    private let key = AnswerFolding.Key(name: "host.test", type: 1, responseCode: 0, server: 1)

    /// `#expect` cannot hold a mutating call, so the table is driven through these.
    private func admit(_ table: inout AnswerFolding, _ key: AnswerFolding.Key, at now: TimeInterval) -> AnswerFolding.Admission {
        table.admit(key, now: now)
    }

    private func expire(_ table: inout AnswerFolding, at now: TimeInterval) -> [AnswerFolding.Held] {
        table.expire(now: now)
    }

    @Test func anAnswerIsPublishedOncePerIntervalAndTheRestAreCounted() {
        var table = AnswerFolding()
        #expect(admit(&table, key, at: 0) == .publish(held: nil))
        for second in 1...4 { #expect(admit(&table, key, at: TimeInterval(second)) == .suppress) }
        #expect(admit(&table, key, at: 6) == .publish(held: AnswerFolding.Held(key: key, count: 4)), "the next one carries the count")
    }

    @Test func anotherNameTypeCodeOrServerIsAnotherAnswer() {
        var table = AnswerFolding()
        _ = admit(&table, key, at: 0)
        let others = [
            AnswerFolding.Key(name: "other.test", type: 1, responseCode: 0, server: 1),
            AnswerFolding.Key(name: "host.test", type: 28, responseCode: 0, server: 1),
            AnswerFolding.Key(name: "host.test", type: 1, responseCode: 3, server: 1),
            AnswerFolding.Key(name: "host.test", type: 1, responseCode: 0, server: 2),
        ]
        for other in others { #expect(admit(&table, other, at: 1) == .publish(held: nil)) }
    }

    @Test func whatWasHeldGoesOutWhenTheIntervalEndsAndOnlyOnce() {
        var table = AnswerFolding()
        _ = admit(&table, key, at: 0)
        _ = admit(&table, key, at: 1)
        #expect(expire(&table, at: 4).isEmpty)
        #expect(expire(&table, at: 5) == [AnswerFolding.Held(key: key, count: 1)])
        #expect(expire(&table, at: 10).isEmpty)
        #expect(admit(&table, key, at: 11) == .publish(held: nil))
    }

    @Test func pastItsCapacityAnAnswerIsPublishedAsItIs() {
        var table = AnswerFolding()
        for index in 0..<DenialSuppression.capacity {
            _ = admit(&table, AnswerFolding.Key(name: "n\(index).test", type: 1, responseCode: 0, server: 1), at: 0)
        }
        let extra = AnswerFolding.Key(name: "extra.test", type: 1, responseCode: 0, server: 1)
        #expect(admit(&table, extra, at: 0) == .publish(held: nil))
        #expect(admit(&table, extra, at: 1) == .publish(held: nil), "it is not followed, so it is never held back")
    }
}
```
