Why: ADR-0012 item 5 and the study ("Delivery and integrity"): every `file:` script carries a SHA-256 in the spec, and the
host refuses to build the seed when a file does not match. The check reads each file once, bounded by `maxFileBytes`
(it reads at most one byte more than the limit, so a huge file is never loaded), refuses a symbolic link, a missing file,
a folder and a mismatch, each with the step's line, and returns the bytes it checked in a `VerifiedScripts` value that
only this check can make. The compiler (next cards) takes that value, so a seed can only carry bytes that were checked.
Known limit, stated in the code: a file swapped between the check and the read is not detected beyond the single read;
the bytes that are hashed are the bytes that are packed.

```swift
// The provisioning library has no caller outside its tests until a later change attaches the seed to a machine, so a
// declaration only the tests reach is expected here; every other dead-code rule still applies.
// @dw:disable referenced-only-by-tests

import CryptoKit
package import Foundation

/// The contents of every `file:` step of a spec, each read once and found to match the SHA-256 the spec gives. Only
/// `ScriptFiles.verify` makes one, so a seed is never built from a file nobody checked.
package struct VerifiedScripts: Sendable {
    private let files: [String: [UInt8]]

    fileprivate init(files: [String: [UInt8]]) {
        self.files = files
    }

    /// The checked bytes of the file a step names by `path`, or nil when no check covered it.
    func contents(of path: String) -> [UInt8]? {
        files[path]
    }
}

/// The host's check of a spec's script files, before anything is packed into a seed (ADR-0012, item 5). It fails closed:
/// the first file that is missing, not a regular file, reached through a symbolic link, larger than the limit or different
/// from its SHA-256 refuses the whole spec. The bytes it hashes are the bytes it returns, read once. A file swapped for a
/// link between the type check and the read is not refused as a link, but only bytes that match the digest are returned.
package enum ScriptFiles {
    package static func verify(
        _ spec: ProvisioningSpec, in directory: URL, limits: ProvisioningLimits = .standard
    ) throws -> VerifiedScripts {
        let base = directory.standardizedFileURL.resolvingSymlinksInPath()
        var contents: [String: [UInt8]] = [:]
        for step in spec.steps {
            guard case .file(let path, let sha256) = step.script else { continue }
            let bytes = try read(path, in: base, line: step.line, limits: limits)
            let actual = SHA256.hash(data: bytes).map { ($0 < 16 ? "0" : "") + String($0, radix: 16) }.joined()
            guard actual == sha256 else {
                throw ProvisioningError(line: step.line, "\(path) does not match its sha256: the spec says \(sha256), the file is \(actual)")
            }
            contents[path] = bytes
        }
        return VerifiedScripts(files: contents)
    }

    private static func read(_ path: String, in base: URL, line: Int, limits: ProvisioningLimits) throws -> [UInt8] {
        let file = base.appending(path: path)
        let values = try? file.resourceValues(forKeys: [.isRegularFileKey, .isSymbolicLinkKey])
        if values?.isSymbolicLink == true { throw ProvisioningError(line: line, "\(path) is a symbolic link") }
        guard values?.isRegularFile == true else { throw ProvisioningError(line: line, "\(path) is not a regular file") }
        guard file.resolvingSymlinksInPath().path == file.standardizedFileURL.path else {
            throw ProvisioningError(line: line, "\(path) goes through a symbolic link")
        }
        guard let handle = try? FileHandle(forReadingFrom: file) else { throw ProvisioningError(line: line, "cannot read \(path)") }
        defer { try? handle.close() }
        let data = (try? handle.read(upToCount: limits.maxFileBytes + 1)) ?? Data()
        guard data.count <= limits.maxFileBytes else {
            throw ProvisioningError(line: line, "\(path) is larger than \(limits.maxFileBytes) bytes")
        }
        return [UInt8](data)
    }
}
```

```swift
import Foundation

/// A new, empty folder under the temporary directory for one test; the test removes it when it ends.
func makeScratchFolder() throws -> URL {
    let folder = FileManager.default.temporaryDirectory.appending(
        path: "provisioning-tests-\(UUID().uuidString)", directoryHint: .isDirectory)
    try FileManager.default.createDirectory(at: folder, withIntermediateDirectories: true)
    return folder
}
```

```swift
import Foundation
import Testing

@testable import Provisioning

/// The script `scripts/hello.sh` of the spec below, and its SHA-256 (computed outside this code, with a separate tool).
private let hello = "#!/bin/sh\necho hello\n"
private let helloDigest = "bfdeaeb08cffb6a36438bcd12dda25417e3cdd36f1e7e482a2849d539225288b"

private func spec(file: String = "scripts/hello.sh", sha256: String = helloDigest) throws -> ProvisioningSpec {
    try ProvisioningSpec(
        parsing: """
            apiVersion: vmrunner.dev/v1alpha1
            kind: Provisioning
            metadata:
              name: node
            spec:
              provision:
              - at: first-boot
                file: \(file)
                sha256: \(sha256)
            """)
}

struct ScriptFilesTests {
    private func write(_ text: String, to path: String, in folder: URL) throws {
        let file = folder.appending(path: path)
        try FileManager.default.createDirectory(at: file.deletingLastPathComponent(), withIntermediateDirectories: true)
        try Data(text.utf8).write(to: file)
    }

    private func refusal(in folder: URL, limits: ProvisioningLimits = .standard) throws -> ProvisioningError {
        let checked = try spec()
        return try #require(throws: ProvisioningError.self) { try ScriptFiles.verify(checked, in: folder, limits: limits) }
    }

    @Test
    func aMatchingFileIsReturnedWithTheBytesThatWereHashed() throws {
        let folder = try makeScratchFolder()
        defer { try? FileManager.default.removeItem(at: folder) }
        try write(hello, to: "scripts/hello.sh", in: folder)
        let scripts = try ScriptFiles.verify(try spec(), in: folder)
        #expect(scripts.contents(of: "scripts/hello.sh") == Array(hello.utf8))
        #expect(scripts.contents(of: "scripts/other.sh") == nil)
    }

    @Test
    func aFileThatDoesNotMatchItsDigestRefusesTheSpecWithTheStepsLine() throws {
        let folder = try makeScratchFolder()
        defer { try? FileManager.default.removeItem(at: folder) }
        try write("#!/bin/sh\necho hello!\n", to: "scripts/hello.sh", in: folder)
        let error = try refusal(in: folder)
        #expect(error.line == 7)
        #expect(error.message.contains("does not match its sha256"), "\(error)")
        #expect(error.message.contains(helloDigest), "\(error)")
    }

    @Test
    func aMissingFileOrAFolderIsRefused() throws {
        let folder = try makeScratchFolder()
        defer { try? FileManager.default.removeItem(at: folder) }
        #expect(try refusal(in: folder).message == "scripts/hello.sh is not a regular file")
        try FileManager.default.createDirectory(at: folder.appending(path: "scripts/hello.sh"), withIntermediateDirectories: true)
        #expect(try refusal(in: folder).message == "scripts/hello.sh is not a regular file")
    }

    @Test
    func aSymbolicLinkIsRefusedEvenToAMatchingFile() throws {
        let folder = try makeScratchFolder()
        defer { try? FileManager.default.removeItem(at: folder) }
        try write(hello, to: "scripts/real.sh", in: folder)
        try FileManager.default.createSymbolicLink(
            at: folder.appending(path: "scripts/hello.sh"), withDestinationURL: folder.appending(path: "scripts/real.sh"))
        let error = try refusal(in: folder)
        #expect(error.line == 7)
        #expect(error.message.contains("symbolic link"), "\(error)")
    }

    @Test
    func aFolderOnTheWayThatIsALinkIsRefused() throws {
        let folder = try makeScratchFolder()
        defer { try? FileManager.default.removeItem(at: folder) }
        try write(hello, to: "elsewhere/hello.sh", in: folder)
        try FileManager.default.createSymbolicLink(
            at: folder.appending(path: "scripts"), withDestinationURL: folder.appending(path: "elsewhere"))
        #expect(try refusal(in: folder).message == "scripts/hello.sh goes through a symbolic link")
    }

    @Test
    func aSpecWithoutFilesReadsNothing() throws {
        let inline = try ProvisioningSpec(
            parsing: """
                apiVersion: vmrunner.dev/v1alpha1
                kind: Provisioning
                metadata:
                  name: node
                spec:
                  provision:
                  - at: first-boot
                    script: |
                      #!/bin/sh
                      true
                """)
        let scripts = try ScriptFiles.verify(inline, in: URL(filePath: "/nonexistent-\(UUID().uuidString)"))
        #expect(scripts.contents(of: "scripts/hello.sh") == nil)
    }

    // MARK: Limits

    @Test
    func aFileAtTheSizeLimitIsReadAndOneByteMoreIsRefused() throws {
        let folder = try makeScratchFolder()
        defer { try? FileManager.default.removeItem(at: folder) }
        try write(hello, to: "scripts/hello.sh", in: folder)
        var limits = ProvisioningLimits.standard
        limits.maxFileBytes = hello.utf8.count
        #expect(try ScriptFiles.verify(try spec(), in: folder, limits: limits).contents(of: "scripts/hello.sh")?.count == 21)
        limits.maxFileBytes = hello.utf8.count - 1
        #expect(try refusal(in: folder, limits: limits).message == "scripts/hello.sh is larger than 20 bytes")
    }
}
```
