Why (read from the code): `GuestTxQueue` keeps up to 2,048 frames, each in a slot of `mtu + 14 + 64` bytes allocated the first
time it is needed. With the largest MTU the stack accepts (65,535, `UserNetConfig.problem`), that is 2,048 × 65,613 bytes, about
128 MiB per guest link if the guest stops reading. A byte budget of 8 MiB caps the slots instead: 2,048 frames at the default
MTU of 1,500 (unchanged), 924 at 9,000, 127 at 65,535. A full queue already refuses frames (TCP retransmits, UDP is lost and
`framesLostToGuest` counts it), so nothing else changes.

```swift
/// blocks costs nothing and one that blocks occasionally stops allocating after the first burst. The queue is
/// bounded in frames and in bytes (the slots together never take more than the byte budget, so a large MTU means fewer
/// slots); a full queue refuses frames, which for TCP means a retransmission and for UDP a lost datagram.
```

   - replace

```swift
    init(slotCapacity: Int, maximumSlots: Int) {
        self.slotCapacity = slotCapacity
        self.maximumSlots = maximumSlots
    }
```

     with

```swift
    /// The most bytes of slots one guest link keeps: 2,048 frames at the default MTU, 127 at the largest.
    static let byteBudget = 8 << 20

    /// - Parameters:
    ///   - slotCapacity: The largest frame a slot holds.
    ///   - maximumSlots: The most frames the queue holds.
    ///   - maximumBytes: The most bytes all slots may take together; at least one slot is always allowed.
    init(slotCapacity: Int, maximumSlots: Int, maximumBytes: Int) {
        self.slotCapacity = slotCapacity
        self.maximumSlots = max(1, min(maximumSlots, maximumBytes / max(slotCapacity, 1)))
    }
```

## Test

Create `Tests/NetStackTests/GuestTxQueueTests.swift` with exactly:

```swift
import Testing

@testable import NetStack

struct GuestTxQueueTests {
    /// How many 100-byte frames the queue takes, out of `attempts`, before it refuses them.
    private func accepted(by queue: GuestTxQueue, attempts: Int = 100) -> Int {
        let frame = [UInt8](repeating: 7, count: 100)
        return (0..<attempts).filter { _ in ByteView.withBytes(of: frame) { queue.enqueue($0) } }.count
    }

    @Test func theQueueHoldsNoMoreSlotsThanItsByteBudgetPaysFor() {
        #expect(accepted(by: GuestTxQueue(slotCapacity: 1_000, maximumSlots: 50, maximumBytes: 4_000)) == 4)
        #expect(accepted(by: GuestTxQueue(slotCapacity: 1_000, maximumSlots: 3, maximumBytes: 4_000)) == 3, "nor more than its frames")
        #expect(accepted(by: GuestTxQueue(slotCapacity: 1_000, maximumSlots: 50, maximumBytes: 10)) == 1, "one slot whatever the budget")
    }

    @Test func theLargestMTUKeepsTheQueueWithinItsBudget() {
        let slot = 65_535 + EthernetHeader.length + 64
        let queue = GuestTxQueue(slotCapacity: slot, maximumSlots: 2_048, maximumBytes: GuestTxQueue.byteBudget)
        #expect(accepted(by: queue, attempts: 300) == GuestTxQueue.byteBudget / slot)
        #expect(GuestTxQueue.byteBudget / (1_500 + EthernetHeader.length + 64) >= 2_048, "the default MTU keeps its 2,048 frames")
    }

    @Test func aDrainedQueueTakesFramesAgain() {
        let queue = GuestTxQueue(slotCapacity: 1_000, maximumSlots: 2, maximumBytes: 4_000)
        #expect(accepted(by: queue) == 2)
        queue.dropFirst()
        #expect(accepted(by: queue) == 1)
        queue.removeAll()
        #expect(queue.isEmpty && accepted(by: queue) == 2)
    }
}
```
