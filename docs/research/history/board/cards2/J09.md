One commit, signed, atomic; `swift run vmr check` 8/8 before it. No board ids, no outside project names in code, docs, comments or commit messages. **No test starts a real machine**: they use `ControllerRig` (fake guests) and scratch folders.

Why: ADR-0056 decision 8. `VMLibrary.startMachine(_:interactive:)` and `suspendMachine(_:)` start a task and return at
once, so a caller cannot know whether the machine started. The control service must answer "running" or say why not. This
card adds `startAndWait(_:)` and `suspendAndWait(_:)`, which do the same work, return when it is done, and say what came of
it (`MachineCommandResult`); the existing menu and toolbar paths call them, so there is one way to start and one to suspend.
A start through `startAndWait` never asks the user anything: host paths the user has not approved refuse it (the
interactive prompt of `startMachine` stays before it, unchanged). The stop variant comes with its caller in J10.

A controller can set `errorMessage` on a success (a notice after a cold boot that replaced a saved state), so the result
looks at the state first: the state the request wanted is success, whatever the message says.

1. Create `Sources/VMRunner/Model/VMLibrary+Commands.swift`:

```swift
import Foundation

/// What came of a request to start, suspend or stop a machine, for a caller that waits for it (the control service, and the
/// menus through the same path).
enum MachineCommandResult: Equatable, Sendable {
    /// The request ran; the machine is now in this state.
    case done(MachineState)
    /// The request was not made, for this reason, written for the person or agent that asked.
    case refused(String)
    /// The request was made and failed, with the controller's message.
    case failed(String)
}

extension VMLibrary {
    /// Starts a machine without asking anything, and returns once the start has finished: the machine runs, or the start
    /// failed. Host paths the user has not approved refuse the start, as for a launch argument: nobody is there to ask.
    func startAndWait(_ id: UUID) async -> MachineCommandResult {
        guard let config = machine(id) else { return .refused("There is no such machine.") }
        guard config.isProvisioned else { return .refused("“\(config.name)” is not set up yet.") }
        guard jobs[id] == nil else { return .refused("“\(config.name)” is being set up.") }
        let controller = controller(for: id)
        guard controller.state.canStart else { return .refused("“\(config.name)” is already \(controller.state.activity).") }
        guard unapprovedHostPaths(for: id).isEmpty else {
            return .refused("“\(config.name)” uses host paths you have not approved in VMRunner. Start it once from the app.")
        }
        await controller.start(config, bundle: bundle(id), library: self, windowSize: displayHint)
        return Self.result(of: controller) { $0 == .running }
    }

    /// Suspends a machine and returns once its memory is saved and it has stopped, or the suspend was refused or failed.
    func suspendAndWait(_ id: UUID) async -> MachineCommandResult {
        guard let config = machine(id) else { return .refused("There is no such machine.") }
        let controller = controller(for: id)
        guard controller.state.allowsSuspend else { return .refused("“\(config.name)” is not running.") }
        // A message left from an earlier request must not read as this one's failure.
        controller.errorMessage = nil
        await controller.suspend()
        return Self.result(of: controller) { $0 == .stopped }
    }

    /// The machine reached the state the request wanted: done, whatever notice the controller holds. Otherwise its message is
    /// the failure; without one, the state it is in.
    static func result(of controller: VMController, reached wanted: (MachineState) -> Bool) -> MachineCommandResult {
        if wanted(controller.state) { return .done(controller.state) }
        if let message = controller.errorMessage { return .failed(message) }
        return .done(controller.state)
    }
}
```

```swift
import Foundation
import TestKit
import Testing

@testable import VMRunner

/// Starting and suspending through the library and waiting for the result, with fake machines in a scratch folder.
@MainActor
struct LibraryCommandTests {
    @Test
    func aStartReturnsOnceTheMachineRuns() async throws {
        let rig = try ControllerRig()
        defer { rig.cleanUp() }
        rig.factory.add()
        #expect(await rig.library.startAndWait(rig.id) == .done(.running))
        #expect(rig.controller.state == .running)
    }

    @Test
    func aSecondStartIsRefusedAndSaysWhatTheMachineIsDoing() async throws {
        let rig = try ControllerRig()
        defer { rig.cleanUp() }
        rig.factory.add()
        _ = await rig.library.startAndWait(rig.id)
        #expect(await rig.library.startAndWait(rig.id) == .refused("“fake” is already running."))
    }

    @Test
    func aStartThatFailsSaysSoAndLeavesTheMachineStopped() async throws {
        let rig = try ControllerRig()
        defer { rig.cleanUp() }
        rig.factory.failure = "no disk"
        let result = await rig.library.startAndWait(rig.id)
        guard case .failed(let message) = result else {
            Issue.record("expected a failure, got \(result)")
            return
        }
        #expect(message.contains("no disk"))
        #expect(rig.controller.state == .stopped)
    }

    @Test
    func anUnknownMachineOrOneThatIsNotSetUpIsRefused() async throws {
        let rig = try ControllerRig()
        defer { rig.cleanUp() }
        #expect(await rig.library.startAndWait(UUID()) == .refused("There is no such machine."))
        var config = rig.config
        config.isProvisioned = false
        rig.library.update(config)
        #expect(await rig.library.startAndWait(rig.id) == .refused("“fake” is not set up yet."))
        #expect(rig.factory.preparations == 0, "nothing was prepared")
    }

    @Test
    func hostPathsTheUserHasNotApprovedRefuseTheStartWithoutAsking() async throws {
        let rig = try ControllerRig()
        defer { rig.cleanUp() }
        var config = rig.config
        config.sharedFolders = [SharedFolder(path: "/tmp/never-approved-\(UUID().uuidString)")]
        rig.library.update(config)
        let result = await rig.library.startAndWait(rig.id)
        #expect(result == .refused("“fake” uses host paths you have not approved in VMRunner. Start it once from the app."))
        #expect(rig.library.pendingApproval == nil, "nobody is asked")
        #expect(rig.factory.preparations == 0)
    }

    @Test
    func aMachineInSetupIsRefused() async throws {
        let directory = try scratchDirectory()
        defer { try? FileManager.default.removeItem(at: directory) }
        let gate = AsyncGate()
        let library = VMLibrary(machinesDirectory: directory, provision: { _, _, _ in try await gate.waitUntilOpen() })
        var config = VMConfig.new(name: "installing", guest: .linux)
        config.isProvisioned = true
        try library.add(config)
        library.beginSetup(config.id)
        let job = try #require(library.jobs[config.id])

        #expect(await library.startAndWait(config.id) == .refused("“installing” is being set up."))

        library.cancelJob(config.id)
        gate.open()
        await job.task?.value
    }

    @Test
    func aSuspendReturnsOnceTheMachineHasStoppedWithItsImageSaved() async throws {
        let rig = try ControllerRig()
        defer { rig.cleanUp() }
        rig.factory.add()
        _ = await rig.library.startAndWait(rig.id)
        #expect(await rig.library.suspendAndWait(rig.id) == .done(.stopped))
        #expect(rig.bundle.hasSavedState)
    }

    @Test
    func aSuspendTheMachineCannotDoIsAFailureWithItsReason() async throws {
        let rig = try ControllerRig()
        defer { rig.cleanUp() }
        rig.factory.suspend = .unavailable(reason: "Filtered networking can't be saved.")
        rig.factory.add()
        _ = await rig.library.startAndWait(rig.id)
        #expect(await rig.library.suspendAndWait(rig.id) == .failed("Filtered networking can't be saved."))
        #expect(rig.controller.state == .running)
        await rig.controller.forceStop()
    }

    @Test
    func suspendingAStoppedMachineIsRefused() async throws {
        let rig = try ControllerRig()
        defer { rig.cleanUp() }
        #expect(await rig.library.suspendAndWait(rig.id) == .refused("“fake” is not running."))
    }

    @Test
    func theMenuPathsStillStartAndSuspend() async throws {
        let rig = try ControllerRig()
        defer { rig.cleanUp() }
        let guest = rig.factory.add()
        rig.library.startMachine(rig.id, interactive: false)
        try await guest.waitFor(.start)
        rig.library.suspendMachine(rig.id)
        try await guest.waitFor(.save)
        // Let the suspend finish before the scratch folder goes.
        await poll(every: .milliseconds(10), attempts: 500, clock: ContinuousClock()) { rig.controller.state == .stopped }
        #expect(rig.controller.state == .stopped)
    }
}
```

## Test

The tests above are the acceptance: a start returns `.done(.running)`; a second start is refused with the state; a failed
preparation returns `.failed` with its message; unknown, not-set-up, in-setup and unapproved-path machines are refused
without preparing anything and without a prompt; a suspend returns `.done(.stopped)` with the image saved; a suspend the
run cannot save returns `.failed` with the reason and the machine keeps running; the menu paths (`startMachine`,
`suspendMachine`) still reach the guest. `LaunchStartTests` and `MachineSuspendTests` keep passing unchanged.
