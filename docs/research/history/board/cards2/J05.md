on a thread of their own and never hold a thread Swift concurrency shares. The socket calls are the console's
(`ConsoleSocket`, internal to `GuestConsole` until now): this card makes the ones the control library needs `package` and
changes nothing else in that file. The client (one request, one answer) drives the protocol end to end over a socket pair in
the tests; the user check is injected, so a test can pretend the client is another user.

import Foundation
import GuestConsole
import Synchronization

/// What reading one request line came to.
enum LineRead: Equatable, Sendable {
    case line([UInt8])
    case tooLong
    case closed
    case timedOut
    case failed(Int32)
}

/// Blocking reads and writes of lines on a socket, bounded in size and time. They wait on the socket, so callers run them
/// on a thread of their own (`runOnThread`).
enum LineIO {
    /// Reads until `buffer` holds a whole line, the line is too long, the peer closes, or `timeoutMilliseconds` pass.
    static func readLine(from descriptor: Int32, into buffer: inout LineBuffer, readBytes: Int, timeoutMilliseconds: Int32) -> LineRead {
        let deadline = ContinuousClock.now + .milliseconds(Int(timeoutMilliseconds))
        var chunk = [UInt8](repeating: 0, count: readBytes)
        while true {
            switch buffer.next() {
            case .line(let bytes): return .line(bytes)
            case .tooLong: return .tooLong
            case .needMore: break
            }
            let left = ContinuousClock.now.duration(to: deadline) / .milliseconds(1)
            guard left > 0, ConsoleSocket.wait(descriptor, milliseconds: Int32(left.rounded(.up))) else { return .timedOut }
            switch ConsoleSocket.read(descriptor, into: &chunk) {
            case .transferred(let count): buffer.append(chunk[..<count])
            case .wouldBlock: continue
            case .endOfStream: return .closed
            case .failed(let code): return .failed(code)
            }
        }
    }

    /// Writes `data` whole; false when the peer is gone or did not read for `stallMilliseconds`.
    static func write(_ data: Data, to descriptor: Int32, stallMilliseconds: Int32) -> Bool {
        do {
            try ConsoleSocket.writeAll(descriptor, Array(data), stallMilliseconds: stallMilliseconds)
            return true
        } catch {
            return false
        }
    }
}

/// Runs blocking `work` on a thread of its own and returns its result, so a wait on a socket never holds one of the threads
/// Swift concurrency shares between tasks.
func runOnThread<T: Sendable>(_ work: @escaping @Sendable () -> T) async -> T {
    await withCheckedContinuation { continuation in
        let thread = Thread { continuation.resume(returning: work()) }
        thread.name = "ControlProtocol.io"
        thread.start()
    }
}

/// Runs `operation` and returns its result, or nil when `timeout` passes first, as `sleep` measures it. The operation is not
/// cancelled: what it started (a machine that boots) goes on, and its result is dropped.
func withDeadline<T: Sendable>(
    _ timeout: Duration, sleep: @escaping @Sendable (Duration) async throws -> Void, _ operation: @escaping @Sendable () async -> T
) async -> T? {
    await withCheckedContinuation { continuation in
        let first = FirstAnswer<T>(continuation)
        let timer = Task {
            guard (try? await sleep(timeout)) != nil else { return }
            first.resume(nil)
        }
        Task {
            let value = await operation()
            first.resume(value)
            timer.cancel()
        }
    }
}

/// Resumes a continuation with the first answer it is given and ignores the others.
private final class FirstAnswer<T: Sendable>: Sendable {
    private let continuation: Mutex<CheckedContinuation<T?, Never>?>

    init(_ continuation: CheckedContinuation<T?, Never>) {
        self.continuation = Mutex(Optional(continuation))
    }

    func resume(_ value: T?) {
        let waiting = continuation.withLock { stored -> CheckedContinuation<T?, Never>? in
            defer { stored = nil }
            return stored
        }
        waiting?.resume(returning: value)
    }
}

import Foundation
import GuestConsole

/// Who is on the other end of a local socket.
public enum ControlPeer {
    /// The user id the kernel reports for the process on the other end (`getpeereid`), or nil when it cannot say.
    public static func user(of descriptor: Int32) -> UInt32? {
        var user: uid_t = 0
        var group: gid_t = 0
        // `getpeereid` writes one uid and one gid.
        guard unsafe getpeereid(descriptor, &user, &group) == 0 else { return nil }
        return user
    }

    /// The user this process runs as.
    public static var currentUser: UInt32 { geteuid() }
}

/// Serves one connected client (ADR-0056): checks that it runs as the allowed user, then reads requests line by line within
/// the limits and writes each answer. The descriptor is closed when it returns.
public struct ControlConnection: Sendable {
    let dispatcher: ControlDispatcher
    let limits: ControlLimits
    let peerUser: @Sendable (Int32) -> UInt32?
    let allowedUser: UInt32
    let sleep: @Sendable (Duration) async throws -> Void

    /// `peerUser` and `allowedUser` say who the client is and who it must be: by default the kernel's answer for the socket
    /// and this process's user. `sleep` measures a request's time limit. Tests replace them.
    public init(
        dispatcher: ControlDispatcher, limits: ControlLimits = ControlLimits(),
        peerUser: @escaping @Sendable (Int32) -> UInt32? = ControlPeer.user(of:), allowedUser: UInt32 = ControlPeer.currentUser,
        sleep: @escaping @Sendable (Duration) async throws -> Void = { try await Task.sleep(for: $0) }
    ) {
        self.dispatcher = dispatcher
        self.limits = limits
        self.peerUser = peerUser
        self.allowedUser = allowedUser
        self.sleep = sleep
    }

    /// Serves `descriptor` until the client closes, sends too much, stays silent past the read limit or has sent
    /// `limits.requestsPerConnection` requests; then closes it. `isOpen` is asked once a request has been read: when it says no
    /// (the listener was stopped) the connection closes without answering, so turning the service off cuts open connections off.
    public func serve(_ descriptor: Int32, while isOpen: @Sendable () -> Bool = { true }) async {
        defer { ConsoleSocket.close(descriptor) }
        let peer = peerUser(descriptor)
        guard peer == allowedUser else {
            dispatcher.report(ControlDenial(reason: .otherUser, peerUser: peer))
            return
        }
        let limits = self.limits
        var buffer = LineBuffer(limit: limits.requestBytes)
        for _ in 0..<limits.requestsPerConnection {
            let (read, rest) = await runOnThread { [buffer] in
                var pending = buffer
                let read = LineIO.readLine(
                    from: descriptor, into: &pending, readBytes: limits.readBytes, timeoutMilliseconds: limits.readTimeoutMilliseconds)
                return (read, pending)
            }
            buffer = rest
            guard isOpen(), let outcome = await reply(to: read) else { return }
            if let line = outcome.line {
                let stall = limits.writeStallMilliseconds
                guard await runOnThread({ LineIO.write(line, to: descriptor, stallMilliseconds: stall) }) else { return }
            }
            if outcome.closes { return }
        }
    }

    /// What to send back for one read, and whether the connection ends after it; nil ends it without a word.
    private func reply(to read: LineRead) async -> (line: Data?, closes: Bool)? {
        switch read {
        case .line(let bytes):
            return (await answer(Data(bytes)), false)
        case .tooLong:
            let error = RPCError.invalidRequest.with("the request is longer than \(limits.requestBytes) bytes")
            return (RPCResponse(id: .null, error: error).encoded(), true)
        case .closed, .timedOut, .failed:
            return nil
        }
    }

    /// The answer line to one request line; nil for a notification, which is neither run nor answered.
    func answer(_ line: Data) async -> Data? {
        switch RPCMessage.parse(line) {
        case .invalid(let response):
            return response.encoded()
        case .notification:
            return nil
        case .request(let request):
            let dispatcher = self.dispatcher
            let response = await withDeadline(limits.requestTimeout, sleep: sleep) { await dispatcher.answer(request) }
            return (response ?? RPCResponse(id: request.id, error: .timedOut)).encoded()
        }
    }
}

```swift
public import Foundation
import GuestConsole

/// Why a client call failed.
public enum ControlClientError: Error, Equatable, Sendable {
    case closed
    case timedOut
    case answerTooLong
    case malformedAnswer
    case system(Int32)
}

/// An answer as the client reads it.
public struct ControlAnswer: Equatable, Sendable {
    public let id: Int?
    public let errorCode: Int?
    public let errorMessage: String?
    /// The `result` member as JSON with sorted keys; nil for an error.
    public let result: Data?

    init(line: Data) throws {
        guard let object = try? JSONSerialization.jsonObject(with: line) as? [String: Any], object["jsonrpc"] as? String == "2.0"
        else { throw ControlClientError.malformedAnswer }
        id = object["id"] as? Int
        if let error = object["error"] as? [String: Any] {
            errorCode = error["code"] as? Int
            errorMessage = error["message"] as? String
            result = nil
        } else if let value = object["result"] {
            errorCode = nil
            errorMessage = nil
            result = try JSONSerialization.data(withJSONObject: value, options: [.fragmentsAllowed, .sortedKeys])
        } else {
            throw ControlClientError.malformedAnswer
        }
    }
}

/// A minimal blocking client of the control service: one request, then its answer. The tests drive the protocol end to end
/// with it; a command-line client can build on it.
public struct ControlClient: Sendable {
    public let descriptor: Int32
    /// How long a call waits for the service to take the request and to answer: longer than the service's own 120 s limit
    /// for an answer, so a slow start is answered rather than abandoned.
    public var timeoutMilliseconds: Int32 = 125_000

    public init(descriptor: Int32) { self.descriptor = descriptor }

    /// Connects to the service's socket.
    public static func connect(to path: String) throws -> Self {
        Self(descriptor: try ConsoleSocket.connect(to: path))
    }

    /// A connected pair of local sockets: a client, and the descriptor to give `ControlConnection.serve(_:)`.
    public static func connectedPair() throws -> (client: Self, server: Int32) {
        guard let pair = ConsoleSocket.streamPair(bufferBytes: 64 * 1_024) else { throw ControlClientError.system(errno) }
        return (Self(descriptor: pair.first), pair.second)
    }

    /// Sends `line` as it is, then a newline.
    public func send(_ line: Data) throws {
        try ConsoleSocket.writeAll(descriptor, Array(line) + [UInt8(ascii: "\n")], stallMilliseconds: timeoutMilliseconds)
    }

    /// The next answer; nil when the service closed the connection.
    public func receive() throws -> ControlAnswer? {
        var buffer = LineBuffer(limit: 1 << 20)
        switch LineIO.readLine(from: descriptor, into: &buffer, readBytes: 4_096, timeoutMilliseconds: timeoutMilliseconds) {
        case .line(let bytes): return try ControlAnswer(line: Data(bytes))
        case .closed: return nil
        case .timedOut: throw ControlClientError.timedOut
        case .tooLong: throw ControlClientError.answerTooLong
        case .failed(let code): throw ControlClientError.system(code)
        }
    }

    /// Sends a request and returns its answer.
    public func call(_ method: String, id: Int = 1, params: [String: RPCValue] = [:]) throws -> ControlAnswer {
        var members: [String: Any] = [:]
        for (key, value) in params {
            switch value {
            case .string(let text): members[key] = text
            case .bool(let flag): members[key] = flag
            }
        }
        let request: [String: Any] = ["jsonrpc": "2.0", "id": id, "method": method, "params": members]
        try send(try JSONSerialization.data(withJSONObject: request, options: [.sortedKeys]))
        guard let answer = try receive() else { throw ControlClientError.closed }
        return answer
    }

    public func close() { ConsoleSocket.close(descriptor) }
}
```
