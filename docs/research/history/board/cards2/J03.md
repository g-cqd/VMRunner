```swift
import CryptoKit
public import Foundation

/// What a token allows. Each method needs one capability; later steps add screenshots, input and exec.
public enum ControlCapability: String, Codable, CaseIterable, Hashable, Sendable {
    case list, state, start, stop

    /// How Settings names it.
    public var label: String {
        switch self {
        case .list: "List machines"
        case .state: "Read state"
        case .start: "Start"
        case .stop: "Stop"
        }
    }
}

/// A client's token as the store keeps it: never the secret, only its digest.
public struct ControlToken: Codable, Equatable, Identifiable, Sendable {
    public let id: UUID
    public let name: String
    public let capabilities: Set<ControlCapability>
    /// The machines the token may act on; nil for every machine whose agent access is on.
    public let machines: [UUID]?
    public let created: Date
    /// SHA-256 of the secret, in lowercase hex.
    let digest: String
}

/// Making, hashing and comparing token secrets.
public enum ControlSecret {
    /// Every secret starts with it, so a secret that leaks into a file or a log is easy to recognise.
    public static let prefix = "vmrc_"

    /// A new secret: 256 bits from the system's cryptographically secure generator, as 64 hex digits after `prefix`.
    public static func make() -> String {
        var generator = SystemRandomNumberGenerator()
        let bytes = (0..<32).map { _ in UInt8.random(in: .min ... .max, using: &generator) }
        return prefix + hex(bytes)
    }

    /// The SHA-256 digest of `secret`, in lowercase hex: what the store keeps.
    public static func digest(of secret: String) -> String {
        hex(Array(SHA256.hash(data: Data(secret.utf8))))
    }

    /// Whether two byte strings are equal, in a time that depends on their length only: every byte is compared, so the time
    /// taken does not tell how much of a guess was right.
    public static func constantTimeEqual(_ lhs: [UInt8], _ rhs: [UInt8]) -> Bool {
        guard lhs.count == rhs.count else { return false }
        var difference: UInt8 = 0
        for index in lhs.indices { difference |= lhs[index] ^ rhs[index] }
        return difference == 0
    }

    static func hex(_ bytes: [UInt8]) -> String {
        bytes.map { byte in byte < 16 ? "0" + String(byte, radix: 16) : String(byte, radix: 16) }.joined()
    }
}
```

```swift
public import Foundation
import Synchronization

/// Why a token could not be made, revoked or read.
public enum ControlTokenError: LocalizedError, Equatable, Sendable {
    case emptyName
    case nameTooLong(limit: Int)
    case noCapability
    case tooMany(limit: Int)
    case unknownToken
    case unreadable(path: String)

    public var errorDescription: String? {
        switch self {
        case .emptyName: "A token needs a name."
        case .nameTooLong(let limit): "A token's name has at most \(limit) characters."
        case .noCapability: "A token needs at least one capability."
        case .tooMany(let limit): "There are already \(limit) tokens; revoke one first."
        case .unknownToken: "There is no such token."
        case .unreadable(let path): "The token list at \(path) cannot be read."
        }
    }
}

/// The tokens, kept as digests in one JSON file (mode 0600) in a folder the caller keeps private. A secret is returned once,
/// when its token is made, and never stored.
public final class ControlTokenStore: Sendable {
    public static let maximumTokens = 32
    public static let maximumNameLength = 64

    public let file: URL
    private let tokens: Mutex<[ControlToken]>

    /// Reads `file`. A missing file is an empty list; a file that does not decode is refused, not replaced.
    public init(file: URL) throws {
        self.file = file
        var stored: [ControlToken] = []
        if FileManager.default.fileExists(atPath: file.path) {
            do {
                stored = try JSONDecoder().decode([ControlToken].self, from: Data(contentsOf: file))
            } catch {
                throw ControlTokenError.unreadable(path: file.path)
            }
        }
        tokens = Mutex(stored)
    }

    /// Every token, oldest first.
    public var all: [ControlToken] { tokens.withLock { $0 } }

    /// Makes a token and returns it with its secret, which is not kept: show it once.
    public func create(
        name: String, capabilities: Set<ControlCapability>, machines: [UUID]? = nil, now: Date = Date()
    ) throws -> (token: ControlToken, secret: String) {
        let trimmed = name.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !trimmed.isEmpty else { throw ControlTokenError.emptyName }
        guard trimmed.count <= Self.maximumNameLength else { throw ControlTokenError.nameTooLong(limit: Self.maximumNameLength) }
        guard !capabilities.isEmpty else { throw ControlTokenError.noCapability }
        let secret = ControlSecret.make()
        let token = ControlToken(
            id: UUID(), name: trimmed, capabilities: capabilities, machines: machines, created: now,
            digest: ControlSecret.digest(of: secret))
        try tokens.withLock { list in
            guard list.count < Self.maximumTokens else { throw ControlTokenError.tooMany(limit: Self.maximumTokens) }
            try write(list + [token])
            list.append(token)
        }
        return (token, secret)
    }

    /// Removes a token: its secret fails from the next request on.
    public func revoke(_ id: UUID) throws {
        try tokens.withLock { list in
            guard list.contains(where: { $0.id == id }) else { throw ControlTokenError.unknownToken }
            let kept = list.filter { $0.id != id }
            try write(kept)
            list = kept
        }
    }

    /// The token whose secret this is, or nil. The presented secret's digest is compared with every stored digest, each in
    /// constant time, whichever matches, so the time taken does not tell which tokens exist.
    public func authenticate(_ secret: String) -> ControlToken? {
        let presented = Array(ControlSecret.digest(of: secret).utf8)
        return tokens.withLock { list in
            var found: ControlToken?
            for token in list where ControlSecret.constantTimeEqual(presented, Array(token.digest.utf8)) {
                found = token
            }
            return found
        }
    }

    private func write(_ list: [ControlToken]) throws {
        let encoder = JSONEncoder()
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
        try encoder.encode(list).write(to: file, options: .atomic)
        try FileManager.default.setAttributes([.posixPermissions: 0o600], ofItemAtPath: file.path)
    }
}
```

```swift
import Foundation

/// A private (0700) folder for one test, removed by `cleanUp()`. Never the app's data folder. A socket test passes `/tmp` as
/// `root`: a local socket path holds at most 103 bytes, and the usual temporary folder is longer than that.
struct ScratchFolder: Sendable {
    let url: URL

    init(in root: URL = FileManager.default.temporaryDirectory) throws {
        url = root.appending(path: "control-\(UUID().uuidString)", directoryHint: .isDirectory)
        try FileManager.default.createDirectory(at: url, withIntermediateDirectories: true, attributes: [.posixPermissions: 0o700])
    }

    func cleanUp() { try? FileManager.default.removeItem(at: url) }
}
```

```swift
import Foundation
import Testing

@testable import ControlProtocol

/// Secrets are 256 random bits; only their SHA-256 digest is kept; digests are compared in constant time.
struct ControlSecretTests {
    @Test
    func aSecretIs256RandomBitsInHexAfterThePrefix() {
        let secret = ControlSecret.make()
        #expect(secret.hasPrefix("vmrc_"))
        let digits = secret.dropFirst(ControlSecret.prefix.count)
        #expect(digits.count == 64)
        #expect(digits.allSatisfy { "0123456789abcdef".contains($0) })
        #expect(ControlSecret.make() != secret)
    }

    @Test
    func theDigestIsSHA256InLowercaseHex() {
        #expect(ControlSecret.digest(of: "abc") == "ba7816bf8f01cfea414140de5dae2223b00361a396177a9cb410ff61f20015ad")
        #expect(ControlSecret.digest(of: "") == "e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855")
    }

    @Test
    func theConstantTimeComparisonSaysEqualOnlyForEqualBytes() {
        let bytes: [UInt8] = [1, 2, 3, 4]
        #expect(ControlSecret.constantTimeEqual(bytes, [1, 2, 3, 4]))
        #expect(!ControlSecret.constantTimeEqual(bytes, [0, 2, 3, 4]), "the first byte differs")
        #expect(!ControlSecret.constantTimeEqual(bytes, [1, 2, 3, 5]), "the last byte differs")
        #expect(!ControlSecret.constantTimeEqual(bytes, [1, 2, 3]), "a shorter string")
        #expect(ControlSecret.constantTimeEqual([], []))
    }
}

/// The store: a token shown once, kept as a digest in a private file, found by its secret, gone once revoked.
struct ControlTokenStoreTests {
    @Test
    func aTokenIsShownOnceAndOnlyItsDigestIsStoredInAPrivateFile() throws {
        let folder = try ScratchFolder()
        defer { folder.cleanUp() }
        let file = folder.url.appending(path: "tokens.json")
        let store = try ControlTokenStore(file: file)
        let made = try store.create(name: "  agent  ", capabilities: [.list, .state])
        #expect(made.token.name == "agent")
        #expect(store.authenticate(made.secret) == made.token)
        let text = try String(contentsOf: file, encoding: .utf8)
        #expect(!text.contains(made.secret), "the secret is never written")
        #expect(text.contains(ControlSecret.digest(of: made.secret)))
        let mode = try FileManager.default.attributesOfItem(atPath: file.path)[.posixPermissions] as? Int
        #expect(mode == 0o600)
    }

    @Test
    func aWrongSecretMatchesNothing() throws {
        let folder = try ScratchFolder()
        defer { folder.cleanUp() }
        let store = try ControlTokenStore(file: folder.url.appending(path: "tokens.json"))
        let made = try store.create(name: "agent", capabilities: [.list])
        #expect(store.authenticate(ControlSecret.make()) == nil)
        #expect(store.authenticate("") == nil)
        #expect(store.authenticate(made.secret + "0") == nil)
        #expect(store.authenticate(String(made.secret.dropLast())) == nil)
    }

    @Test
    func tokensSurviveAReloadAndARevokedOneFailsAtOnceAndAfterAReload() throws {
        let folder = try ScratchFolder()
        defer { folder.cleanUp() }
        let file = folder.url.appending(path: "tokens.json")
        let store = try ControlTokenStore(file: file)
        let first = try store.create(name: "one", capabilities: [.list])
        let second = try store.create(name: "two", capabilities: [.start], machines: [UUID()])
        let reloaded = try ControlTokenStore(file: file)
        #expect(reloaded.all == store.all)
        #expect(reloaded.authenticate(second.secret)?.machines == second.token.machines)

        try store.revoke(first.token.id)
        #expect(store.authenticate(first.secret) == nil)
        #expect(store.authenticate(second.secret) == second.token)
        #expect(try ControlTokenStore(file: file).authenticate(first.secret) == nil)
        #expect(throws: ControlTokenError.unknownToken) { try store.revoke(first.token.id) }
    }

    @Test
    func badNamesAnEmptyCapabilitySetAndAFullListAreRefused() throws {
        let folder = try ScratchFolder()
        defer { folder.cleanUp() }
        let store = try ControlTokenStore(file: folder.url.appending(path: "tokens.json"))
        #expect(throws: ControlTokenError.emptyName) { _ = try store.create(name: "  ", capabilities: [.list]) }
        #expect(throws: ControlTokenError.nameTooLong(limit: 64)) {
            _ = try store.create(name: String(repeating: "n", count: 65), capabilities: [.list])
        }
        #expect(throws: ControlTokenError.noCapability) { _ = try store.create(name: "n", capabilities: []) }
        for index in 0..<ControlTokenStore.maximumTokens { _ = try store.create(name: "t\(index)", capabilities: [.list]) }
        #expect(throws: ControlTokenError.tooMany(limit: 32)) { _ = try store.create(name: "one more", capabilities: [.list]) }
        #expect(store.all.count == 32)
    }

    @Test
    func aFileThatDoesNotDecodeIsRefusedAndKept() throws {
        let folder = try ScratchFolder()
        defer { folder.cleanUp() }
        let file = folder.url.appending(path: "tokens.json")
        try Data("not json".utf8).write(to: file)
        #expect(throws: ControlTokenError.unreadable(path: file.path)) { _ = try ControlTokenStore(file: file) }
        #expect(try String(contentsOf: file, encoding: .utf8) == "not json")
    }
}
```
