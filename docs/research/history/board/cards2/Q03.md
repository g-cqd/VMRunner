/// The bounds the device puts on what a guest may ask of it. The guest is untrusted: every count, size and length it sends is
/// checked against these before anything is allocated or copied.
public struct GPULimits: Equatable, Sendable {
    /// How many 2D resources may exist at once; one more is refused with an out-of-memory answer.
    public var maxResources = 256
    /// The largest host copy of one resource, in bytes (width × height × 4): 64 MiB, a 4096 × 4096 surface.
    public var maxResourceBytes = 64 << 20
    /// The host copies of all resources together, in bytes.
    public var maxTotalResourceBytes = 256 << 20
    /// The most guest memory pieces one resource's backing may list: a 64 MiB surface in 4 KiB pages.
    public var maxBackingEntries = 16_384
    /// The size of the one display the device reports.
    public var displayWidth: UInt32 = 1280
    public var displayHeight: UInt32 = 800

    public init() {}

    /// The longest request the device reads: an attach-backing header followed by the most entries it accepts.
    public var maxCommandBytes: Int { GPUWireSize.attachBacking + GPUWireSize.memoryEntry * maxBackingEntries }
}

/// A cursor over a copy of guest bytes that reads little-endian integers and returns `nil` instead of reading past the end.
struct ByteReader {
    private let bytes: [UInt8]
    private(set) var offset = 0

    init(_ bytes: [UInt8]) {
        self.bytes = bytes
    }

    /// The bytes not read yet.
    var remaining: Int { bytes.count - offset }

    mutating func u8() -> UInt8? {
        guard remaining >= 1 else { return nil }
        defer { offset += 1 }
        return bytes[offset]
    }

    mutating func u32() -> UInt32? {
        guard remaining >= 4 else { return nil }
        var value: UInt32 = 0
        for index in 0..<4 { value |= UInt32(bytes[offset + index]) << (8 * index) }
        offset += 4
        return value
    }

    mutating func u64() -> UInt64? {
        guard let low = u32(), let high = u32() else { return nil }
        return UInt64(high) << 32 | UInt64(low)
    }

    /// Moves past `count` bytes; `false`, and no move, when fewer remain.
    mutating func skip(_ count: Int) -> Bool {
        guard count >= 0, remaining >= count else { return false }
        offset += count
        return true
    }
}

/// Appends little-endian integers to a byte array, for the device's answers.
struct ByteWriter {
    private(set) var bytes: [UInt8] = []

    init(capacity: Int) {
        bytes.reserveCapacity(capacity)
    }

    mutating func u8(_ value: UInt8) {
        bytes.append(value)
    }

    mutating func u32(_ value: UInt32) {
        for index in 0..<4 { bytes.append(UInt8(truncatingIfNeeded: value >> (8 * index))) }
    }

    mutating func u64(_ value: UInt64) {
        u32(UInt32(truncatingIfNeeded: value))
        u32(UInt32(truncatingIfNeeded: value >> 32))
    }

    mutating func zeros(_ count: Int) {
        bytes.append(contentsOf: repeatElement(0, count: count))
    }
}

    public static let getDisplayInfo = 24
    public static let resourceCreate2D = 40
    public static let resourceUnref = 32
    public static let setScanout = 48
    public static let resourceFlush = 48
    public static let transferToHost2D = 56

    public static let memoryEntry = 16
    public static let detachBacking = 32
    public static let getEDID = 32
    public static let cursor = 56
    /// A display-information answer: the header and 16 modes of 24 bytes.
    public static let displayInfo = 408
    /// The number of modes a display-information answer always carries.
    public static let displayModes = 16
}

public enum GPUCommandType: UInt32, Sendable, CaseIterable {
    case getDisplayInfo = 0x0100
    case resourceCreate2D = 0x0101
    case resourceUnref = 0x0102
    case setScanout = 0x0103
    case resourceFlush = 0x0104
    case transferToHost2D = 0x0105
    case resourceAttachBacking = 0x0106
    case resourceDetachBacking = 0x0107

    case updateCursor = 0x0300
    case moveCursor = 0x0301

        self.flags = flags
        self.fenceID = fenceID
        self.contextID = contextID
        self.ringIndex = ringIndex
    }

/// A rectangle in a resource's or a scanout's pixels.

    public var x: UInt32
    public var y: UInt32
    public var width: UInt32
    public var height: UInt32

    public init(x: UInt32, y: UInt32, width: UInt32, height: UInt32) {
        self.x = x
        self.y = y
        self.width = width
        self.height = height
    }

    /// Whether the rectangle lies inside a surface of `width` × `height`, computed without overflow.
    public func fits(width surfaceWidth: UInt32, height surfaceHeight: UInt32) -> Bool {
        UInt64(x) + UInt64(width) <= UInt64(surfaceWidth) && UInt64(y) + UInt64(height) <= UInt64(surfaceHeight)

    public func intersects(_ other: Self) -> Bool {

    }
}

/// A decoded 2D request.
public enum GPUCommand: Equatable, Sendable {
    case getDisplayInfo
    case resourceCreate2D(resourceID: UInt32, format: UInt32, width: UInt32, height: UInt32)
    case resourceUnref(resourceID: UInt32)
    case setScanout(rect: GPURect, scanoutID: UInt32, resourceID: UInt32)
    case resourceFlush(rect: GPURect, resourceID: UInt32)
    case transferToHost2D(rect: GPURect, offset: UInt64, resourceID: UInt32)
    case resourceAttachBacking(resourceID: UInt32, entries: [GPUMemoryEntry])
    case resourceDetachBacking(resourceID: UInt32)
    case getEDID(scanoutID: UInt32)
    case updateCursor(GPUCursor)
    case moveCursor(GPUCursor)

    /// The resource the request names, for the command log; 0 when it names none.
    public var resourceID: UInt32 {
        switch self {
        case .getDisplayInfo, .getEDID: 0
        case .resourceCreate2D(let id, _, _, _), .resourceUnref(let id), .setScanout(_, _, let id), .resourceFlush(_, let id),
            .transferToHost2D(_, _, let id), .resourceAttachBacking(let id, _), .resourceDetachBacking(let id):
            id
        case .updateCursor(let cursor), .moveCursor(let cursor): cursor.resourceID
        }
    }
}

/// Why a request could not be decoded, and the answer it gets: a request too short for its type, or of a type the device does
/// not implement, is answered as unsupported (`errUnspec`); a request whose values break a limit as an invalid parameter.
public enum GPUDecodeFailure: Error, Equatable, Sendable {
    case shorterThanHeader(length: Int)
    case unsupported(type: UInt32)
    case shorterThanCommand(type: UInt32, needed: Int, length: Int)
    case longerThanLimit(length: Int)
    case tooManyEntries(count: UInt32)
    case entriesMissing(count: UInt32, length: Int)
    case invalidEntry(index: Int)

    public var responseType: GPUResponseType {
        switch self {
        case .shorterThanHeader, .unsupported, .shorterThanCommand: .errUnspec
        case .longerThanLimit, .tooManyEntries, .entriesMissing, .invalidEntry: .errInvalidParameter
        }
    }
}

/// A request as read from the guest: its header (zeroed when the request is too short to carry one) and its command or the
/// reason it has none.
public struct GPURequest: Equatable, Sendable {
    public let header: GPUControlHeader
    public let command: Result<GPUCommand, GPUDecodeFailure>
}

/// Decodes one request from the bytes the device copied out of guest memory. The copy is made once, before decoding, so the
/// guest cannot change a value between its check and its use; every length, count and offset is checked here or by the
/// resource store, and nothing is allocated for the guest before its size has been checked against `GPULimits`.
public enum GPUDecoder {
    public static func decode(_ bytes: [UInt8], limits: GPULimits) -> GPURequest {
        var reader = ByteReader(bytes)
        guard let header = readHeader(&reader) else {
            return GPURequest(header: .unreadable, command: .failure(.shorterThanHeader(length: bytes.count)))
        }
        guard bytes.count <= limits.maxCommandBytes else {
            return GPURequest(header: header, command: .failure(.longerThanLimit(length: bytes.count)))
        }
        guard let type = GPUCommandType(rawValue: header.type) else {
            return GPURequest(header: header, command: .failure(.unsupported(type: header.type)))
        }
        let needed = fixedSize(of: type)
        guard bytes.count >= needed else {
            let failure = GPUDecodeFailure.shorterThanCommand(type: header.type, needed: needed, length: bytes.count)
            return GPURequest(header: header, command: .failure(failure))
        }
        return GPURequest(header: header, command: body(of: type, from: &reader, limits: limits))
    }

    private static func readHeader(_ reader: inout ByteReader) -> GPUControlHeader? {
        guard let type = reader.u32(), let flags = reader.u32(), let fence = reader.u64(), let context = reader.u32(),
            let ring = reader.u8(), reader.skip(3)
        else { return nil }
        return GPUControlHeader(type: type, flags: flags, fenceID: fence, contextID: context, ringIndex: ring)
    }

    /// The size of the fixed part of a request of `type`, header included.
    static func fixedSize(of type: GPUCommandType) -> Int {
        switch type {
        case .getDisplayInfo: GPUWireSize.getDisplayInfo
        case .resourceCreate2D: GPUWireSize.resourceCreate2D
        case .resourceUnref: GPUWireSize.resourceUnref
        case .setScanout: GPUWireSize.setScanout
        case .resourceFlush: GPUWireSize.resourceFlush
        case .transferToHost2D: GPUWireSize.transferToHost2D
        case .resourceAttachBacking: GPUWireSize.attachBacking
        case .resourceDetachBacking: GPUWireSize.detachBacking
        case .getEDID: GPUWireSize.getEDID
        case .updateCursor, .moveCursor: GPUWireSize.cursor
        }
    }

    /// The command after the header. The caller checked that the fixed part is present, so the reads of the fixed part
    /// cannot fail; they still fall back to an answer rather than trap.
    private static func body(
        of type: GPUCommandType, from reader: inout ByteReader, limits: GPULimits
    ) -> Result<GPUCommand, GPUDecodeFailure> {
        let short = GPUDecodeFailure.shorterThanCommand(type: type.rawValue, needed: fixedSize(of: type), length: 0)
        let command: GPUCommand?
        switch type {
        case .getDisplayInfo: command = .getDisplayInfo
        case .resourceCreate2D: command = resourceCreate2D(&reader)
        case .resourceUnref: command = reader.u32().map { .resourceUnref(resourceID: $0) }
        case .setScanout: command = setScanout(&reader)
        case .resourceFlush: command = resourceFlush(&reader)
        case .transferToHost2D: command = transferToHost2D(&reader)
        case .resourceAttachBacking: return attachBacking(&reader, limits: limits)
        case .resourceDetachBacking: command = reader.u32().map { .resourceDetachBacking(resourceID: $0) }
        case .getEDID: command = reader.u32().map { .getEDID(scanoutID: $0) }
        case .updateCursor: command = cursor(&reader).map { .updateCursor($0) }
        case .moveCursor: command = cursor(&reader).map { .moveCursor($0) }
        }
        return command.map { .success($0) } ?? .failure(short)
    }

    private static func rect(_ reader: inout ByteReader) -> GPURect? {
        guard let x = reader.u32(), let y = reader.u32(), let width = reader.u32(), let height = reader.u32() else { return nil }
        return GPURect(x: x, y: y, width: width, height: height)
    }

    private static func resourceCreate2D(_ reader: inout ByteReader) -> GPUCommand? {
        guard let id = reader.u32(), let format = reader.u32(), let width = reader.u32(), let height = reader.u32() else { return nil }
        return .resourceCreate2D(resourceID: id, format: format, width: width, height: height)
    }

    private static func setScanout(_ reader: inout ByteReader) -> GPUCommand? {
        guard let rect = rect(&reader), let scanout = reader.u32(), let id = reader.u32() else { return nil }
        return .setScanout(rect: rect, scanoutID: scanout, resourceID: id)
    }

    private static func resourceFlush(_ reader: inout ByteReader) -> GPUCommand? {
        guard let rect = rect(&reader), let id = reader.u32() else { return nil }
        return .resourceFlush(rect: rect, resourceID: id)
    }

    private static func transferToHost2D(_ reader: inout ByteReader) -> GPUCommand? {
        guard let rect = rect(&reader), let offset = reader.u64(), let id = reader.u32() else { return nil }
        return .transferToHost2D(rect: rect, offset: offset, resourceID: id)
    }

    private static func cursor(_ reader: inout ByteReader) -> GPUCursor? {
        guard let scanout = reader.u32(), let x = reader.u32(), let y = reader.u32(), reader.skip(4), let id = reader.u32(),
            let hotX = reader.u32(), let hotY = reader.u32()
        else { return nil }
        return GPUCursor(scanoutID: scanout, x: x, y: y, resourceID: id, hotX: hotX, hotY: hotY)
    }

    /// The entry count is checked against the limit and against the bytes actually present before the entries array is
    /// made, so a guest cannot make the device allocate for entries it did not send.
    private static func attachBacking(_ reader: inout ByteReader, limits: GPULimits) -> Result<GPUCommand, GPUDecodeFailure> {
        guard let id = reader.u32(), let count = reader.u32() else {
            let type = GPUCommandType.resourceAttachBacking.rawValue
            return .failure(.shorterThanCommand(type: type, needed: GPUWireSize.attachBacking, length: 0))
        }
        guard count <= limits.maxBackingEntries else { return .failure(.tooManyEntries(count: count)) }
        let present = reader.remaining / GPUWireSize.memoryEntry
        guard Int(count) <= present else { return .failure(.entriesMissing(count: count, length: reader.remaining)) }
        var entries: [GPUMemoryEntry] = []
        entries.reserveCapacity(Int(count))
        for index in 0..<Int(count) {
            guard let address = reader.u64(), let length = reader.u32(), reader.skip(4) else {
                return .failure(.entriesMissing(count: count, length: reader.remaining))
            }
            guard length > 0, address.addingReportingOverflow(UInt64(length)).overflow == false else {
                return .failure(.invalidEntry(index: index))
            }
            entries.append(GPUMemoryEntry(address: address, length: length))
        }
        return .success(.resourceAttachBacking(resourceID: id, entries: entries))
    }
}

/// One mode of a display-information answer.
public struct GPUDisplayMode: Equatable, Sendable {
    public var rect: GPURect
    public var enabled: Bool

    public init(rect: GPURect, enabled: Bool) {
        self.rect = rect
        self.enabled = enabled
    }
}

/// An answer the device writes back into the request's write buffers.
public enum GPUResponse: Equatable, Sendable {
    case noData
    /// Up to 16 modes; the answer always carries 16, the missing ones zeroed and disabled.
    case displayInfo([GPUDisplayMode])
    case error(GPUResponseType)

    private static let disabledMode = GPUDisplayMode(rect: GPURect(x: 0, y: 0, width: 0, height: 0), enabled: false)

    public var type: GPUResponseType {
        switch self {
        case .noData: .okNoData
        case .displayInfo: .okDisplayInfo
        case .error(let type): type
        }
    }

    /// The bytes of the answer to a request with `request`'s header. The fence flag, the fence id and the context are echoed
    /// when the request asked for a fence, and the ring index when it named one. The device handles a request completely
    /// before it answers, so an answer that carries a fence is written only once the request's effect is done.
    public func encoded(answering request: GPUControlHeader) -> [UInt8] {
        let fenced = request.flags & GPUControlHeader.fenceFlag != 0
        let flags = request.flags & (GPUControlHeader.fenceFlag | GPUControlHeader.ringIndexFlag)
        var writer = ByteWriter(capacity: GPUWireSize.displayInfo)
        writer.u32(type.rawValue)
        writer.u32(flags)
        writer.u64(fenced ? request.fenceID : 0)
        writer.u32(fenced ? request.contextID : 0)
        writer.u8(flags & GPUControlHeader.ringIndexFlag != 0 ? request.ringIndex : 0)
        writer.zeros(3)
        if case .displayInfo(let modes) = self {
            for index in 0..<GPUWireSize.displayModes {
                let mode = index < modes.count ? modes[index] : Self.disabledMode
                writer.u32(mode.rect.x)
                writer.u32(mode.rect.y)
                writer.u32(mode.rect.width)
                writer.u32(mode.rect.height)
                writer.u32(mode.enabled ? 1 : 0)
                writer.u32(0)
            }
        }
        return writer.bytes
    }
}

/// A small deterministic generator (SplitMix64), so a random test fails the same way on every run.
struct SeededGenerator: RandomNumberGenerator {
    private var state: UInt64

    init(seed: UInt64) {
        state = seed
    }

    mutating func next() -> UInt64 {
        state &+= 0x9E37_79B9_7F4A_7C15
        var value = state
        value = (value ^ (value >> 30)) &* 0xBF58_476D_1CE4_E5B9
        value = (value ^ (value >> 27)) &* 0x94D0_49BB_1331_11EB
        return value ^ (value >> 31)
    }
}

/// The bytes of a request: a header, then each field as a little-endian 32-bit word, then `tail`.
func request(
    _ type: UInt32, flags: UInt32 = 0, fence: UInt64 = 0, context: UInt32 = 0, ring: UInt8 = 0, words: [UInt32] = [],
    tail: [UInt8] = []
) -> [UInt8] {
    var writer = ByteWriter(capacity: 64)
    writer.u32(type)
    writer.u32(flags)
    writer.u64(fence)
    writer.u32(context)
    writer.u8(ring)
    writer.zeros(3)
    for word in words { writer.u32(word) }
    return writer.bytes + tail
}

func request(_ type: GPUCommandType, words: [UInt32] = [], tail: [UInt8] = []) -> [UInt8] {
    request(type.rawValue, words: words, tail: tail)
}

/// Memory entries as an attach-backing request carries them.
func entryBytes(_ entries: [(address: UInt64, length: UInt32)]) -> [UInt8] {
    var writer = ByteWriter(capacity: entries.count * 16)
    for entry in entries {
        writer.u64(entry.address)
        writer.u32(entry.length)
        writer.u32(0)
    }
    return writer.bytes
}

/// A transfer request: rectangle, 64-bit offset, resource.
func transfer(_ rect: GPURect, offset: UInt64, resource: UInt32) -> [UInt8] {
    let words = [
        rect.x, rect.y, rect.width, rect.height, UInt32(truncatingIfNeeded: offset), UInt32(truncatingIfNeeded: offset >> 32), resource, 0,
    ]
    return request(.transferToHost2D, words: words)
}

/// Reads the little-endian word at `offset` of an answer.
func word(_ bytes: [UInt8], at offset: Int) -> UInt32 {
    (0..<4).reduce(0) { $0 | UInt32(bytes[offset + $1]) << (8 * $1) }
}

/// Random and mutated requests: whatever the guest sends, the decoder returns, never traps, and never builds more entries
/// than the limit allows.
struct GPUDecoderRandomTests {
    private static let limits = GPULimits()

    private static func check(_ bytes: [UInt8]) {
        let decoded = GPUDecoder.decode(bytes, limits: limits)
        if bytes.count >= GPUWireSize.header {
            #expect(decoded.header.type == word(bytes, at: 0))
        } else {
            #expect(decoded.header == .unreadable)
        }
        if case .success(.resourceAttachBacking(_, let entries)) = decoded.command {
            #expect(entries.count <= limits.maxBackingEntries)
            #expect(entries.allSatisfy { $0.length > 0 && $0.address.addingReportingOverflow(UInt64($0.length)).overflow == false })
        }
        let answer = GPUResponse.error(.errUnspec).encoded(answering: decoded.header)
        #expect(answer.count == GPUWireSize.header)
    }

    @Test
    func randomBytesOfRandomLengthsDecodeWithoutTrapping() {
        var generator = SeededGenerator(seed: 0x5EED_0001)
        for _ in 0..<3000 {
            let length = Int.random(in: 0...600, using: &generator)
            var bytes = (0..<length).map { _ in UInt8.random(in: 0...255, using: &generator) }
            // Half of them carry a known type, so the decoder gets past the header.
            if length >= 4, Bool.random(using: &generator), let type = GPUCommandType.allCases.randomElement(using: &generator) {
                for index in 0..<4 { bytes[index] = UInt8(truncatingIfNeeded: type.rawValue >> (8 * index)) }
            }
            Self.check(bytes)
        }
    }

    @Test
    func mutatedWellFormedRequestsDecodeWithoutTrapping() {
        var generator = SeededGenerator(seed: 0x5EED_0002)
        let seeds = GPUDecoderTests.wellFormed.map(\.bytes)
        for round in 0..<3000 {

        }
