import ControlProtocol
import Foundation
import GuestConsole

/// The exit statuses of `vmr control`, from `sysexits.h`, besides 0 (done), 1 (the service refused the call) and
/// `ToolError.usageStatus` (64: a bad argument, no token, or a token that is not one).
enum ControlExit {
    /// The token file is missing or cannot be read (`EX_NOINPUT`).
    static let noInput: Int32 = 66
    /// Nothing serves the socket: the service is off or the app is not running (`EX_UNAVAILABLE`).
    static let unavailable: Int32 = 69
    /// No answer in time, or the service's own -32005 (`EX_TEMPFAIL`): what the call started may still finish.
    static let timedOut: Int32 = 75
    /// An answer this client cannot read, or none (`EX_PROTOCOL`).
    static let protocolError: Int32 = 76
    /// The token lacks the right (-32001), the token file is open to other users, or the socket is another user's
    /// (`EX_NOPERM`).
    static let notPermitted: Int32 = 77
}

/// Where `vmr control` finds the service and the token it presents.
enum ControlTarget {
    /// The environment variable a token can be given in. A token is never an argument: `ps` shows arguments to every user.
    static let tokenVariable = "VMR_CONTROL_TOKEN"
    /// The app's own variable for another data folder (CONTRIBUTING, `VMRUNNER_DATA_DIR`).
    static let dataDirectoryVariable = "VMRUNNER_DATA_DIR"
    /// A token file is at most this long; a token is 69 characters.
    static let tokenFileBytes = 4_096

    /// `socket` when given (relative to `currentDirectory`), otherwise `Control/control.sock` in the app's data folder: the one
    /// `VMRUNNER_DATA_DIR` names when it is set and not empty, otherwise `Library/Application Support/VMRunner` in `home`.
    static func socketPath(_ socket: String?, environment: [String: String], home: URL, currentDirectory: URL) throws -> String {
        if let socket {
            return URL(fileURLWithPath: socket, relativeTo: currentDirectory).standardizedFileURL.path
        }
        let data: URL
        if let override = environment[dataDirectoryVariable], !override.isEmpty {
            guard override.hasPrefix("/"), !override.split(separator: "/").contains("..") else {
                throw ToolError.usage("\(dataDirectoryVariable) must be an absolute path without `..`: \(override)")
            }
            data = URL(fileURLWithPath: override, isDirectory: true).standardizedFileURL
        } else {
            data = home.appending(path: "Library/Application Support/VMRunner", directoryHint: .isDirectory)
        }
        return data.appending(path: "Control/control.sock").path
    }

    /// The token: the contents of `file` when given (relative to `currentDirectory`), otherwise `VMR_CONTROL_TOKEN`; nil when
    /// neither is given. A file must be a regular file of this user that no other user may read or write, like a private
    /// key.
    static func token(file: String?, environment: [String: String], currentDirectory: URL) throws -> String? {
        if let file {
            return try checked(try contents(of: URL(fileURLWithPath: file, relativeTo: currentDirectory).standardizedFileURL), from: file)
        }
        guard let text = environment[tokenVariable], !text.isEmpty else { return nil }
        return try checked(text, from: tokenVariable)
    }

    private static func contents(of url: URL) throws -> String {
        // `attributesOfItem` describes a link itself and does not follow it.
        guard let attributes = try? FileManager.default.attributesOfItem(atPath: url.path) else {
            throw ToolError("cannot read the token file \(url.path)", status: ControlExit.noInput)
        }
        guard attributes[.type] as? FileAttributeType == .typeRegular else {
            throw ToolError("the token file \(url.path) is not a regular file", status: ControlExit.noInput)
        }
        guard (attributes[.ownerAccountID] as? NSNumber)?.uint32Value == getuid() else {
            throw ToolError("the token file \(url.path) belongs to another user", status: ControlExit.notPermitted)
        }
        let mode = (attributes[.posixPermissions] as? NSNumber)?.intValue ?? 0o777
        guard mode & 0o077 == 0 else {
            let shown = String(mode & 0o777, radix: 8)
            throw ToolError(
                "the token file \(url.path) is open to other users (mode \(shown)); make it private: chmod 600 \(url.path)",
                status: ControlExit.notPermitted)
        }
        guard (attributes[.size] as? NSNumber)?.intValue ?? 0 <= tokenFileBytes,
            let data = try? Data(contentsOf: url), data.count <= tokenFileBytes, let text = String(bytes: data, encoding: .utf8)
        else {
            throw ToolError(
                "cannot read the token file \(url.path) as text, or it is longer than \(tokenFileBytes) bytes", status: ControlExit.noInput)
        }
        return text
    }

    private static func checked(_ text: String, from source: String) throws -> String {
        let token = text.trimmingCharacters(in: .whitespacesAndNewlines)
        guard token.hasPrefix(ControlSecret.prefix), !token.contains(where: \.isWhitespace) else {
            throw ToolError.usage("\(source) does not hold a control token (one starts with \(ControlSecret.prefix))")
        }
        return token
    }
}

/// How `vmr control` reports an error thrown while calling the service: a message and an exit status.
enum ControlFailure {
    static func toolError(_ error: any Error, socket: String) -> ToolError {
        switch error {
        case ConsoleError.notServed:
            ToolError(
                "nothing serves \(socket): the control service is off (VMRunner's Settings, Automation) or the app is not running",
                status: ControlExit.unavailable)
        case let error as ConsoleError:
            ToolError("cannot reach \(socket): \(error)", status: ControlExit.unavailable)
        case ControlClientError.timedOut:
            ToolError("\(error); what the call started may still finish", status: ControlExit.timedOut)
        case ControlClientError.peerIsAnotherUser:
            ToolError("\(socket): \(error)", status: ControlExit.notPermitted)
        default:
            ToolError("\(error)", status: ControlExit.protocolError)
        }
    }
}

import ControlProtocol
import Foundation

/// What `vmr control status` found: whether something serves the socket and, with a token, whether the service accepts it.
struct ControlStatus {
    enum TokenCheck: String {
        case absent, unchecked, accepted, refused
    }

    let socket: String
    /// Whether a connection to the socket was accepted.
    var listening = false
    /// Why the service could not be used; nil when it answered.
    var failure: ToolError?
    var token = TokenCheck.absent
    /// How many machines the accepted token lists.
    var machines: Int?
    /// The service's error for a refused token.
    var refusal: String?

    /// Looks for the socket, connects to it, and with `token` lists the machines to check the token.
    static func check(socket: String, token: String?) -> Self {
        var status = Self(socket: socket)
        // Until the service has answered, a given token is unchecked, not absent.
        if token != nil { status.token = .unchecked }
        guard FileManager.default.fileExists(atPath: socket) else {
            status.failure = ToolError(
                "no socket at \(socket): the control service is off (VMRunner's Settings, Automation) or the app is not running",
                status: ControlExit.unavailable)
            return status
        }
        do {
            try ControlClient.connect(to: socket).close()
            status.listening = true
            guard let token else { return status }
            let answer = try ControlCaller(token: token, socket: socket).call(.list)
            if let code = answer.errorCode {
                status.token = .refused
                status.refusal = "\(answer.errorMessage ?? "refused") (\(code))"
            } else {
                status.token = .accepted
                let list = try? JSONDecoder().decode([String: [ControlMachine]].self, from: answer.result ?? Data())
                status.machines = list?["machines"]?.count
            }
        } catch {
            status.failure = ControlFailure.toolError(error, socket: socket)
        }
        return status
    }

    /// 0 when the service answered and the token, if any, was accepted.
    var exitStatus: Int32 {
        if let failure { return failure.status }
        return token == .refused ? ControlExit.notPermitted : 0
    }

    /// What a person reads.
    var lines: [String] {
        guard listening else { return [] }
        var lines = ["the control service is listening at \(socket)"]
        switch token {
        case .absent: lines.append("no token given (--token-file FILE or \(ControlTarget.tokenVariable)): not checked")
        case .unchecked: lines.append("a token was given but could not be checked")
        case .accepted: lines.append("the token is accepted: it lists \(machines ?? 0) machine(s)")
        case .refused: lines.append("the token is refused: \(refusal ?? "")")
        }
        return lines
    }

    /// The same as one JSON object, keys sorted.
    var json: String {
        var object: [String: Any] = ["socket": socket, "listening": listening, "token": token.rawValue]
        if let failure { object["problem"] = failure.message }
        if let machines { object["machines"] = machines }
        if let refusal { object["refusal"] = refusal }
        let data = (try? JSONSerialization.data(withJSONObject: object, options: [.sortedKeys, .withoutEscapingSlashes])) ?? Data()
        return String(bytes: data, encoding: .utf8) ?? "{}"
    }
}

import ArgumentParser
import ControlProtocol
import Foundation

struct ControlCommand: AsyncParsableCommand {
    static let configuration = CommandConfiguration(
        commandName: "control",
        abstract: "Call the local control service of a running VMRunner app.",
        discussion: """
            The app serves its control service on a private local socket once it is turned on in Settings (Automation): \
            Control/control.sock in the app's data folder, which is $VMRUNNER_DATA_DIR when that is set; --socket names \
            another. A call presents a control token made in Settings, read from --token-file FILE (a file only you can read: \
            chmod 600) or from $VMR_CONTROL_TOKEN; never from an argument, which every user can see with ps. --json prints \
            the service's answer as JSON.

            Exit status: 0 done; 1 the service refused the call (no such machine, agent access off, refused); 64 a usage \
            error, no token, or a token that is not one; 66 the token file cannot be read; 69 nothing serves the socket (the \
            service is off or the app is not running); 75 no answer in time, and what the call started may still finish; 76 \
            an answer this client cannot read; 77 not permitted (the token lacks the capability, the token file is open to \
            other users, or another user serves the socket).
            """,
        subcommands: [Status.self]
    )

    /// `--socket` and `--token-file`, shared by every subcommand.
    struct TargetOptions: ParsableArguments {
        @Option(
            name: .long, help: ArgumentHelp("The control socket (default: Control/control.sock in the data folder).", valueName: "path"))
        var socket: String?

        @Option(
            name: .long,
            help: ArgumentHelp("A file holding the token, readable by you only (default: $VMR_CONTROL_TOKEN).", valueName: "file"))
        var tokenFile: String?

        func socketPath() throws -> String {
            try ControlTarget.socketPath(
                socket, environment: ProcessInfo.processInfo.environment, home: FileManager.default.homeDirectoryForCurrentUser,
                currentDirectory: Self.currentDirectory)
        }

        /// The token, or nil when none is given.
        func token() throws -> String? {
            try ControlTarget.token(
                file: tokenFile, environment: ProcessInfo.processInfo.environment, currentDirectory: Self.currentDirectory)
        }

        private static var currentDirectory: URL {
            URL(fileURLWithPath: FileManager.default.currentDirectoryPath, isDirectory: true)
        }
    }

    struct Status: AsyncParsableCommand {
        static let configuration = CommandConfiguration(
            abstract: "Say whether the control service answers and, with a token, whether it is accepted.",
            discussion: """
                Looks for the socket and connects to it; when a token is given (it is optional here), lists the machines with \
                it. Exits 0 when the service answers and the token, if any, is accepted; 69 when nothing serves the socket; 77 \
                when the token is refused.
                """
        )

        @OptionGroup var target: TargetOptions

        @Flag(help: "Print the result as one JSON object.")
        var json = false

        func run() async throws {
            try await perform {
                // The token is read first: a token file open to other users is refused before anything is sent.
                let token = try target.token()
                let status = ControlStatus.check(socket: try target.socketPath(), token: token)
                if json {
                    Output.out(status.json)
                } else {
                    for line in status.lines { Output.out(line) }
                }
                if let failure = status.failure { throw failure }
                return status.exitStatus
            }
        }
    }
}
