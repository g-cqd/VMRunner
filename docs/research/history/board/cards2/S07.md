- **`ContainerWorkload`** (a `Workload` with no capabilities and no kill switch) wraps one `ContainerProcess`. `start()` prepares the scratch folder (`<machine bundle>/container-scratch`, 0700, the new `VMBundle.containerScratch`), spawns the runner, and returns when the container is up: the runner reported its `boot` or `run` phase, or the container printed or ended first. After the start, the runner's end is the machine stopping: status 0 or 75 (`passed`, `stopped`) is a clean stop; status 1 with a non-zero container status is "The container exited with status N."; any other status is the runner's last error or "The container runner stopped with status N."; a signal is "The container runner was ended by signal N." unless the app asked for the stop (a stop request or a kill), which is never an error whatever ended the runner.
```swift
import Foundation

/// A container machine's workload: one container runner process, followed through its events.
///
/// `start()` returns when the container is up, that is when the runner reports that it booted or started the command, or when
/// the container printed or ended first; a runner that ends before that fails the start with what it said. After that the
/// runner's end is reported as the machine stopping: cleanly when the runner passed or was stopped, with a message when the
/// container or the runner failed. A stop the app asked for is never an error, whatever signal ended the runner.
@MainActor
final class ContainerWorkload: Workload {
    /// The runner's exit statuses that are not failures: it passed, or it was stopped before the command ended.
    private static let cleanStatuses: Set<Int32> = [0, 75]
    /// The runner's exit status for "the container ran and its result was wrong".
    private static let failedStatus: Int32 = 1

    private let process: ContainerProcess
    private let scratch: URL
    private var starting: CheckedContinuation<Void, any Error>?
    private var stopAsked = false
    private var killed = false

    init(process: ContainerProcess, scratch: URL) {
        self.process = process
        self.scratch = scratch
    }

    var capabilities: WorkloadCapabilities { [] }
    var canStop: Bool { process.isRunning }
    var onGuestStopped: (@MainActor () -> Void)?
    var onStoppedWithError: (@MainActor (_ message: String) -> Void)?
    var networkCutoff: NetworkCutoff? { nil }
    var workloadLog: ContainerLog? { process.log }

    func start() async throws {
        try ContainerScratch.prepare(scratch)
        process.onEvent = { [weak self] in self?.apply($0) }
        process.onExit = { [weak self] in self?.ended($0) }
        try process.start()
        // No event or exit can be applied before this suspends: they run on this actor.
        try await withCheckedThrowingContinuation { starting = $0 }
    }

    /// Writes the stop line; the runner stops the container and exits, or the process is ended by signal after a grace.
    func requestStop() throws {
        guard process.isRunning else { throw ContainerProcessError(reason: "The container is not running.") }
        stopAsked = true
        process.stop()
    }

    /// Ends the runner at once and returns when it has ended.
    func stop() async throws {
        guard process.isRunning else { return }
        killed = true
        process.kill()
        _ = await process.waitForExit()
    }

    // MARK: Following the runner

    private func apply(_ event: ContainerEvent) {
        guard starting != nil else { return }
        switch event {
        case .phase(let name, _):
            if name == "boot" || name == "run" { becomeReady() }
        case .output, .exited:
            becomeReady()
        case .error, .finished:
            break
        }
    }

    private func becomeReady() {
        let pending = starting
        starting = nil
        pending?.resume()
    }

    private func ended(_ exit: ContainerExit) {
        let failure = failure(of: exit)
        if let pending = starting {
            starting = nil
            if let failure {
                pending.resume(throwing: ContainerProcessError(reason: failure))
                return
            }
            // The start returns, but the run is over: report it, in this order.
            pending.resume()
        }
        if let failure { onStoppedWithError?(failure) } else { onGuestStopped?() }
    }

    /// Why the run failed, or `nil` when it ended as it should.
    private func failure(of exit: ContainerExit) -> String? {
        if exit.signaled { return stopAsked || killed ? nil : "The container runner was ended by signal \(exit.status)." }
        if Self.cleanStatuses.contains(exit.status) { return nil }
        let log = process.log
        if exit.status == Self.failedStatus, let status = log.containerStatus, status != 0 {
            return "The container exited with status \(status)."
        }
        return log.errors.last ?? "The container runner stopped with status \(exit.status)."
    }
}
```

```swift
import Foundation

/// Builds the workload of a container machine: the container runner shipped inside the app, started with the machine's
/// container settings and its own working folder.
struct ContainerMachineFactory: MachineFactory {
    static let missingRunnerMessage =
        "This build of the app does not include the container runner; build it with vmr build-app."

    /// Where the app bundle keeps the runner. A build that is not an app bundle has none there.
    static var bundledRunner: URL { Bundle.main.bundleURL.appending(path: "Contents/Helpers/ContainerRunner") }

    let runner: URL

    init(runner: URL = Self.bundledRunner) {
        self.runner = runner
    }

    func prepare(
        _ config: VMConfig, bundle: VMBundle, display: (width: Int, height: Int), suspendAllowed: Bool
    ) async throws -> PreparedMachine {
        guard FileManager.default.isExecutableFile(atPath: runner.path) else {
            throw ContainerProcessError(reason: Self.missingRunnerMessage)
        }
        guard let spec = config.container else {
            throw ContainerProcessError(reason: "“\(config.name)” has no container settings.")
        }
        let scratch = bundle.containerScratch
        let process = ContainerProcess(executable: runner, arguments: ContainerRunnerArguments.make(for: spec, scratch: scratch))
        return PreparedMachine(
            machine: ContainerWorkload(process: process, scratch: scratch), network: nil,
            suspend: .unavailable(reason: "A container can't be suspended."))
    }
}
```

```swift
import SwiftUI

/// What a container machine shows in place of a screen: the container's output, the time each runner phase took, how the run
/// ended, and the runner's own diagnostics under a disclosure. The text is bounded and already filtered by `ContainerLog`.
struct ContainerLogView: View {
    let log: ContainerLog

    var body: some View {
        VStack(alignment: .leading, spacing: 0) {
            ScrollView {
                Text(log.output.markedText)
                    .font(.system(.body, design: .monospaced))
                    .textSelection(.enabled)
                    .frame(maxWidth: .infinity, alignment: .leading)
                    .padding(10)
            }
            .defaultScrollAnchor(.bottom)
            .accessibilityLabel("Container output")
            Divider()
            footer
        }
        .background(.background)
    }

    private var footer: some View {
        VStack(alignment: .leading, spacing: 6) {
            if let summary = Self.phaseSummary(log.phases) {
                Text(summary).foregroundStyle(.secondary)
            }
            if let status = Self.statusLine(for: log) {
                Text(status)
            }
            ForEach(Array(log.errors.enumerated()), id: \.offset) { _, message in
                Text(message).foregroundStyle(.red)
            }
            if !log.diagnostics.text.isEmpty {
                DisclosureGroup("Runner diagnostics") {
                    ScrollView {
                        Text(log.diagnostics.markedText)
                            .font(.system(.caption, design: .monospaced))
                            .textSelection(.enabled)
                            .frame(maxWidth: .infinity, alignment: .leading)
                    }
                    .frame(maxHeight: 120)
                }
            }
        }
        .font(.callout)
        .frame(maxWidth: .infinity, alignment: .leading)
        .padding(10)
    }

    /// `pull 1.2 s · boot 0.9 s`, or `nil` before the first phase ends.
    static func phaseSummary(_ phases: [ContainerLog.Phase]) -> String? {
        guard !phases.isEmpty else { return nil }
        return phases.map { "\($0.name) \($0.seconds.formatted(.number.precision(.fractionLength(1)))) s" }.joined(separator: " · ")
    }

    /// How the run ended, once it did.
    static func statusLine(for log: ContainerLog) -> String? {
        if let status = log.containerStatus { return "The container exited with status \(status)." }
        guard let status = log.finishedStatus else { return nil }
        return "The container runner finished with status \(status)."
    }
}
```

```swift
// MARK: The first boot of a virtual machine

extension VMController {
    /// Provisioning is only consumed by a boot that keeps its result: a disposable clone or a recovery boot
    /// would discard the account that was just created.
    private func recordFirstBoot(of config: VMConfig, storedMACAddress: String?, in library: any MachineStore) {
        guard config.workload == .vm, !config.hasBooted, !config.ephemeral, !config.bootIntoRecovery else { return }
        var updated = config
        updated.hasBooted = true
        updated.provisioning.password = ""
        // Putting the stored address back is an invariant guard: ephemeral starts never reach this today.
        library.update(updated.persistable(storedMACAddress: storedMACAddress))
    }
}

// MARK: Starting a machine that is not a virtual machine

extension VMController {
    /// Starts a workload that has no display, MAC address, disposable disk or saved state: none of the setup `start` does for a
    /// virtual machine applies, so the launch carries nothing of it.
    private func startWithoutGuestSetup(_ config: VMConfig, bundle: VMBundle, library: any MachineStore) async {
        let noDisplay = StartDisplay.Decision(
            width: 0, height: 0, source: .configuredSizeWithoutWindow, cap: .notApplicable, rememberedSizeIgnored: false)
        launch = MachineLaunch(
            config: config, bundle: bundle, library: library, display: noDisplay, windowSize: nil, suspendAllowed: false,
            storedMACAddress: nil)
        await drive(.startRequested, run: run)
    }
}
```
