Why: see ADR-0055 (D19). The provider keeps fetching the netboot `SHA256SUMS` from `deb.debian.org`; a new trust,

subkey (D19 lets the other two signatures be skipped), checks that it names `trixie` and point release 13.7 or later, and
believes the `SHA256SUMS` only when its SHA-256 is the one `InRelease` lists for `main/installer-arm64/current/images/SHA256SUMS`.
The subkey was read on 2026-10-22 from `https://ftp-master.debian.org/keys/archive-key-13.asc` (the primary key
`04B5 4C3C DCA7 9751 B16B C6B5 2256 29DF 75B1 88BD` is on `https://ftp-master.debian.org/keys.html`); it expires on 2035-03-28
(`2_058_699_029`); `gpg` verified that day's `InRelease` with it (captured in `cards2/fixtures/raw/`). The real `InRelease` is
140 KB, too large for a test file under the file-length limit, so the tests sign synthetic ones; the live check is the owner's.

2. `ChecksumTrust.swift`: the first line `import Foundation` becomes the two lines `import CryptoKit` and `import Foundation`. Then
   insert directly before the line `/// The steps the signed trusts share.` (the doc comment of `private enum SignatureSupport`):

```swift
/// Debian signs no checksum list of its installer images by itself: the archive's `InRelease`, a cleartext message that several
/// archive keys sign, lists the SHA-256 of every index of the suite, the images' `SHA256SUMS` among them. The checksum file is
/// believed when its SHA-256 is the one the signed `InRelease` lists for its path, and the suite is the release this app serves
/// and not older than its floor (ADR-0055).
struct DebianReleaseTrust: ChecksumTrust {
    /// The checksum file's path in `InRelease`, relative to the suite: `main/installer-arm64/current/images/SHA256SUMS`.
    var indexPath: String
    /// The release the suite must be (`Codename:`), for example `trixie`: a signed `InRelease` of another release is genuine and
    /// still not the one this app serves.
    var codename: String
    /// The oldest point release accepted (`Version:`), for example `13.7`.
    var minimumVersion: String
    var keys: [PGP.PinnedKey]

    func vouch(
        for checksumFile: Data, at checksumURL: URL, image imageURL: URL, now: Date, fetch: SmallFetch
    ) async throws -> VouchedChecksum {
        let address = checksumURL.absoluteString
        guard address.hasSuffix("/" + indexPath), let releaseURL = URL(string: String(address.dropLast(indexPath.count)) + "InRelease")
        else { throw SetupError.missingFile("the suite of \(checksumURL.lastPathComponent)") }
        let name = releaseURL.lastPathComponent
        let trusted = try SignatureSupport.keys(keys, for: name)
        _ = try DownloadPolicy.validate(releaseURL)
        let document = try await fetch(releaseURL, Fetch.smallLimit)
        let signed = try SignatureSupport.refusing(name) { () throws(PGP.Failure) in
            try PGP.Check.verifyClearSigned(Array(document), keys: trusted, now: now)
        }
        let release = DebianRelease(text: try LinuxProvider.text(of: Data(signed), at: releaseURL))
        guard release.codename == codename else {
            throw SetupError.signatureRejected(file: name, reason: "it describes Debian \(release.codename), not \(codename)")
        }
        guard let version = ReleaseVersion(release.version), let floor = ReleaseVersion(minimumVersion), !(version < floor) else {
            throw SetupError.releaseTooOld(file: name, oldest: minimumVersion)
        }
        guard release.sha256[indexPath] == SHA256.hash(data: checksumFile).hexString else {
            throw SetupError.signatureRejected(
                file: checksumURL.lastPathComponent, reason: "its SHA-256 is not the one the signed \(name) lists for it")
        }
        return VouchedChecksum(text: try LinuxProvider.text(of: checksumFile, at: checksumURL))
    }
}

/// What this app reads of a Debian `Release` file: the suite's codename and version, and the SHA-256 of each index.
struct DebianRelease {
    var codename = ""
    var version = ""
    /// The lowercase SHA-256 that the `SHA256:` section lists for each path.
    var sha256: [String: String] = [:]

    init(text: String) {
        var inSHA256 = false
        for line in text.split(separator: "\n") {
            guard line.hasPrefix(" ") else {
                inSHA256 = line == "SHA256:"
                if line.hasPrefix("Codename: ") { codename = String(line.dropFirst("Codename: ".count)) }
                if line.hasPrefix("Version: ") { version = String(line.dropFirst("Version: ".count)) }
                continue
            }
            let fields = line.split(separator: " ")
            if inSHA256, fields.count == 3, fields[0].utf8.count == 64 { sha256[String(fields[2])] = fields[0].lowercased() }
        }
    }
}
```

```swift
import CryptoKit
import Foundation
import Testing

@testable import VMRunner

private let suite = "https://deb.debian.org/debian/dists/stable/"
private let indexPath = "main/installer-arm64/current/images/SHA256SUMS"
private let checksums = "7352cf14ab29a565c8c8b5089757882ce7fbbb06431973208c97ba51b6af6486  ./netboot/mini.iso\n"

struct DebianReleaseTests {
    @Test func onlyTheSHA256SectionNamesDigestsAndTheHeaderNamesTheRelease() {
        let digest = String(repeating: "ab", count: 32)
        let text = [
            "Origin: Debian", "Version: 13.7", "Codename: trixie", "MD5Sum:", " 46994e39f2b234f4932a672f71576330   184036 \(indexPath)",
            "SHA256:", " \(digest)   184036 \(indexPath)", " short   1 other/file",

        #expect(release.codename == "trixie" && release.version == "13.7")
        #expect(release.sha256 == [indexPath: digest])
    }

/// `InRelease` messages signed, like Debian's, by a key that is not pinned and by the test key that is.
struct DebianReleaseTrustTests {
    private let signer: TestSigner
    private let other: TestSigner

    init() throws {
        signer = try SharedSigner.get()
        other = try TestSigner()
    }

    private var trust: DebianReleaseTrust {
        DebianReleaseTrust(indexPath: indexPath, codename: "trixie", minimumVersion: "13.7", keys: [signer.pinned])
    }

    private func inRelease(version: String = "13.7", codename: String = "trixie", digest: String? = nil) throws -> String {
        let listed = digest ?? SHA256.hash(data: Data(checksums.utf8)).hexString
        let lines = [
            "Origin: Debian", "Suite: stable", "Version: \(version)", "Codename: \(codename)", "SHA256:",
            " \(listed) \(checksums.utf8.count) \(indexPath)",
        ]
        let canonical = Array(lines.joined(separator: "\r\n").utf8)
        let foreign = try other.signatureBody(over: canonical, SignatureDraft(type: 0x01))
        let mine = try signer.signatureBody(over: canonical, SignatureDraft(type: 0x01))
        let text = "-----BEGIN PGP SIGNED MESSAGE-----\nHash: SHA256\n\n" + lines.joined(separator: "\n") + "\n"
        return text + utf8String(TestSigner.armor(TestSigner.packet(foreign) + TestSigner.packet(mine)))
    }

    private func vouch(_ release: String, using used: DebianReleaseTrust? = nil) async throws -> VouchedChecksum {
        let mirror = FixtureMirror([suite + "InRelease": release])
        let url = try #require(URL(string: suite + indexPath))
        return try await (used ?? trust).vouch(
            for: Data(checksums.utf8), at: url, image: url.deletingLastPathComponent(), now: pgpTestNow, fetch: mirror.fetch)
    }

    /// Why `vouch` refused `release`, in words; nil when it did not refuse.
    private func refusal(_ release: String, using used: DebianReleaseTrust? = nil) async -> String? {
        do {
            _ = try await vouch(release, using: used)
            return nil
        } catch SetupError.signatureRejected(_, let reason) {
            return reason
        } catch SetupError.releaseTooOld(_, let oldest) {
            return "older than \(oldest)"
        } catch {
            return "another error: \(error)"
        }
    }

    @Test func theIndexIsBelievedWhenTheSignedReleaseListsItsDigest() async throws {
        #expect(try await vouch(inRelease()).text == checksums)
    }

    @Test func anotherDigestAnotherReleaseOrAnOlderPointReleaseIsRefusedForItsOwnReason() async throws {
        let wrongDigest = try inRelease(digest: String(repeating: "0", count: 64))
        #expect(await refusal(wrongDigest)?.contains("not the one the signed InRelease lists") == true)
        #expect(await refusal(try inRelease(codename: "forky"))?.contains("not trixie") == true)
        #expect(await refusal(try inRelease(version: "13.6")) == "older than 13.7")
    }

    @Test func aReleaseFileSignedOnlyByOtherKeysIsRefused() async throws {
        let lines = ["Codename: trixie", "Version: 13.7"]
        let foreign = try other.signatureBody(over: Array(lines.joined(separator: "\r\n").utf8), SignatureDraft(type: 0x01))
        let text = "-----BEGIN PGP SIGNED MESSAGE-----\nHash: SHA256\n\n" + lines.joined(separator: "\n") + "\n"
        let refused = await refusal(text + utf8String(TestSigner.armor(TestSigner.packet(foreign))))
        #expect(refused?.contains("not pinned for this download") == true)
    }

    @Test func aGenuineReleaseSignedByAnExpiredPinnedKeyIsRefused() async throws {
        var expired = signer.pinned
        expired.expires = UInt32(pgpTestNow.timeIntervalSince1970)
        let used = DebianReleaseTrust(indexPath: indexPath, codename: "trixie", minimumVersion: "13.7", keys: [expired])
        #expect(await refusal(try inRelease(), using: used)?.contains("has expired") == true)
        #expect(await refusal(try inRelease()) == nil, "the same file is accepted while the key is valid")
    }

    @Test func theNetbootProviderUsesThisTrustWithThePinnedArchiveKey() throws {
        let provider = LinuxProvider.debianNetbootMini
        let used = try #require(provider.trust as? DebianReleaseTrust)
        #expect(used.keys == [PGP.PinnedKeys.debianArchive13] && used.codename == "trixie" && used.minimumVersion == "13.7")
        #expect(try provider.checksumURL(for: LinuxRelease(path: "netboot/mini.iso", version: "")).absoluteString == suite + indexPath)
    }
}
```
