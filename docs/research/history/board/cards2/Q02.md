## Why

The runner needs a machine before it can answer anything. Decisions (INDEX-Q, owner decisions 2 to 5): an EFI boot from a
raw disk image (a stock cloud image boots unchanged), a cloud-init seed as a **second, read-only virtio disk** (the app's
own real run of 2026-10-23 found that path works; USB did not), entropy, the serial console written to `console.log` in the
scratch directory (output only, nothing types into the guest), **no network** (the guest needs nothing from it), no
keyboard, no pointer, and **none of the framework's graphics devices**: the custom GPU is the guest's only display. Caps
follow CONTRIBUTING's real-guest conditions and are enforced by the parser: at most 2 CPUs and 2048 MiB, a hard limit of
900 s by default (3600 s at most) after which the runner stops the machine, a scratch directory that is never `/`, the
home folder or anything in `~/Library`.

The device configuration (`VZCustomVirtioDeviceConfiguration`, compiled against the 27.0 SDK): device ID 16, PCI class
3 / subclass 0x80, 2 queues (control, cursor), `mandatoryFeatures.subset1 = 1` (`VIRTIO_F_VERSION_1`; the header says
the framework sets it in any case), no optional feature (no 3D, EDID, blob or context bits: two-dimensional only), the
16-byte configuration space `{events_read 0, events_clear 0, num_scanouts 1, num_capsets 0}`, no save and restore, and a
`VZCustomVirtioDeviceDelegateProvider` on the main queue. The provider holds its delegate weakly (header), so
`RunnerMain` keeps the `GPUDevice`. For now the device only records its life cycle (created, `DRIVER_OK` with the
negotiated features, each queue notification) and gives every element back unanswered with `returnToQueue()`: enough to
see whether the guest enumerates the device and binds a driver. Q06 replaces the draining with real command servicing.

Two runtime facts found while writing this card (both checked on this macOS 27.0.1 host with `--dry-run`, no machine
started): the framework raises an Objective-C exception (`Invalid fileHandleForWriting`) for `FileHandle.nullDevice` as a
serial port's output, so a dry run opens `/dev/null` instead; and `validate()` refuses the configuration without the
virtualization entitlement (`VZErrorDomain` code 2) and accepts it, custom device included, once the binary is signed ad
hoc with it.

/// Why a command line was refused.
package enum RunnerArgumentError: Error, Equatable, CustomStringConvertible {
    case unknownOption(String)
    case missingValue(option: String)
    case notANumber(option: String, value: String)
    case outOfRange(option: String, value: Int, allowed: ClosedRange<Int>)
    case missing(option: String)
    case notAbsolute(option: String, path: String)
    case refusedDirectory(String)

    package var description: String {
        switch self {
        case .unknownOption(let option): "unknown option '\(option)'"
        case .missingValue(let option): "\(option) needs a value"
        case .notANumber(let option, let value): "\(option) takes a whole number, not '\(value)'"
        case .outOfRange(let option, let value, let allowed):
            "\(option) \(value) is outside \(allowed.lowerBound)...\(allowed.upperBound)"
        case .missing(let option): "\(option) is required"
        case .notAbsolute(let option, let path): "\(option) needs an absolute path without '..', not '\(path)'"
        case .refusedDirectory(let reason): reason
        }
    }
}

/// What one boot of the runner uses: a scratch directory for everything it writes, a raw disk image, an optional seed image,
/// and caps on the guest's resources and run time.
package struct RunnerOptions: Equatable, Sendable {
    package var directory: String
    package var disk: String
    package var seed: String?
    package var cpuCount = 2
    package var memoryMiB = 2048
    package var maxSeconds = 900
    /// Build and validate the configuration, start nothing.
    package var dryRun = false

    package static let cpuRange = 1...2
    package static let memoryRange = 1024...2048
    package static let secondsRange = 10...3600

    /// The machine's files inside the scratch directory.
    package var variableStorePath: String { directory + "/efi-vars.fd" }
    package var consoleLogPath: String { directory + "/console.log" }
    package var pidPath: String { directory + "/runner.pid" }
}

/// What the command line asks for.
package enum RunnerRequest: Equatable, Sendable {
    case help
    case describe
    case run(RunnerOptions)
}

/// The runner's command line, read without touching the file system.
package enum RunnerArguments {
    package static let usage = """
        usage: VirtioGPURunner --directory <scratch> --disk <raw image> [--seed <iso>] [options]
               VirtioGPURunner --describe | --help

        Boots one Linux guest whose only display is the host-implemented virtio GPU device. Needs macOS 27 or later.
          --directory <path>   absolute scratch directory for everything the run writes (not /, the home folder or ~/Library)
          --disk <path>        the guest's raw, bootable disk image (EFI)
          --seed <path>        an image attached read-only as a second disk (a cloud-init seed)
          --cpus <n>           1 to 2 (default 2)
          --memory-mib <n>     1024 to 2048 (default 2048)
          --max-seconds <n>    10 to 3600 (default 900); the machine is stopped then, whatever it does
          --dry-run            build and validate the configuration, start nothing
          --describe           print the virtio identity and the configuration space the guest sees
        """

    /// Reads `arguments` (the program name already removed). `home` is the user's home folder, refused as a scratch
    /// directory with everything in its `Library`.
    package static func parse(_ arguments: [String], home: String) throws(RunnerArgumentError) -> RunnerRequest {
        var values: [String: String] = [:]
        var dryRun = false
        var index = 0
        while index < arguments.count {
            let argument = arguments[index]
            index += 1
            switch argument {
            case "-h", "--help": return .help
            case "--describe": return .describe
            case "--dry-run": dryRun = true
            case "--directory", "--disk", "--seed", "--cpus", "--memory-mib", "--max-seconds":
                guard index < arguments.count else { throw .missingValue(option: argument) }
                values[argument] = arguments[index]
                index += 1
            default: throw .unknownOption(argument)
            }
        }
        guard let directory = values["--directory"] else { throw .missing(option: "--directory") }
        guard let disk = values["--disk"] else { throw .missing(option: "--disk") }
        var options = RunnerOptions(
            directory: try scratchDirectory(directory, home: home), disk: try absolute(disk, option: "--disk"),
            seed: try values["--seed"].map { path throws(RunnerArgumentError) in try absolute(path, option: "--seed") })
        options.cpuCount = try number(values["--cpus"], option: "--cpus", in: RunnerOptions.cpuRange) ?? options.cpuCount
        options.memoryMiB = try number(values["--memory-mib"], option: "--memory-mib", in: RunnerOptions.memoryRange) ?? options.memoryMiB
        options.maxSeconds =
            try number(values["--max-seconds"], option: "--max-seconds", in: RunnerOptions.secondsRange) ?? options.maxSeconds
        options.dryRun = dryRun
        return .run(options)
    }

    private static func number(_ text: String?, option: String, in range: ClosedRange<Int>) throws(RunnerArgumentError) -> Int? {
        guard let text else { return nil }
        guard let value = Int(text) else { throw .notANumber(option: option, value: text) }
        guard range.contains(value) else { throw .outOfRange(option: option, value: value, allowed: range) }
        return value
    }

    /// `path` without a trailing slash, when it is absolute and has no `.` or `..` component.
    private static func absolute(_ path: String, option: String) throws(RunnerArgumentError) -> String {
        let components = path.split(separator: "/", omittingEmptySubsequences: true)
        guard path.hasPrefix("/"), !components.contains(".."), !components.contains(".") else {
            throw .notAbsolute(option: option, path: path)
        }
        return "/" + components.joined(separator: "/")
    }

    /// Refuses the root of the disk, the home folder and anything in its `Library`, by their names (the run creates the
    /// directory itself, so a link planted there in advance is the caller's own).
    private static func scratchDirectory(_ path: String, home: String) throws(RunnerArgumentError) -> String {
        let directory = try absolute(path, option: "--directory")
        let homeFolder = (try? absolute(home, option: "home")) ?? home
        if directory == "/" { throw .refusedDirectory("the root of the disk is not a scratch directory") }
        if directory == homeFolder { throw .refusedDirectory("your home folder is not a scratch directory") }
        let library = homeFolder + "/Library"
        if directory == library || directory.hasPrefix(library + "/") {
            throw .refusedDirectory("\(directory) is inside your Library; give a scratch directory under /tmp")
        }
        return directory
    }
}

    static func line(_ text: String, error: Bool = false) {
        let handle = error ? FileHandle.standardError : FileHandle.standardOutput
        try? handle.write(contentsOf: Data((text + "\n").utf8))

    /// `bytes` as two lower-case hexadecimal digits each, separated by spaces.
    static func hex(_ bytes: [UInt8]) -> String {
        bytes.map { byte in

            return digits.count == 1 ? "0" + digits : digits
        }.joined(separator: " ")
    }

import Foundation
import Virtualization

/// The host side of the virtio GPU device. The framework calls it on the device queue, which is the main queue, so its
/// state is main-actor state. For now it records the device's life cycle and gives every element back unanswered: that
/// shows whether the guest finds the device, binds a driver and sends commands.
@available(macOS 27.0, *)
@MainActor
final class GPUDevice: NSObject, VZCustomVirtioDeviceConfigurationDelegate, VZCustomVirtioDeviceDelegate {
    private let started = ContinuousClock.now
    private var notifications = 0
    private var elements = 0

    /// One line, prefixed with the seconds since the device object was made.
    func log(_ text: String) {
        let elapsed = ContinuousClock.now - started
        let milliseconds = elapsed.components.seconds * 1000 + elapsed.components.attoseconds / 1_000_000_000_000_000
        Output.line("[\(milliseconds / 1000).\(String(milliseconds % 1000 + 1000).dropFirst())] \(text)")
    }

    func customVirtioConfiguration(
        _ deviceConfiguration: VZCustomVirtioDeviceConfiguration, didCreateDevice device: VZCustomVirtioDevice
    ) {
        device.delegate = self
        log("device created")
    }

    func customVirtioDeviceDidAcceptDriverOk(_ device: VZCustomVirtioDevice) {
        let features = device.negotiatedFeatures
        log("DRIVER_OK, negotiated features subset0 \(features?.subset0 ?? 0) subset1 \(features?.subset1 ?? 0)")
    }

    func customVirtioDevice(_ device: VZCustomVirtioDevice, didReceiveNotificationFor queue: VZVirtioQueue) {
        notifications += 1
        var drained = 0
        while let element = queue.nextElement() {
            drained += 1
            element.returnToQueue()
        }
        elements += drained
        log(
            "queue \(queue.queueIndex): \(drained) element(s) given back unanswered (\(notifications) notifications, \(elements) in all)"
        )
    }

    func customVirtioDeviceWillStop(_ device: VZCustomVirtioDevice) { log("device stopping") }
    func customVirtioDeviceWillPause(_ device: VZCustomVirtioDevice) { log("device pausing") }
    func customVirtioDeviceWillResume(_ device: VZCustomVirtioDevice) { log("device resuming") }
    func customVirtioDeviceWillReset(_ device: VZCustomVirtioDevice) { log("device reset") }
}

import Foundation
import VirtioGPURunnerCore
import Virtualization

/// Why the configuration could not be built.
struct RunnerFailure: Error, CustomStringConvertible {
    let description: String

    init(_ description: String) {
        self.description = description
    }
}

/// A headless Linux machine whose only display is the host-implemented GPU device: EFI boot from a raw disk, an optional
/// read-only seed disk, entropy, a serial console written to a file, no network, no keyboard, no pointer and none of the
/// framework's own graphics devices.
@available(macOS 27.0, *)
@MainActor
enum MachineBuilder {
    /// With `creatingFiles` the EFI variable store and the console log are created in the scratch directory (a real start);
    /// without it nothing is written (a dry run) and the console goes nowhere.
    static func build(_ options: RunnerOptions, device: GPUDevice, creatingFiles: Bool) throws -> VZVirtualMachineConfiguration {
        let configuration = VZVirtualMachineConfiguration()
        configuration.cpuCount = options.cpuCount
        configuration.memorySize = UInt64(options.memoryMiB) << 20
        configuration.platform = VZGenericPlatformConfiguration()
        configuration.bootLoader = try bootLoader(options, creatingFiles: creatingFiles)
        configuration.storageDevices = try storage(options)
        configuration.entropyDevices = [VZVirtioEntropyDeviceConfiguration()]
        configuration.serialPorts = [try console(options, creatingFiles: creatingFiles)]
        configuration.customVirtioDevices = [gpu(device)]
        return configuration
    }

    /// The virtio GPU: device 16 on a display class, a control and a cursor queue, version 1 and no optional feature
    /// (two-dimensional only), one scanout and no capability set.
    static func gpu(_ device: GPUDevice) -> VZCustomVirtioDeviceConfiguration {
        let gpu = VZCustomVirtioDeviceConfiguration()
        gpu.deviceID = GPUWire.deviceID
        gpu.pciClassID = GPUWire.pciClassID
        gpu.pciSubclassID = GPUWire.pciSubclassID
        gpu.virtioQueueCount = GPUWire.queueCount
        gpu.mandatoryFeatures.subset1 = GPUWire.versionOneFeatureSubset1
        let space = GPUConfigSpace(eventsRead: 0, eventsClear: 0, scanoutCount: 1, capsetCount: 0)
        gpu.deviceSpecificConfiguration = VZVirtioDeviceSpecificConfiguration(configurationData: Data(space.bytes))
        gpu.supportsSaveRestore = false
        // The provider keeps its delegate weakly: the runner holds `device` for as long as the machine lives.
        gpu.provider = VZCustomVirtioDeviceDelegateProvider(deviceQueue: .main, delegate: device)
        return gpu
    }

    private static func requireFile(_ path: String) throws {
        guard FileManager.default.fileExists(atPath: path) else { throw RunnerFailure("\(path) does not exist") }
    }

    private static func bootLoader(_ options: RunnerOptions, creatingFiles: Bool) throws -> VZBootLoader {
        let loader = VZEFIBootLoader()
        let url = URL(filePath: options.variableStorePath)
        if FileManager.default.fileExists(atPath: url.path) || !creatingFiles {
            loader.variableStore = VZEFIVariableStore(url: url)
        } else {
            do {
                loader.variableStore = try VZEFIVariableStore(creatingVariableStoreAt: url)
            } catch {
                throw RunnerFailure("cannot create the EFI variable store \(url.path): \(error.localizedDescription)")
            }
        }
        return loader
    }

    private static func storage(_ options: RunnerOptions) throws -> [VZStorageDeviceConfiguration] {
        var devices: [VZStorageDeviceConfiguration] = []
        for (path, readOnly) in [(options.disk, false)] + (options.seed.map { [($0, true)] } ?? []) {
            try requireFile(path)
            do {
                let attachment = try VZDiskImageStorageDeviceAttachment(url: URL(filePath: path), readOnly: readOnly)
                devices.append(VZVirtioBlockDeviceConfiguration(attachment: attachment))
            } catch {
                throw RunnerFailure("cannot attach \(path): \(error.localizedDescription)")
            }
        }
        return devices
    }

    /// The guest's serial console, output only, to `console.log` (created private) or, in a dry run, to nothing.
    private static func console(_ options: RunnerOptions, creatingFiles: Bool) throws -> VZSerialPortConfiguration {
        let port = VZVirtioConsoleDeviceSerialPortConfiguration()
        let output: FileHandle
        if creatingFiles {
            guard FileManager.default.createFile(atPath: options.consoleLogPath, contents: nil, attributes: [.posixPermissions: 0o600]),
                let handle = FileHandle(forWritingAtPath: options.consoleLogPath)
            else { throw RunnerFailure("cannot create \(options.consoleLogPath)") }
            output = handle
        } else {
            // The framework refuses `FileHandle.nullDevice` (an exception, not an error); an opened /dev/null is accepted.
            guard let handle = FileHandle(forWritingAtPath: "/dev/null") else { throw RunnerFailure("cannot open /dev/null") }
            output = handle
        }
        port.attachment = VZFileHandleSerialPortAttachment(fileHandleForReading: nil, fileHandleForWriting: output)
        return port
    }
}

import Foundation
import VirtioGPURunnerCore
import Virtualization

/// One run: the scratch directory, the configuration, the machine, and the hard time limit after which the machine is
/// stopped whatever it does. Everything runs on the main queue (the framework's, the device's and the runner's).
@available(macOS 27.0, *)
@MainActor
final class RunnerMain: NSObject, VZVirtualMachineDelegate {
    private let options: RunnerOptions
    /// Held here because the device provider keeps its delegate weakly.
    private let device = GPUDevice()
    private var machine: VZVirtualMachine?

    init(options: RunnerOptions) {
        self.options = options
    }

    /// Builds and validates the configuration without writing a file or starting anything.
    func validate() throws {
        let configuration = try MachineBuilder.build(options, device: device, creatingFiles: false)
        try configuration.validate()
    }

    /// Creates the scratch directory, builds, validates and starts the machine, and arms the time limit.
    func start() throws {
        try FileManager.default.createDirectory(
            atPath: options.directory, withIntermediateDirectories: true, attributes: [.posixPermissions: 0o700])
        let pid = Data("\(ProcessInfo.processInfo.processIdentifier)\n".utf8)
        guard FileManager.default.createFile(atPath: options.pidPath, contents: pid, attributes: [.posixPermissions: 0o600]) else {
            throw RunnerFailure("cannot write \(options.pidPath)")
        }
        let configuration = try MachineBuilder.build(options, device: device, creatingFiles: true)
        try configuration.validate()
        let machine = VZVirtualMachine(configuration: configuration)
        machine.delegate = self
        self.machine = machine
        device.log("starting: \(options.cpuCount) CPUs, \(options.memoryMiB) MiB, limit \(options.maxSeconds) s")
        machine.start { result in
            MainActor.assumeIsolated {
                switch result {
                case .success: self.device.log("machine running")
                case .failure(let error): self.finish("the machine did not start: \(error.localizedDescription)", status: 1)
                }
            }
        }
        let limit = options.maxSeconds
        Task { @MainActor in
            try? await Task.sleep(for: .seconds(limit))
            self.stopAtLimit()
        }
    }

    private func stopAtLimit() {
        guard let machine else { return }
        device.log("time limit reached, stopping the machine")
        machine.stop { error in
            MainActor.assumeIsolated {
                self.finish("stopped at the time limit" + (error.map { ": \($0.localizedDescription)" } ?? ""), status: 0)
            }
        }
    }

    private func finish(_ text: String, status: Int32) {
        device.log(text)
        exit(status)
    }

    nonisolated func guestDidStop(_ virtualMachine: VZVirtualMachine) {
        MainActor.assumeIsolated { finish("the guest stopped", status: 0) }
    }

    nonisolated func virtualMachine(_ virtualMachine: VZVirtualMachine, didStopWithError error: any Error) {
        MainActor.assumeIsolated { finish("the machine stopped with an error: \(error.localizedDescription)", status: 1) }
    }
}
