Why: the steps become cloud-init configuration (ADR-0012 item 4; the study's table "Where a script can run"):
`early-boot` is a `bootcmd` entry (every boot, before the network), `first-boot` a `runcmd` entry (once per instance),
`every-boot` a script in cloud-init's per-boot folder. Each script travels base64-encoded (so no quoting of its text is
needed), is unpacked into `/var/lib/cloud/vmrunner` and run with its `env` values; the only quoting is of those values, for the shell
and then for YAML, both done by one function each. The output depends on nothing but the inputs, so it is pinned by golden
tests. What this version cannot honour is refused with the step's line instead of being dropped: installer phases (no
installer seed media yet), `ready` and `pre-shutdown` and `user: user` (no guest agent yet), and any `secretRef` (a secret
never goes in the seed). `meta-data` gives the instance identifier, which the caller chooses (the change that attaches the
seed decides which), and the host name.

None of this has run in a guest: the shape follows cloud-init's documented modules, and the attach step verifies it.

```swift
// The provisioning library has no caller outside its tests until a later change attaches the seed to a machine, so a
// declaration only the tests reach is expected here; every other dead-code rule still applies.
// @dw:disable referenced-only-by-tests

import Foundation

/// Turns a checked spec into the text files of a cloud-init NoCloud seed. Every output is a function of its inputs alone:
/// the same spec, scripts and instance identifier give the same bytes.
package enum SeedCompiler {
    /// Where a step's script is unpacked and run in the guest: on the root file system, which allows running files, and
    /// under cloud-init's own folder, which every cloud image has. Not `/run`: Debian and Ubuntu mount it `noexec`.
    static let scriptDirectory = "/var/lib/cloud/vmrunner"

    /// The `meta-data` file: the identifier cloud-init keys its once-per-instance work on, and the guest's host name.
    static func metaData(name: String, instanceID: String) throws -> String {
        let allowed = instanceID.allSatisfy { $0.isASCII && ($0.isLetter || $0.isNumber || $0 == "." || $0 == "_" || $0 == "-") }
        guard (1...64).contains(instanceID.utf8.count), allowed else {
            throw ProvisioningError(line: nil, "the instance identifier must be 1 to 64 ASCII letters, digits, ., _ or -")
        }
        return "instance-id: \(instanceID)\nlocal-hostname: \(name)\n"
    }

    /// The cloud-config made from the steps, in the spec's order: `bootcmd` for early-boot, `runcmd` for first-boot, and a
    /// file in cloud-init's per-boot folder for every-boot. A step this version cannot run is refused with its line.
    static func cloudConfig(for steps: [ProvisioningStep], scripts: VerifiedScripts) throws -> String {
        var bootcmd: [String] = []
        var runcmd: [String] = []
        var perBoot: [String] = []
        for (offset, step) in steps.enumerated() {
            let number = String(offset + 1)
            let label = "step-" + String(repeating: "0", count: max(0, 3 - number.count)) + number
            switch step.phase {
            case .earlyBoot:
                let run = try command(for: step, label: label, scripts: scripts)
                bootcmd.append("- ['/bin/sh', '-c', \(yamlQuoted(run))]")
            case .firstBoot:
                let run = try command(for: step, label: label, scripts: scripts)
                runcmd.append("- ['/bin/sh', '-c', \(yamlQuoted(run))]")
            case .everyBoot:
                let run = try command(for: step, label: label, scripts: scripts)
                let wrapper = Data("#!/bin/sh\n\(run)\n".utf8).base64EncodedString()
                perBoot.append(
                    "- path: /var/lib/cloud/scripts/per-boot/vmrunner-\(label)\n  permissions: '0700'\n  encoding: b64\n"
                        + "  content: '\(wrapper)'")
            case .preInstall, .postInstall, .installError:
                throw ProvisioningError(
                    line: step.line, "\(step.phase.rawValue) runs in an installer, and installer seed media are not built yet")
            case .ready, .preShutdown:
                throw ProvisioningError(line: step.line, "\(step.phase.rawValue) needs the guest agent, which is not built yet")
            }
        }
        var text = "#cloud-config\n"
        if !bootcmd.isEmpty { text += "bootcmd:\n" + bootcmd.joined(separator: "\n") + "\n" }
        if !runcmd.isEmpty { text += "runcmd:\n" + runcmd.joined(separator: "\n") + "\n" }
        if !perBoot.isEmpty { text += "write_files:\n" + perBoot.joined(separator: "\n") + "\n" }
        return text
    }

    /// A shell command that unpacks the step's script into `scriptDirectory` and runs it with the step's values. The script
    /// is base64 (letters, digits, `+`, `/`, `=`: nothing a shell or YAML reads specially); only the values are quoted.
    private static func command(for step: ProvisioningStep, label: String, scripts: VerifiedScripts) throws -> String {
        guard step.user == .system else {
            throw ProvisioningError(line: step.line, "a step that runs as the user needs the guest agent, which is not built yet")
        }
        guard step.secrets.isEmpty else {
            throw ProvisioningError(
                line: step.line, "a secret is delivered by the guest agent, which is not built yet; it is never written into the seed")
        }
        let bytes = try scriptBytes(of: step, scripts: scripts)
        let encoded = Data(bytes).base64EncodedString()
        let path = "\(scriptDirectory)/\(label)"
        let values = step.variables.map { " " + shellQuoted("\($0.name)=\($0.value)") }.joined()
        let unpack = "mkdir -p \(scriptDirectory) && printf %s \(encoded) | base64 -d > \(path)"
        return unpack + " && chmod 700 \(path) && exec env\(values) \(path)"
    }

    private static func scriptBytes(of step: ProvisioningStep, scripts: VerifiedScripts) throws -> [UInt8] {
        switch step.script {
        case .inline(let text):
            return Array(text.utf8)
        case .file(let path, _):
            guard let bytes = scripts.contents(of: path) else {
                throw ProvisioningError(line: step.line, "\(path) was not checked against its sha256")
            }
            guard bytes.starts(with: Array("#!".utf8)) else {
                throw ProvisioningError(line: step.line, "\(path) does not start with #! and its interpreter")
            }
            return bytes
        }
    }

    /// `text` in single quotes for a POSIX shell; a single quote inside becomes `'\''`.
    private static func shellQuoted(_ text: String) -> String {
        "'" + text.replacing("'", with: "'\\''") + "'"
    }

    /// `text` as a YAML single-quoted scalar; a single quote inside is doubled.
    private static func yamlQuoted(_ text: String) -> String {
        "'" + text.replacing("'", with: "''") + "'"
    }
}
```

import Foundation
import Testing

@testable import Provisioning

/// A spec of one step: `rest` continues the line `  - at: `.
private func oneStep(_ rest: String) throws -> ProvisioningSpec {
    try ProvisioningSpec(
        parsing: "apiVersion: vmrunner.dev/v1alpha1\nkind: Provisioning\nmetadata:\n  name: node\nspec:\n  provision:\n  - at: " + rest)
}

private let trueScript = "\n    script: |\n      #!/bin/sh\n      true\n"

struct SeedCompilerTests {
    private func noFiles(_ spec: ProvisioningSpec) throws -> VerifiedScripts {
        try ScriptFiles.verify(spec, in: FileManager.default.temporaryDirectory)
    }

    private func refusal(_ spec: ProvisioningSpec) throws -> ProvisioningError {
        let scripts = try noFiles(spec)
        return try #require(throws: ProvisioningError.self) { try SeedCompiler.cloudConfig(for: spec.steps, scripts: scripts) }
    }

    @Test
    func eachPhaseGoesToItsModuleInTheSpecsOrderAndTheOutputIsGolden() throws {
        let spec = try ProvisioningSpec(
            parsing: """
                apiVersion: vmrunner.dev/v1alpha1
                kind: Provisioning
                metadata:
                  name: golden
                spec:
                  provision:
                  - at: early-boot
                    script: |
                      #!/bin/sh
                      echo early
                  - at: first-boot
                    script: |
                      #!/bin/sh
                      echo "$GREETING"
                    env:
                      GREETING: it's here
                  - at: every-boot
                    script: |
                      #!/bin/sh
                      echo again
                """)
        let expected = [
            "#cloud-config",
            "bootcmd:",
            "- ['/bin/sh', '-c', 'mkdir -p /var/lib/cloud/vmrunner && printf %s IyEvYmluL3NoCmVjaG8gZWFybHkK "
                + "| base64 -d > /var/lib/cloud/vmrunner/step-001 && chmod 700 /var/lib/cloud/vmrunner/step-001 && "
                + "exec env /var/lib/cloud/vmrunner/step-001']",
            "runcmd:",
            "- ['/bin/sh', '-c', 'mkdir -p /var/lib/cloud/vmrunner && printf %s IyEvYmluL3NoCmVjaG8gIiRHUkVFV"
                + "ElORyIK | base64 -d > /var/lib/cloud/vmrunner/step-002 && chmod 700 /var/lib/cloud/vmrunner/step"
                + "-002 && exec env ''GREETING=it''\\''''s here'' /var/lib/cloud/vmrunner/step-002']",
            "write_files:",
            "- path: /var/lib/cloud/scripts/per-boot/vmrunner-step-003",
            "  permissions: '0700'",
            "  encoding: b64",
            "  content: 'IyEvYmluL3NoCm1rZGlyIC1wIC92YXIvbGliL2Nsb3VkL3ZtcnVubmVyICYmIHByaW50ZiAlcyBJeUV2WW1s"
                + "dUwzTm9DbVZqYUc4Z1lXZGhhVzRLIHwgYmFzZTY0IC1kID4gL3Zhci9saWIvY2xvdWQvdm1ydW5uZXIvc3RlcC0wMDMgJiYg"
                + "Y2htb2QgNzAwIC92YXIvbGliL2Nsb3VkL3ZtcnVubmVyL3N0ZXAtMDAzICYmIGV4ZWMgZW52IC92YXIvbGliL2Nsb3VkL3Zt"
                + "cnVubmVyL3N0ZXAtMDAzCg=='",
        ].joined(separator: "\n") + "\n"
        let scripts = try noFiles(spec)
        #expect(try SeedCompiler.cloudConfig(for: spec.steps, scripts: scripts) == expected)
        #expect(try SeedCompiler.cloudConfig(for: spec.steps, scripts: scripts) == expected, "the same input gives the same text")
    }

    @Test
    func theMetaDataNamesTheInstanceAndTheHost() throws {
        #expect(try SeedCompiler.metaData(name: "golden", instanceID: "i-0123") == "instance-id: i-0123\nlocal-hostname: golden\n")
        _ = try SeedCompiler.metaData(name: "golden", instanceID: String(repeating: "a", count: 64))
        for bad in ["", "has space", "a/b", String(repeating: "a", count: 65)] {
            let error = try #require(throws: ProvisioningError.self) { try SeedCompiler.metaData(name: "golden", instanceID: bad) }
            #expect(error.line == nil)
            #expect(error.message.contains("instance identifier"), "\(error)")
        }
    }

    @Test
    func aFileStepCarriesTheCheckedBytes() throws {
        let folder = try makeScratchFolder()
        defer { try? FileManager.default.removeItem(at: folder) }
        try Data("#!/bin/sh\necho hello\n".utf8).write(to: folder.appending(path: "hello.sh"))
        let spec = try oneStep(
            "first-boot\n    file: hello.sh\n    sha256: bfdeaeb08cffb6a36438bcd12dda25417e3cdd36f1e7e482a2849d539225288b\n")
        let config = try SeedCompiler.cloudConfig(for: spec.steps, scripts: try ScriptFiles.verify(spec, in: folder))
        #expect(config.contains("printf %s \(Data("#!/bin/sh\necho hello\n".utf8).base64EncodedString()) | base64 -d"))
    }

    @Test
    func aFileTheCheckDidNotCoverOrWithoutAnInterpreterIsRefused() throws {
        let folder = try makeScratchFolder()
        defer { try? FileManager.default.removeItem(at: folder) }
        let unchecked = try oneStep(
            "first-boot\n    file: hello.sh\n    sha256: bfdeaeb08cffb6a36438bcd12dda25417e3cdd36f1e7e482a2849d539225288b\n")
        let otherSpecsScripts = try noFiles(try oneStep("first-boot" + trueScript))
        let error = try #require(throws: ProvisioningError.self) {
            try SeedCompiler.cloudConfig(for: unchecked.steps, scripts: otherSpecsScripts)
        }
        #expect(error.line == 7)
        #expect(error.message == "hello.sh was not checked against its sha256")

        try Data("echo no shebang\n".utf8).write(to: folder.appending(path: "plain.sh"))
        let plain = try oneStep(
            "first-boot\n    file: plain.sh\n    sha256: 3c0cb907fb7a9918b0b318076e7d6eef110f7c8e4e3c2fbc15ec0d9507314f86\n")
        let scripts = try ScriptFiles.verify(plain, in: folder)
        let noInterpreter = try #require(throws: ProvisioningError.self) {
            try SeedCompiler.cloudConfig(for: plain.steps, scripts: scripts)
        }
        #expect(noInterpreter.message == "plain.sh does not start with #! and its interpreter")
    }

    /// The seed is readable by everything in the guest: a step that needs a secret is refused, never compiled without it.
    @Test
    func aStepWithASecretIsRefusedRatherThanCompiledWithoutIt() throws {
        let error = try refusal(try oneStep("first-boot" + trueScript + "    env:\n      TOKEN:\n        secretRef: k3s-token\n"))
        #expect(error.line == 7)
        #expect(error.message.contains("never written into the seed"), "\(error)")
    }

    @Test(arguments: ["pre-install", "post-install", "install-error", "ready", "pre-shutdown"])
    func aPhaseThisVersionCannotRunIsRefusedWithTheStepsLine(phase: String) throws {
        let error = try refusal(try oneStep(phase + trueScript))
        #expect(error.line == 7)
        #expect(error.message.hasPrefix(phase), "\(error)")
    }

    @Test
    func aStepThatRunsAsTheUserIsRefused() throws {
        let error = try refusal(try oneStep("first-boot\n    user: user" + trueScript))
        #expect(error.line == 7)
        #expect(error.message.contains("runs as the user"), "\(error)")
    }

    @Test
    func noStepsGiveAnEmptyCloudConfig() throws {
        #expect(try SeedCompiler.cloudConfig(for: [], scripts: try noFiles(try oneStep("first-boot" + trueScript))) == "#cloud-config\n")
    }
}
