/// The two queues of the device.
public enum GPUQueueKind: UInt16, Sendable {
    case control = 0
    case cursor = 1
}

/// One request as the device handled it.
public struct CommandLogEntry: Equatable, Sendable {
    /// 1 for the first request since the device was created; never reset.
    public let sequence: UInt64
    public let queue: GPUQueueKind
    /// The request's type as the guest sent it (0 when the request was too short to carry one).
    public let commandType: UInt32
    public let resourceID: UInt32
    /// The answer's type; `nil` for a cursor request, which gets no answer.
    public let responseType: GPUResponseType?
    /// The request's length in bytes.
    public let byteCount: Int
}

/// The last requests the device handled, in a ring of fixed capacity, plus counts of every request and answer since the
/// device was created. Memory is bounded whatever the guest sends: one entry per slot, one counter per distinct type.
public struct CommandLog: Sendable {
    public let capacity: Int
    private var ring: [CommandLogEntry] = []
    private var next = 0
    public private(set) var total: UInt64 = 0
    private var commandCounts: [UInt32: UInt64] = [:]
    private var responseCounts: [GPUResponseType: UInt64] = [:]
    /// Distinct command types counted; one past this, further unknown types share the counter of type 0xFFFF_FFFF.
    private static let maxCommandTypes = 64

    public init(capacity: Int = 1024) {
        self.capacity = max(capacity, 1)
        ring.reserveCapacity(self.capacity)
    }

    public mutating func append(_ entry: CommandLogEntry) {
        if ring.count < capacity {
            ring.append(entry)
        } else {
            ring[next] = entry
        }
        next = (next + 1) % capacity
        total += 1
        let type = commandCounts[entry.commandType] != nil || commandCounts.count < Self.maxCommandTypes ? entry.commandType : 0xFFFF_FFFF
        commandCounts[type, default: 0] += 1
        if let response = entry.responseType { responseCounts[response, default: 0] += 1 }
    }

    /// The entries still held, oldest first.
    public var entries: [CommandLogEntry] {
        ring.count < capacity ? ring : Array(ring[next...] + ring[..<next])
    }

    /// One line per request type and per answer type, in hexadecimal order, with how often each was seen.
    public func summary() -> [String] {
        let commands = commandCounts.sorted { $0.key < $1.key }.map { "command 0x\(String($0.key, radix: 16)): \($0.value)" }
        let responses = responseCounts.sorted { $0.key.rawValue < $1.key.rawValue }.map {
            "response 0x\(String($0.key.rawValue, radix: 16)): \($0.value)"
        }
        return ["requests: \(total)"] + commands + responses
    }
}

/// The pixels of the scanout after a flush: the scanout's rectangle of its resource, rows packed `width × 4` bytes apart.
public struct ScanoutUpdate: Equatable, Sendable {
    public let width: UInt32
    public let height: UInt32
    public let format: GPUPixelFormat
    public let pixels: [UInt8]
}

/// What the device does with one request: the answer to write back (`nil` on the cursor queue, whose requests get none), the
/// log entry, and the new picture when the request flushed the scanout.
public struct HandledCommand: Sendable {
    public let response: [UInt8]?
    public let logEntry: CommandLogEntry
    public let scanoutUpdate: ScanoutUpdate?
}

/// The state of the 2D device: its resources, its one scanout and the cursor. It decodes and carries out each request the
/// guest sends, completely, before the answer is returned, so a fenced request is done when its answer arrives.
public struct GPUDeviceState: Sendable {
    public private(set) var store: ResourceStore
    /// The resource shown on scanout 0 and the rectangle of it that is shown, or `nil` while the scanout is off.
    public private(set) var scanout: (resourceID: UInt32, rect: GPURect)?
    public private(set) var cursor: GPUCursor?
    private var sequence: UInt64 = 0

    public init(limits: GPULimits = GPULimits()) {
        store = ResourceStore(limits: limits)
    }

    /// Handles the bytes of one request, copied once out of guest memory by the caller.
    public mutating func handle(_ bytes: [UInt8], queue: GPUQueueKind, memory: some GuestMemoryReader) -> HandledCommand {
        sequence += 1
        let request = GPUDecoder.decode(bytes, limits: store.limits)
        var update: ScanoutUpdate?
        let response: GPUResponse?
        switch (queue, request.command) {
        case (_, .failure(let failure)):
            response = queue == .control ? .error(failure.responseType) : nil
        case (.control, .success(let command)):
            (response, update) = control(command, memory: memory)
        case (.cursor, .success(let command)):
            cursorRequest(command)
            response = nil
        }
        let entry = CommandLogEntry(
            sequence: sequence, queue: queue, commandType: request.header.type, resourceID: (try? request.command.get())?.resourceID ?? 0,
            responseType: response?.type, byteCount: bytes.count)
        return HandledCommand(response: response?.encoded(answering: request.header), logEntry: entry, scanoutUpdate: update)
    }

    /// Forgets every resource, the scanout and the cursor, as a device reset does. The sequence keeps counting.
    public mutating func reset() {
        store.removeAll()
        scanout = nil
        cursor = nil
    }

    private mutating func control(_ command: GPUCommand, memory: some GuestMemoryReader) -> (GPUResponse, ScanoutUpdate?) {
        switch command {
        case .getDisplayInfo:
            let rect = GPURect(x: 0, y: 0, width: store.limits.displayWidth, height: store.limits.displayHeight)
            return (.displayInfo([GPUDisplayMode(rect: rect, enabled: true)]), nil)
        case .resourceCreate2D(let id, let format, let width, let height):
            return (answer(store.create2D(resourceID: id, format: format, width: width, height: height)), nil)
        case .resourceUnref(let id):
            if scanout?.resourceID == id { scanout = nil }
            return (answer(store.unref(id)), nil)
        case .setScanout(let rect, let scanoutID, let id):
            return (answer(setScanout(rect: rect, scanoutID: scanoutID, resourceID: id)), nil)
        case .resourceFlush(let rect, let id):
            return flush(rect: rect, resourceID: id)
        case .transferToHost2D(let rect, let offset, let id):
            return (answer(store.transferToHost2D(resourceID: id, rect: rect, offset: offset, memory: memory)), nil)
        case .resourceAttachBacking(let id, let entries):
            return (answer(store.attachBacking(id, entries: entries)), nil)
        case .resourceDetachBacking(let id):
            return (answer(store.detachBacking(id)), nil)
        case .getEDID, .updateCursor, .moveCursor:
            // EDID is not offered, and cursor requests belong on the cursor queue.
            return (.error(.errUnspec), nil)
        }
    }

    private func answer(_ type: GPUResponseType) -> GPUResponse {
        type == .okNoData ? .noData : .error(type)
    }

    /// Resource 0 turns the scanout off; any other must exist and hold the rectangle shown.
    private mutating func setScanout(rect: GPURect, scanoutID: UInt32, resourceID id: UInt32) -> GPUResponseType {
        guard scanoutID == 0 else { return .errInvalidScanoutID }
        guard id != 0 else {
            scanout = nil
            return .okNoData
        }
        guard let resource = store.resource(id) else { return .errInvalidResourceID }
        guard rect.width > 0, rect.height > 0, rect.fits(width: resource.width, height: resource.height) else {
            return .errInvalidParameter
        }
        scanout = (id, rect)
        return .okNoData
    }

    /// A flush of the scanout's resource that touches the scanout's rectangle gives the new picture of the whole scanout.
    private func flush(rect: GPURect, resourceID id: UInt32) -> (GPUResponse, ScanoutUpdate?) {
        guard let resource = store.resource(id) else { return (.error(.errInvalidResourceID), nil) }
        guard rect.fits(width: resource.width, height: resource.height) else { return (.error(.errInvalidParameter), nil) }
        guard let scanout, scanout.resourceID == id, rect.intersects(scanout.rect) else { return (.noData, nil) }
        return (.noData, Self.picture(of: resource, in: scanout.rect))
    }

    /// The rectangle's rows of the resource; the rectangle was checked to fit when the scanout was set.
    private static func picture(of resource: GPUResource, in rect: GPURect) -> ScanoutUpdate {
        let rowBytes = Int(rect.width) * GPUPixelFormat.bytesPerPixel
        var pixels: [UInt8] = []
        pixels.reserveCapacity(rowBytes * Int(rect.height))
        for row in 0..<Int(rect.height) {
            let start = (Int(rect.y) + row) * resource.stride + Int(rect.x) * GPUPixelFormat.bytesPerPixel
            pixels.append(contentsOf: resource.pixels[start..<start + rowBytes])
        }
        return ScanoutUpdate(width: rect.width, height: rect.height, format: resource.format, pixels: pixels)
    }

    /// Cursor requests are recorded, not drawn: the scanout picture has no cursor.
    private mutating func cursorRequest(_ command: GPUCommand) {
        switch command {
        case .updateCursor(let new):
            cursor = new
        case .moveCursor(let moved):
            // A move keeps the image and hot spot of the last update.
            cursor = GPUCursor(
                scanoutID: moved.scanoutID, x: moved.x, y: moved.y, resourceID: cursor?.resourceID ?? 0, hotX: cursor?.hotX ?? 0,
                hotY: cursor?.hotY ?? 0)
        default:
            break
        }
    }
}

/// The bounded log of requests and its counts.
struct CommandLogTests {
    private static func entry(_ sequence: UInt64, type: UInt32 = 0x0100, response: GPUResponseType? = .okNoData) -> CommandLogEntry {
        CommandLogEntry(sequence: sequence, queue: .control, commandType: type, resourceID: 0, responseType: response, byteCount: 24)

    @Test
    func theRingKeepsTheLastEntriesOldestFirst() {
        var log = CommandLog(capacity: 3)
        for sequence in 1...2 { log.append(Self.entry(UInt64(sequence))) }
        #expect(log.entries.map(\.sequence) == [1, 2])
        for sequence in 3...7 { log.append(Self.entry(UInt64(sequence))) }
        #expect(log.entries.map(\.sequence) == [5, 6, 7])
        #expect(log.total == 7)
        #expect(CommandLog(capacity: 0).capacity == 1)
    }

    @Test
    func theSummaryCountsEveryRequestAndAnswerSinceTheStart() {
        var log = CommandLog(capacity: 2)
        log.append(Self.entry(1, type: 0x0101))
        log.append(Self.entry(2, type: 0x0101, response: .errOutOfMemory))
        log.append(Self.entry(3, type: 0x0300, response: nil))
        #expect(
            log.summary() == [
                "requests: 3", "command 0x101: 2", "command 0x300: 1", "response 0x1100: 1", "response 0x1201: 1",

    @Test

    }

/// Random and mutated requests through the whole device: it never traps, every answer is a well-formed answer of a known
/// type and size, and the host copies never exceed the limit.
struct GPUDeviceStateRandomTests {
    private static let knownTypes = Set(GPUResponseType.allCases.map(\.rawValue))

    private static func limits() -> GPULimits {
        var limits = GPULimits()
        limits.maxResources = 8
        limits.maxResourceBytes = 64 * 64 * 4
        limits.maxTotalResourceBytes = 3 * 64 * 64 * 4
        return limits
    }

    private static func check(_ handled: HandledCommand, queue: GPUQueueKind, state: GPUDeviceState) {
        if queue == .cursor {
            #expect(handled.response == nil)
        } else if let answer = handled.response {
            let type = word(answer, at: 0)
            #expect(Self.knownTypes.contains(type))
            #expect(answer.count == (type == GPUResponseType.okDisplayInfo.rawValue ? GPUWireSize.displayInfo : GPUWireSize.header))
        } else {
            Issue.record("a control request got no answer")
        }
        #expect(state.store.totalBytes <= state.store.limits.maxTotalResourceBytes)
        #expect(state.store.count <= state.store.limits.maxResources)
    }

    /// A request that is valid in shape, with small random values, so the store is exercised and not only the decoder.
    private static func plausible(using generator: inout SeededGenerator) -> [UInt8] {
        let small = { (generator: inout SeededGenerator) in UInt32.random(in: 0...70, using: &generator) }
        switch Int.random(in: 0..<7, using: &generator) {
        case 0: return request(.resourceCreate2D, words: [small(&generator) % 6, 2, small(&generator), small(&generator)])
        case 1:
            let count = Int.random(in: 0...3, using: &generator)
            let entries = (0..<count).map { _ in (UInt64(0x4000 + small(&generator) * 64), UInt32(small(&generator) * 256)) }
            return request(.resourceAttachBacking, words: [small(&generator) % 6, UInt32(count)], tail: entryBytes(entries))
        case 2:
            let rect = GPURect(x: small(&generator), y: small(&generator), width: small(&generator), height: small(&generator))
            return transfer(rect, offset: UInt64(small(&generator)) * 4, resource: small(&generator) % 6)
        case 3: return request(.setScanout, words: [0, 0, small(&generator), small(&generator), 0, small(&generator) % 6])
        case 4: return request(.resourceFlush, words: [0, 0, small(&generator), small(&generator), small(&generator) % 6, 0])
        case 5: return request(.resourceUnref, words: [small(&generator) % 6, 0])
        default: return request(.resourceDetachBacking, words: [small(&generator) % 6, 0])
        }
    }

    @Test
    func randomBytesOnBothQueuesAreAnsweredWellFormed() {
        var generator = SeededGenerator(seed: 0x5EED_0003)
        var state = GPUDeviceState(limits: Self.limits())
        var memory = FakeGuestMemory()
        memory.add(at: 0x4000, count: 64 * 1024)
        for _ in 0..<2500 {

            var bytes = (0..<length).map { _ in UInt8.random(in: 0...255, using: &generator) }
            if length >= 4, let type = GPUCommandType.allCases.randomElement(using: &generator) {
                for index in 0..<4 { bytes[index] = UInt8(truncatingIfNeeded: type.rawValue >> (8 * index)) }
            }
            let queue: GPUQueueKind = Bool.random(using: &generator) ? .control : .cursor
            Self.check(state.handle(bytes, queue: queue, memory: memory), queue: queue, state: state)
        }
    }

    @Test
    func plausibleAndMutatedSequencesKeepTheStoreWithinItsLimits() {
        var generator = SeededGenerator(seed: 0x5EED_0004)
        var state = GPUDeviceState(limits: Self.limits())
        var memory = FakeGuestMemory()
        memory.add(at: 0x4000, count: 64 * 1024, seed: 7)

                #expect(update.pixels.count == Int(update.width) * Int(update.height) * 4)
            }
            if (round + 1).isMultiple(of: 500) { state.reset() }
