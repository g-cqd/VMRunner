Why: `TerminalMode.raw` restores the terminal's settings on a normal return or a throw only (its `defer`). Raw mode turns
off the signal keys (Ctrl-C goes to the guest, as it should), but a `kill`, the terminal hanging up or a closed output
still deliver SIGTERM, SIGINT, SIGQUIT, SIGHUP or SIGPIPE, whose default action ends the process with the terminal left
raw. There is no signal handling anywhere in the repository today.

Design (decided): no handler runs in signal context, and nothing in this repository needs `nonisolated(unsafe)` or a
global. `SignalWatch` sets each watched signal to be ignored and receives it through a dispatch source instead (the kernel
still reports an ignored signal to such a source; this is the documented way to watch signals with dispatch); the source's
handler runs on an ordinary thread and writes the signal's number into a non-blocking pipe, under a lock that `stop()`
closes first, so no handler writes to a closed descriptor. The relay polls the pipe's read end (O03's `wake`) and returns
`.interrupted`; the stack then unwinds normally: `TerminalMode.raw`'s `defer` restores the settings, `ConsoleAttach` resets
the colours, prints `ended by signal N` and exits with `128 + N`, the shell's convention. `stop()` gives each signal its
default action back. A crash of the client itself (a trap) cannot be covered safely from inside the process; O05's help
text says `stty sane` restores the terminal. The repository's lint forbids `DispatchQueue` (the custom rule
`no_dispatch_queue`); the sources use the default queue and the name does not appear. `TerminalMode.raw` gains
`on descriptor:` (default standard input) so a test can run it on a pseudo-terminal.

```swift
import Foundation

/// Puts a terminal into raw mode (no echo, no line editing, no signal keys) while `body` runs, then puts it back, whether
/// `body` returns or throws. Does nothing when `descriptor` is not a terminal. A signal that would end the process is not
/// covered here: the console client turns those into an ordinary end of `body` with ``SignalWatch``.
enum TerminalMode {
    static func raw<Result>(on descriptor: Int32 = STDIN_FILENO, _ body: () throws -> Result) rethrows -> Result {
        var original = termios()
        // `tcgetattr` fills one termios; `tcsetattr` reads one.
        guard isatty(descriptor) == 1, unsafe tcgetattr(descriptor, &original) == 0 else { return try body() }
        var raw = original
        unsafe cfmakeraw(&raw)
        guard unsafe tcsetattr(descriptor, TCSANOW, &raw) == 0 else { return try body() }
        defer { _ = unsafe tcsetattr(descriptor, TCSANOW, &original) }
        return try body()
    }
}
```

2. Create `Sources/DevTools/Guest/SignalWatch.swift` with exactly:

```swift
import Foundation
import Synchronization

/// Turns the signals that would end the process into a byte on a pipe while it is active, so a loop that polls
/// ``descriptor`` ends on its own path and the terminal is put back before the process exits.
///
/// No code runs in signal context. Each signal is set to be ignored and is received through a dispatch source instead (the
/// kernel reports a signal to such a source even when it is ignored); the source's handler runs on an ordinary thread and
/// writes the signal's number to the pipe. ``stop()`` gives each signal its default action back.
final class SignalWatch {
    /// The signals the console client watches: the terminal hanging up, an interrupt or a quit sent with `kill` (raw mode
    /// sends the keys to the guest instead), the output closing, and a request to terminate.
    static let ending: [Int32] = [SIGHUP, SIGINT, SIGQUIT, SIGPIPE, SIGTERM]

    /// Readable once a watched signal arrived: one byte per signal, its number.
    let descriptor: Int32
    private let writer: Int32
    private let signals: [Int32]
    private let gate = Gate()
    private var sources: [any DispatchSourceSignal] = []

    /// Whether the pipe may still be written. A handler checks it and writes under the lock, and ``stop()`` closes it under
    /// the lock before closing the pipe, so no handler writes to a closed descriptor.
    private final class Gate: Sendable {
        let isOpen = Mutex(true)
    }

    /// Nil when the pipe cannot be made; the caller then goes on without a watch.
    init?(signals: [Int32]) {
        var ends: [Int32] = [-1, -1]
        // `pipe` fills exactly two descriptors.
        guard unsafe pipe(&ends) == 0 else { return nil }
        for end in ends { _ = fcntl(end, F_SETFL, fcntl(end, F_GETFL) | O_NONBLOCK) }
        descriptor = ends[0]
        writer = ends[1]
        self.signals = signals
        let shared = gate
        let target = ends[1]
        for number in signals {
            signal(number, SIG_IGN)
            let source = DispatchSource.makeSignalSource(signal: number)
            source.setEventHandler {
                shared.isOpen.withLock { isOpen in
                    guard isOpen else { return }
                    var byte = UInt8(truncatingIfNeeded: number)
                    // One byte from the stack. A full pipe drops it: one byte is enough to wake the loop.
                    _ = unsafe write(target, &byte, 1)
                }
            }
            source.resume()
            sources.append(source)
        }
    }

    deinit {
        stop()
    }

    /// The number of the first watched signal not yet read, or nil when none arrived; does not wait.
    func caught() -> Int32? {
        var byte: UInt8 = 0
        // One byte into the stack.
        return unsafe read(descriptor, &byte, 1) == 1 ? Int32(byte) : nil
    }

    /// Stops watching: no handler writes any more, the sources are cancelled, each signal gets its default action back and
    /// the pipe is closed. Calling it again does nothing.
    func stop() {
        let wasOpen = gate.isOpen.withLock { isOpen in
            defer { isOpen = false }
            return isOpen
        }
        guard wasOpen else { return }
        for source in sources { source.cancel() }
        sources.removeAll()
        for number in signals { signal(number, SIG_DFL) }
        close(descriptor)
        close(writer)
    }
}
```

```swift
import Foundation
import GuestConsole

/// `vmr guest console`: this terminal attached to a guest's console. What the guest prints passes through a
/// ``ConsoleOutputFilter`` before it reaches the terminal. However the session ends (detach, end of input, the console
/// closing, an error, or one of ``SignalWatch/ending``), the terminal's settings come back and the colours the guest may have
/// left set are reset.
enum ConsoleAttach {
    static func run(socket: String) throws -> Int32 {
        Output.err("attached to \(socket); Ctrl-] detaches")
        let watch = SignalWatch(signals: SignalWatch.ending)
        defer { watch?.stop() }
        let ending: ConsoleRelay.Ending
        do {
            ending = try TerminalMode.raw {
                try ConsoleRelay.run(
                    socketPath: socket, input: STDIN_FILENO, output: STDOUT_FILENO, filter: ConsoleOutputFilter(),
                    wake: watch?.descriptor ?? -1)
            }
        } catch let error as ConsoleError {
            resetScreen()
            throw GuestExec.failure(error)
        }
        resetScreen()
        switch ending {
        case .consoleClosed: Output.err("\nthe console closed")
        case .inputEnded: Output.err("\nend of input")
        case .detached: Output.err("\ndetached")
        case .interrupted:
            let number = watch?.caught() ?? SIGTERM
            Output.err("\nended by signal \(number)")
            return 128 + number
        }
        return 0
    }

    /// Resets what the guest's output may have left set on the terminal; nothing when standard output is not a terminal.
    private static func resetScreen() {
        guard isatty(STDOUT_FILENO) == 1 else { return }
        try? FileHandle.standardOutput.write(contentsOf: Data(ConsoleRelay.closingBytes(filtered: true)))
    }
}
```

```swift
import Foundation
import Testing

@testable import DevTools

/// Raw mode on a real pseudo-terminal pair, and the signal watch, without a guest.
struct TerminalModeTests {
    private struct Failure: Error {}

    /// A fresh pseudo-terminal pair; `body` gets the terminal's end.
    private func withTerminal(_ body: (Int32) throws -> Void) throws {
        var controller: Int32 = -1
        var terminal: Int32 = -1
        try #require(unsafe openpty(&controller, &terminal, nil, nil, nil) == 0)
        defer {
            close(terminal)
            close(controller)
        }
        try body(terminal)
    }

    private func settings(_ descriptor: Int32) -> termios {
        var current = termios()
        _ = unsafe tcgetattr(descriptor, &current)
        return current
    }

    /// The kernel sets PENDIN (retype pending input) when a terminal leaves raw mode, so that flag is left out of the comparison.
    private func sameFlags(_ lhs: termios, _ rhs: termios) -> Bool {
        let pending = tcflag_t(PENDIN)
        return lhs.c_iflag == rhs.c_iflag && lhs.c_oflag == rhs.c_oflag && lhs.c_cflag == rhs.c_cflag
            && lhs.c_lflag & ~pending == rhs.c_lflag & ~pending
    }

    @Test
    func rawModeHoldsWhileTheBodyRunsAndTheOriginalSettingsComeBackAfter() throws {
        try withTerminal { terminal in
            let before = settings(terminal)
            #expect(before.c_lflag & tcflag_t(ICANON) != 0, "a new pseudo-terminal starts in canonical mode")
            let during = TerminalMode.raw(on: terminal) { settings(terminal) }
            #expect(during.c_lflag & tcflag_t(ICANON | ECHO | ISIG) == 0)
            #expect(!sameFlags(during, before))
            #expect(sameFlags(settings(terminal), before))
        }
    }

    @Test
    func theOriginalSettingsComeBackWhenTheBodyThrows() throws {
        try withTerminal { terminal in
            let before = settings(terminal)
            #expect(throws: Failure.self) {
                try TerminalMode.raw(on: terminal) {
                    #expect(settings(terminal).c_lflag & tcflag_t(ECHO) == 0)
                    throw Failure()
                }
            }
            #expect(sameFlags(settings(terminal), before))
        }
    }

    @Test
    func aDescriptorThatIsNotATerminalIsLeftAlone() throws {
        var ends: [Int32] = [-1, -1]
        try #require(unsafe pipe(&ends) == 0)
        defer { for end in ends { close(end) } }
        #expect(TerminalMode.raw(on: ends[0]) { 42 } == 42)
    }

    /// SIGUSR1 stands in for the signals the console watches: the same path, and nothing else in the test process uses it.
    @Test(.timeLimit(.minutes(1)))
    func aWatchedSignalBecomesAByteOnThePipeAndDoesNotEndTheProcess() throws {
        let watch = try #require(SignalWatch(signals: [SIGUSR1]))
        defer { watch.stop() }
        // The source starts listening shortly after it is made, so the signal is sent until the pipe answers.
        var arrived = false
        for _ in 0..<100 where !arrived {
            kill(getpid(), SIGUSR1)
            var entry = pollfd(fd: watch.descriptor, events: Int16(POLLIN), revents: 0)
            arrived = unsafe poll(&entry, 1, 100) == 1
        }
        #expect(arrived)
        #expect(watch.caught() == SIGUSR1)
        #expect(SignalWatch.ending == [SIGHUP, SIGINT, SIGQUIT, SIGPIPE, SIGTERM])
    }
}
```

echo and signal keys are off; after a normal return the four flag words equal the original (but for PENDIN, which the kernel sets on leaving raw mode:
measured, the restored `c_lflag` is the original plus that one bit); after a body that throws, the same. A pipe (not a terminal) is left alone and the body's value comes back. The watch: with SIGUSR1 watched (a stand-in
