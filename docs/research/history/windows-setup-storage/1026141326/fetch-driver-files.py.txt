import concurrent.futures, hashlib, json, os, struct, urllib.request
from pathlib import Path
os.umask(0o077)
root = Path('/tmp/vmr-windows-storage-1026141326')
notes = Path('/tmp/vmr-windows-storage-notes-1026141326')
source = Path('/tmp/vmr-windows-storage-notes-1026134146')
url = json.loads((source/'drivers.json').read_text())['url']
size = 877373440
expected = {Path(line.split()[1]).name: line.split()[0]
            for line in (source/'driver-files.sha256').read_text().splitlines()}
assert len(expected) == 3

def read(offset, count):
    if not (0 < count <= 1 << 20 and 0 <= offset <= size-count):
        raise ValueError('ISO extent outside bounds')
    request = urllib.request.Request(url, headers={'Range': f'bytes={offset}-{offset+count-1}'})
    with urllib.request.urlopen(request, timeout=20) as response:
        if response.url != url or response.status != 206:
            raise ValueError('Range request refused or redirected')
        if response.headers.get('Content-Range') != f'bytes {offset}-{offset+count-1}/{size}':
            raise ValueError('Unexpected Content-Range')
        data = response.read(count+1)
    if len(data) != count: raise ValueError('Wrong range size')
    return data

pvd = read(16*2048, 2048)
if pvd[:7] != b'\x01CD001\x01': raise ValueError('Not an ISO9660 primary descriptor')
def extent(record):
    if len(record) < 34: raise ValueError('Short directory entry')
    return struct.unpack_from('<I',record,2)[0]*2048, struct.unpack_from('<I',record,10)[0]
current = extent(pvd[156:190])
def entries(location):
    data = read(*location); pos = 0; result = {}
    while pos < len(data):
        length = data[pos]
        if length == 0:
            pos = (pos//2048+1)*2048; continue
        if length < 34 or pos+length > len(data): raise ValueError('Bad directory record')
        record = data[pos:pos+length]; count = record[32]
        if 33+count > length: raise ValueError('Bad name length')
        raw = record[33:33+count]
        if raw not in [b'\0',b'\1']:
            name = raw.decode('ascii').split(';')[0].rstrip('.').upper()
            result[name] = (extent(record), bool(record[25]&2))
        pos += length
    return result
for component in ['VIOSTOR','W11','ARM64']:
    current, directory = entries(current)[component]
    if not directory: raise ValueError('Expected directory')
files = entries(current)
(root/'driver-files').mkdir(mode=0o700)
def fetch(name):
    location, directory = files[name.upper()]
    if directory: raise ValueError('Expected file')
    data = read(*location)
    if hashlib.sha256(data).hexdigest() != expected[name]:
        raise ValueError('Driver file does not match previously verified package')
    (root/'driver-files'/name).write_bytes(data)
    return {'name':name,'bytes':len(data),'sha256':expected[name]}
with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
    records = list(pool.map(fetch,expected))
p = (root/'driver-files/viostor.sys').read_bytes()
pe = struct.unpack_from('<I',p,0x3c)[0]
if p[:2] != b'MZ' or p[pe:pe+4] != b'PE\0\0' or struct.unpack_from('<H',p,pe+4)[0] != 0xaa64:
    raise ValueError('Driver is not ARM64 PE')
(notes/'drivers.json').write_text(json.dumps({'url':url,'method':'bounded HTTP ranges; each file checked against hashes from the previously full-ISO-verified package','files':records},indent=2)+'\n')
(notes/'download.exit').write_text('0\n')
print('Fetched and hash-verified three ARM64 driver files without re-downloading the full ISO')
